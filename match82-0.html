public static void mostWater( Integer[] height ){

        Integer maximumArea=0;

        for(int left= 0; left<height.length-1; left++){

            for(int right= left+1; right<height.length; right++){

                int width= right-left;
                int Tallness;
                if (height[right]>height[left]){
                    Tallness= height[left];
                }
                else{
                    Tallness= height[right];
                }

                int area= width*Tallness;
                if(area>maximumArea){
                    maximumArea= area;
                }

            }

        }

        System.out.println(maximumArea);

    }

public static Integer rowRotation( Integer examWeek, String[][] matrix ){

        Integer counter= 1;

        while(true){

            if (counter == examWeek){
                break;
            }

            for(int columnIndex= 0; columnIndex<matrix[0].length; columnIndex++){

                String tempDown= matrix[0][columnIndex];
                String tempUp= matrix[matrix.length-1][columnIndex];

                for(int rowIndex= 0; rowIndex<matrix.length; rowIndex++){
                    matrix[rowIndex][columnIndex]= tempUp;
                    if(rowIndex==matrix.length-1){
                        break;
                    }
                    else{
                        tempUp= tempDown;
                        tempDown= matrix[rowIndex+1][columnIndex];
                    }
                }

            }

            counter++;

        }

        counter--;

        Integer friendRow= 0;
        String friendSeat= "AA";



        for (int i=0; i<matrix.length; i++){
            for (int j=0; j<matrix[0].length; j++){
                if (matrix[i][j].equals(friendSeat)){
                    friendRow= i;
                    break;
                }
            }
        }

        Arr.print2D(matrix);
        friendRow+=1;

        return friendRow;
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){

        Integer newRow= matrix.length/2;
        Integer newColumn= matrix[0].length/2;
        Integer[][] compressedMatrix= new Integer[newRow][newColumn];

        for(int i=0; i< matrix.length; i++){
            for(int j=0; j< matrix[0].length; j++){


                Integer sum= matrix[i][j] + matrix[i+1][j] + matrix[i][j+1] + matrix[i+1][j+1];
                compressedMatrix[i/2][j/2]= sum;
                j++;
            }
            i++;
        }

        return compressedMatrix;
    }
