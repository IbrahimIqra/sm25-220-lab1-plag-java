public static void mostWater( Integer[] height ){
      int min = 0;
      int max = -1;
      for(int i=0;i<height.length-1;i++)
      {
        for(int j=i+1;j<height.length;j++)
        {
          if(height[i]<height[j])
          {
            min = height[i];
          }
          else
          {
            min = height[j];
          }
            int wid= j-i;
            int q= min*wid;
            if(q>max)
            {
              max=q;
            }
      }
    }
    System.out.print(max);      
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){
       int row = matrix.length;
      int col = matrix[0].length;
      Integer[][] arr = new Integer[row/2][col/2];
for(int i=0;i<row;i++)
      { 
       for(int j=0;j<col;j++)
        { 
         if(i%2==0 && j%2==0)
         {


          int sum = matrix[i][j]+matrix[i+1][j]+matrix[i][j+1]+matrix[i+1][j+1];
         arr[i/2][j/2]= sum;
         }
        }
        
      }
        return arr;
    }

 public static void playGame( Integer[][] arena ){              
      int row = arena.length;
      int col = arena[0].length;
      int count=0;
      for(int i=0;i<row;i++)
      {
        for(int j=0;j<col;j++)
        {
          if(arena[i][j]%50==0 && arena[i][j]!=0)
          {
           if(i+1<row && arena[i+1][j]==2)
           {
             count++;
           }
          if(i-1>=0 && arena[i-1][j]==2)
           {
             count++;
           }
            if(j+1<col && arena[i][j+1]==2)
           {
             count++;
           }
         if(j-1>=0 && arena[i][j-1]==2)
           {
             count++;
           }
          }
        }
      }
      int points = count*2;
      if(points<10)
      {
      System.out.println("Points Gained: "+points+". Your team is out.\n");
      }
      else
      {
       System.out.println("Points Gained: "+points+". Your team has survived the game.");
      }




    }
