#class AssgnTask1{
    public static void mostWater( Integer[] height ){     
      int waterArea = 0;  //val of max area of water 
     // water may overflow if container is slant (question hint)
        int min_h;  //var for height compar.


        for(int i=0;i<height.length;i++){  //o.p start from left of graph
            for(int j=i+1;j<height.length;j++){  //i.p start from +1 of left
                int breadth = j-i;  //for area calc.


                if(height[i]<height[j]){  //compares to find smaller height 
                    min_h = height[i]; 
                }else{
                    min_h = height[j];
                }
           //var. needed for area calc. 
                int area = breadth*min_h;  //ini area
                if(area>waterArea){  //compares of areas
                    waterArea = area;
                }}}                  
        System.out.println(waterArea);
    }

    public static Integer rowRotation( Integer examWeek, String[][] matrix ){
    int rows = matrix.length; 
    int columns = matrix[0].length; 
    int friendSeat = 0;  // var for seat.
    //hint from question: rotation=week-1
    //column stays constant, row shifts for rotations
    //no column(no printing)


    for(int i=0;i<examWeek-1;i++){
            String[] store = matrix[rows-1];  //empty array to store last row 
            for(int j = rows-1;j>0;j--){  
                matrix[j] = matrix[j-1];  // matrix shifts downw
            }
            matrix[0]= store;  //last row goes to first 
        }
    //print entire matrix,rows and cols always needed
    for(int i=0;i<rows;i++){
        for(int j=0;j<columns;j++){
            System.out.print("| "+matrix[i][j]+"  ");
        }


        System.out.println("|");
    }
    //find the friend's posi
    for(int i=0;i<rows;i++){
        for(int j=0;j<columns;j++){
            if(matrix[i][j]=="AA"){
                friendSeat = i + 1;
            }}}
   return friendSeat;
}

#class AssgnTask3{


    public static Integer[][] compressMatrix( Integer[][] matrix ){
    int rows = matrix.length;
    int columns = matrix[0].length;
    //new array = half of the given
    Integer[][] output = new Integer[rows/2][columns/2]; 
    //hint: summation of (i,j),(i+1,j),(i,j+1),(i+1,j+1)
    //increase will be +2 bcs matrix compression is 2x2
    for (int i=0;i<rows;i+=2){ 
        for (int j=0;j<columns;j+=2){
           int sum1 =matrix[i][j];
           int sum2= matrix[i+1][j];
           int sum3 = matrix[i][j+1];
           int sum4 =matrix[i+1][j+1];
           int totalSum = sum1+sum2+sum3+sum4;          
           output[i][j] = totalSum; //since output matrix's half the given
        }}  
    return output;
}

#class AssgnTask4{


    public static void playGame(Integer[][] arena){
    int rows = arena.length;         
    int columns = arena[0].length;      
    int points = 0; //to calc. team points             
    for(int i=0;i<rows;i++){ 
        for(int j=0; j<columns;j++){
            int position = arena[i][j]; //to find position of players
            if(position!=0 && position!=1 && position%50==0){ //finding multiples of 50 and avoid 0 1
            if(j>0 && arena[i][j-1]==2){  //chk left
              points+=2;
            }   
            if(j<columns-1 && arena[i][j+1]==2){  //chk right
              points+=2;
            }     
            if(i>0 && arena[i-1][j]==2){  //chk up
              points+=2;
            }           
            if(i<rows-1 && arena[i+1][j]==2){  //chk down
              points+=2;
            }                    
            }
      }}
    //Checking if team has survived or not
    if(points<10){
        System.out.println("Points Gained: "+points+"." + "Your team is out.");
    }else{
        System.out.println("Points Gained: "+points+"." + "Your team has survived the game.");
    }
    }
