class AssgnTask1{

    public static int mostWater( Integer[] height ){

        
        int n = height.length;
        int max_Area =0;

        for( int i = 0; i < n; i++){
            for ( int j = i+1; j<n; j++){



                int h;
                if (height[i] < height[j]) {
                    h=height[i];
                }
                else {
                    h = height[j];
                }

                int w = j-i;
                int area = h*w;

                if (area > max_Area) {
                    max_Area = area;
                }
            }
        }
        System.out.println(max_Area);
        return max_Area;
    }

public class AssgnTask2{




    public static Integer rowRotation( Integer examWeek, String[][] matrix ){
      int n = matrix.length;
      int k = (examWeek % n)-1;
      int i = 0;
      int j = matrix.length-1;


      
      while(i<j){
        String[] temp =  matrix[i];
        matrix[i] = matrix[j];
        matrix[j] = temp;
        i++;
        j--;
      }

      i = 0;
      j = k-1;

      while(i<j)  {
        String[] temp =  matrix[i];
        matrix[i] = matrix[j];
        matrix[j] = temp;
        i++ ;
        j--;
      }


      i = k+1;
      j = matrix.length-1;
      
      while(i<j){
        String[] temp =  matrix[i];
        matrix[i] = matrix[j];
        matrix[j] = temp;
        i++;
        j--;
      }


      for(int row=0; row<matrix.length; row++) {


        for(int col=0; col<matrix[row].length; col++){

          if(matrix[row][col]=="AA"){
            return row+1;
          }
        }


 }
      
      return -1;
    
    }

import java.util.Arrays;

class AssgnTask3{




    public static Integer[][] compressMatrix( Integer[][] matrix ){



    int row = matrix.length;
    int column = matrix[0].length;
    int r = row / 2;
    int c = column / 2;
    Integer[][] compressed = new Integer[r][c];


    int rw = 0, cl = 0;



    for (int i = 0; i < row; i += 2){
        for (int j = 0; j < column; j += 2){
            int sum = matrix[i][j] + matrix[i + 1][j] + matrix[i][j + 1] + matrix[i + 1][j + 1];
            compressed[rw][cl] = sum;
            cl++;

        }

        rw++;
        cl = 0;
    }



    System.out.println(Arrays.deepToString(compressed));  
    return compressed;

  }
