public static void mostWater( Integer[] height ){
        int leftMost = 0;
        int rightMost = height.length-1;
        int Area = 0;
        while(leftMost < rightMost){
            int H_LeftMost = height[leftMost];
            int H_RightMost = height[rightMost];
            int width = rightMost - leftMost;
            int AreaNow = Math.min(H_LeftMost, H_RightMost)*width;
            Area = Math.max(Area, AreaNow);

            if(H_LeftMost < H_RightMost){
               leftMost++;
            }
            else{
                rightMost--;
            }
        }
         System.out.println(Area);
    }

public static Integer rowRotation(Integer examWeek, String[][] matrix) {
        int totalRotations = examWeek - 1;


        int rows = matrix.length;
        for (int r = 0; r < totalRotations; r++) {
            String[] lastRow = matrix[rows - 1];
            for (int j = rows - 1; j > 0; j--) {
                matrix[j] = matrix[j - 1];
            }
            matrix[0] = lastRow;
        }
        int insideRow = -1;
        boolean found = false;


        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < matrix[i].length; j++) {
                if ("AA".equals(matrix[i][j])) {
                    insideRow = i;
                    found = true;
                    break;
                }
            }
            if(found){
          	break;
           }
        }
        for (int i = 0; i < rows; i++) {
            System.out.print("|  ");
            for (int j = 0; j < matrix[i].length; j++) {
                System.out.print(matrix[i][j] + " | ");
            }
            System.out.println();
        }
        return insideRow+1;
    }

 public static Integer[][] compressMatrix( Integer[][] matrix ){
        int rows = matrix.length;
        int cols = matrix[0].length;
        Integer[][] compMatrix = new Integer[rows/2][cols/2];
        for(int i = 0; i < rows; i+= 2){
            for(int j = 0; j < cols; j+= 2){


                compMatrix[i/2][j/2] = matrix[i][j]+matrix[i][j + 1]+matrix[i + 1][j]+matrix[i + 1][j + 1];
            }
        }
        return compMatrix;
    }

 public static void playGame( Integer[][] arena ){
        int rows = arena.length;


        int cols = arena[0].length;
        int pointsTotal = 0;
        for (int i = 0; i < rows; i++ ){
            for (int j = 0; j < cols; j++ ){
                int position = arena[i][j];
                if(position >= 50 && position%50==0){


                    int pointCount = 0;
                    if(i>0 && arena[i-1][j] == 2){
                        pointCount++;
                    }
                    if(i < rows-1 && arena[i+1][j]==2){
                        pointCount++;
                    }
                    if(j>0 && arena[i][j-1] == 2){
                        pointCount++;
                    }
                    if(j<cols-1 && arena[i][j+1] == 2){
                        pointCount++;
                    }
                    pointsTotal += pointCount;
                }
            }
        }
        int points = pointsTotal*2;
        if (points>= 10) {
            System.out.println("Points Gained: " + points + ". Your team has survived the game.");
        } else {
            System.out.println("Points Gained: " + points + ". Your team is out.");
        }
    }
