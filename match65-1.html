//Assignment Task 01: Container with Most Water
class AssgnTask1{
    


    public static void mostWater( Integer[] height ){
int left =0;
int right= height.length-1;
int maxArea=0;
while(left<right){
    int h;
    if(height[left]<height[right]){
        h=height[left];
    }
else{
    h=height[right];
}
    int w= right-left;
    int area=h*w;
    if (maxArea < area){


        maxArea=area;
    }
    if(height[left]<height[right]){
        left++;
    }
    else{
        right--;
    }
}
System.out.println(maxArea);

    }

//Assignment Task 03: Matrix Compression
class AssgnTask3{

    public static Integer[][] compressMatrix( Integer[][] matrix){
        Integer row=matrix.length;


        Integer col=matrix[0].length;
        Integer[][] result=new Integer[row/2][col/2];
        for(int i=0; i<row; i+=2){
         for(int j=0; j<col; j+=2){
            int sum=matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+matrix[i+1][j+1];
            result[i/2][j/2]=sum;
         } 
        }
        return result;
    }

class AssgnTask4{

    public static void playGame( Integer[][] arena ){
Integer totalpoints=0;
Integer row=arena.length;
Integer col=arena[0].length;
for(int i=0; i<row; i++){
    for(int j=0; j<col; j++){
      Integer value=arena[i][j];


if(value!=0 && value%50==0){
   if( i>0 && arena[i-1][j]==2) {
    totalpoints +=2;
   }
   if(i<row-1 && arena[i+1][j]==2){
totalpoints +=2;
   }
   if(j>0 && arena[i][j-1]==2){
totalpoints +=2;
   }
   if( j<col-1 && arena[i][j+1]==2){
    totalpoints +=2;
   }
}
    }
}
if (totalpoints >= 10){

}
