class AssgnTask1{
    public static void mostWater( Integer[] height ){
      
      int m_area=0;
      int l=0;
      int r=height.length-1;
      for(int i=0;i<height.length;i++){
        for(int j=i+1;j<height.length;j++){
          int min_ht=minHeight(height[i],height[j]);
          int w=j-i;
          int area=min_ht*w;
          m_area=maxArea(m_area,area);
        }
      }
      System.out.print(m_area);
    }
    public static Integer minHeight( Integer left,Integer right){
      
      if(left<right){
        return left;
      }
      else{
        return right;
      }
    }
    public static Integer maxArea( Integer max,Integer cal){
      
      if(max<cal){
        return cal;
      }
      else{
        return max;
      }
    }

class AssgnTask2{


    public static Integer rowRotation( Integer examWeek, String[][] matrix ){
      
      int r=matrix.length;
      int c=matrix[0].length;
      boolean flag=false;
      int find=0;


      for(int i=0;i<examWeek-1;i++){
        String []last=matrix[r-1];
        for(int j=r-1;j>0;j--) {
          matrix[j]=matrix[j-1];
        }
        matrix[0]=last;
      }


      for(int i=0;i<r;i++){
        for(int j=0;j<c;j++){
          if(matrix[i][j].equals("AA")){
            find=i+1;
            flag=true;
            break;
          }
        }
        if(flag){
          break;
        }
      }
      return find;
    }

class AssgnTask3{


    public static Integer[][] compressMatrix( Integer[][] matrix ){
      
      int r=matrix.length;
      int c=matrix[0].length;
      Integer[][] n_mat=new Integer [2][2];
      for(int i=0;i<r;i+=r/2){
        for(int j=0;j<c;j+=c/2){


          int sum=matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+matrix[i+1][j+1];
          n_mat[i/2][j/2]=sum;
        }
      }
      return n_mat;
    }

class AssgnTask4{


    public static void playGame( Integer[][] arena ){
      
      int r=arena.length;
      int c=arena[0].length;
      int total=0;
      for(int i=0;i<r;i++){
        for(int j =0;j<c;j++){
          if(arena[i][j]%50==0 && arena[i][j]!=0) {
            total+=find(arena,i,j);
          }
        }
      }
      if(total<10){
        System.out.println("Points Gained: "+total+". Your team is out.");
      }
      else{
        System.out.println("Points Gained: "+total+". Your team has survived the game.");
      }
    }
    public static int find( Integer[][] arena,int i,int j ){
      int r=arena.length;
      int c=arena[0].length;
      int sum=0;


      if (i-1 >= 0 && arena[i-1][j]==2) {
        sum+=2;       
      }
      if (i+1 < r && arena[i+1][j]==2) {
        sum+=2;   
      }
      if (j-1 >= 0 && arena[i][j-1]==2) {
        sum+=2;      
      }
      if (j+1 < c && arena[i][j+1]==2) {
        sum+=2;       
      }
      return sum;
    }
