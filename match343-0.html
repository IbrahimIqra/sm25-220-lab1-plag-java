//Assignment Task 01: Container with Most Water

    
    public static void mostWater(Integer[] height) {
        int i = 0, j = height.length - 1, res = 0;
        while (i < j) {
            int h = Math.min(height[i], height[j]);


            res = Math.max(res, h * (j - i));
            if (height[i] < height[j]) i++;
            else j--;
        }
        System.out.print(res);
    }

        int rotations = (examWeek - 1) % n;
        for (int r = 0; r < rotations; r++) {
            String[] temp = matrix[n - 1];
            for (int i = n - 1; i > 0; i--) {
                matrix[i] = matrix[i - 1];

//Assignment Task 03: Matrix Compression




    public static Integer[][] compressMatrix(Integer[][] matrix) {
        int rows = matrix.length;
        int cols = matrix[0].length;
        Integer[][] result = new Integer[rows / 2][cols / 2];
        for (int i = 0; i < rows; i += 2) {
            for (int j = 0; j < cols; j += 2) {
                int sum = matrix[i][j] 
                    + matrix[i][j + 1] 
                    + matrix[i + 1][j] 
                    + matrix[i + 1][j + 1];
                result[i / 2][j / 2] = sum;
            }
        }
        return result;
    }

//Assignment Task 04: Game Arena

    public static void playGame(Integer[][] arena) {


        int rows = arena.length, cols = arena[0].length;
        int totalCount = 0;
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if (arena[i][j] > 2) {


                    int cnt = 0;
                    if (i > 0 && arena[i-1][j] == 2) cnt++;
                    if (i < rows - 1 && arena[i+1][j] == 2) cnt++;
                    if (j > 0 && arena[i][j-1] == 2) cnt++;
                    if (j < cols - 1 && arena[i][j+1] == 2) cnt++;
                    totalCount += cnt;
                }
            }
        }


        int points = totalCount * 2;
        if (points >= 10) {
            System.out.print("Points Gained: " + points + ". Your team has survived the game.\n");
        } else {
            System.out.print("Points Gained: " + points + ". Your team is out.\n");
        }
    }
