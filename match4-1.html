﻿//Assignment Task 01: Container with Most Water
class AssgnTask1{




    public static void mostWater( Integer[] height ){


        int answer = 0;
        for(int r = 0; r < height.length; r++){
            for(int c = r + 1; c < height.length; c++){
                if(height[r] <= height[c]){
                    if(height[r] * (c - r) > answer){
                        answer = height[r] * (c - r);
                    }
                }
                else{
                    if(height[c] * (c - r) > answer){
                        answer = height[c] * (c - r);
                    }
                }
            }
        }
        System.out.println(answer);
        return;
    }


    //DO NOT CHANGE ANY DRIVER CODE BELOW THIS LINE
    public static void main(String[] args){
        Integer[] array = {1,8,6,2,5,4,8,3,7};
        System.out.println("Given Array: ");
        Arr.print(array);
        System.out.println("\nExpected Output:");
        System.out.print("49");
        System.out.print("\nYour Output:\n");
        mostWater( array );


    }
}


//Assignment Task 02: Row Rotation Policy
class AssgnTask2{




    public static Integer rowRotation( Integer examWeek, String[][] matrix ){


        int a = -1, b = -1;
        for(int r = 0; r < matrix.length; r++){
            for(int c = 0; c < matrix[0].length; c++){
                if(matrix[r][c] == "AA"){
                    a = r;
                    b = c;
                    break;
                }
            }          
        }
        a += examWeek;
        if(a > matrix.length){
            a -= matrix.length;
        }
        return a;
    }


    //DO NOT CHANGE ANY DRIVER CODE BELOW THIS LINE
    public static void main(String[] args){
        String[][] seatStatus = {
                {"A" , "B"  , "C"  , "D"  , "E"},

//Assignment Task 03: Matrix Compression
class AssgnTask3{




    public static Integer[][] compressMatrix( Integer[][] matrix ){


        Integer [][] answer = new Integer[matrix.length / 2][matrix[0].length / 2];
        for(int r = 0; r < matrix.length - 1; r += 2){
            for(int c = 0; c < matrix[0].length - 1; c += 2){
                answer[r / 2][c / 2] = matrix[r][c] + matrix[r][c + 1] + matrix[r + 1][c] + matrix[r + 1][c + 1];
            }
        }
        return answer;
    }


    //DO NOT CHANGE ANY DRIVER CODE BELOW THIS LINE
    public static void main(String[] args){
        Integer[][] matrix = {
            { 1 , 2 , 3 , 4 },
            { 5 , 6 , 7 , 8 },
            { 1 , 3 , 5 , 2 },
            {-2 , 0 , 6 ,-3 }
        };
        System.out.println("Given Matrix: ");
        Arr.print2D(matrix);
        
        System.out.println("\nExpected Output:");
        System.out.print("| 14 | 22 |\n| 2  | 10 |\n");
        
        System.out.print("\nYour Output:\n");
        Integer[][] returnedArray = compressMatrix( matrix );
        Arr.print2D( returnedArray );
    }
}


//Assignment Task 04: Game Arena
class AssgnTask4{




    public static void playGame( Integer[][] arena ){
        int total = 0;
        for(int r = 0; r < arena.length; r++){
            for(int c = 0; c < arena[0].length; c++){
                if(arena[r][c] >= 50 && arena[r][c] % 50 == 0){
                    if(r > 0 && arena[r - 1][c] == 2){
                        total += arena[r - 1][c];
                    }
                    if(c > 0 && arena[r][c - 1] == 2){
                        total += arena[r][c - 1];
                    }
                    if(r < arena.length - 1 && arena[r + 1][c] == 2){
                        total += arena[r + 1][c];
                    }
                    if(c < arena[0].length - 1 && arena[r][c + 1] == 2){
                        total += arena[r][c + 1];
                    }
                }
            }
        }
        if(total >= 10){
            System.out.println("Points Gained: " + total + ". Your team has survived the game.");
        }
        else{
            System.out.println("Points Gained: " + total + ". Your team is out.");
        }
    }


    //DO NOT CHANGE ANY DRIVER CODE BELOW THIS LINE
    public static void main(String[] args){
        Integer[][] arena = {
            {0,2,2,0},
            {50,1,2,0},
            {2,2,2,0},
            {1,100,2,0}
        };
        System.out.println("Given Arena: ");
        Arr.print2D(arena);
        
        System.out.println("\nExpected Output:");
        System.out.print("Points Gained: 6. Your team is out.\n");
        
        System.out.print("\nYour Output:\n");
        playGame( arena );


        System.out.print("\n======================\n");


        Integer[][] arena1 = {
            {0,2,2,0,2},

            {0,200,2,0,0}
        };
        System.out.println("\nGiven Arena: ");
        Arr.print2D(arena1);
        
        System.out.println("\nExpected Output:");
        System.out.print("Points Gained: 14. Your team has survived the game.\n");
        
        System.out.print("\nYour Output:\n");
        playGame( arena1 );
    }
}
