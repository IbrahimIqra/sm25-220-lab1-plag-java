public static void playGame(Integer[][] arena) {
    int row = arena.length;
    int col = arena[0].length;
    int totalPoints = 0;

int[] dx = {-1, 1, 0, 0};
int[] dy = {0, 0, -1, 1};

for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {

  if (arena[i][j] != 0 && arena[i][j] % 50 == 0) {     //multiple of 50 and not 0

                for (int d = 0; d < 4; d++) {
                    int newRow = i + dx[d];
                    int newCol = j + dy[d];

                    if (newRow >= 0 && newRow < row &&
                        newCol >= 0 && newCol < col &&
                        arena[newRow][newCol] == 2) {

                        totalPoints += 2;
                    }
                }
            }
        }
    }
    if (totalPoints >= 10) {
        System.out.println(totalPoints +" Your team has survived the game.");
    } else {
        System.out.println(totalPoints + " Your team is out.");
