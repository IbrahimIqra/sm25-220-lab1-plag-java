 public static void mostWater( Integer[] height ){
        int h1=height[0];
        int h2=height[height.length-1];
        int width=height.length-1;
        int area=0;
        int temp=0;
        int trav1=1;
        int trav2=height.length;
        int i=0;
        int j=height.length-1;
        
        while(trav1<trav2){
            if(h1<=h2){
                temp=width*h1;
                width--;
                h1=height[++i];
                trav1++;
                if(temp>area){
                    area=temp;
                }
            }
            else{
                temp=width*h2;
                width--;
                h2=height[--j];
                trav2--; 
                if(temp>area){
                    area=temp;}
            }
        }
        System.out.println(area);
    }

public static Integer rowRotation( Integer examWeek, String[][] matrix ){
        int retrieved_rowNumber=0;
        for(int i=1;i<examWeek;i++){
            for( int col=0;col<matrix[0].length;col++){
                String temp=null;
                temp=matrix[matrix.length-1][col];
                for( int row=matrix.length-1;row>0;row--){
                    matrix[row][col]=matrix[row-1][col];
                }
                matrix[0][col]=temp;
            }
        }
        Arr.print2D(matrix);
        for(int row=0;row<matrix.length;row++){
            for(int col=0;col<matrix[0].length;col++){
                if(matrix[row][col].equals("AA")){
                retrieved_rowNumber=row+1;
                break;
                }
            }
            if(retrieved_rowNumber>0){break;}
        }
        return retrieved_rowNumber;
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){
        int row=matrix.length;
        int col=matrix[0].length;
        Integer[][] new_matrix=new Integer[row/2][col/2];
        
        for(int r=0;r<matrix.length;r+=2){
            int sum=0;
            for(int c=0;c<matrix[0].length;c+=2){


                sum=matrix[r][c]+matrix[r+1][c]+matrix[r][c+1]+matrix[r+1][c+1];
                new_matrix[r/2][c/2]=sum;
            }
        }
        return new_matrix;
    }
