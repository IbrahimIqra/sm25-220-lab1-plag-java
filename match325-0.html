public static void mostWater( Integer[] height ){
       
        Integer maximumArea=0;
       Integer leftPoint=0;
       Integer rightPoint=height.length-1;

       while(leftPoint<rightPoint){
            Integer minHeight= Math.min(height[leftPoint],height[rightPoint]);
             Integer width= rightPoint-leftPoint;
             Integer area= minHeight*width;
           if(maximumArea<area){


                maximumArea=area;
           }
           if(height[leftPoint]<height[rightPoint]){
            leftPoint++;
           }
           else{
            rightPoint--;
           }
                
              
            }
            System.out.println(maximumArea);
            

        }

public static Integer[][] compressMatrix( Integer[][] matrix ){

        Integer [][]compressedMatrix= new Integer[2][2];
        Integer rows= matrix.length;
        Integer cols= matrix[0].length;
        Integer sum=0;
        for(int i=0;i<rows-1;i+=2){
           for(int j=0;j<cols-1;j+=2){
                sum=0;


                sum+= matrix[i][j]+matrix[i][j+1] + matrix[i+1][j] + matrix[i+1][j+1];
                compressedMatrix[i/2][j/2]= sum;

            }
        }
        
        return compressedMatrix;
    }

public static void playGame( Integer[][] arena ){

                
        
        int pointsGained=0;
        int rows= arena.length;
        int cols= arena[0].length;


        for(int i=0;i<rows;i++){
            for(int j=0;j<cols;j++){
             if(arena[i][j]%50==0 && arena[i][j]!=0){
                if(i>0 && arena[i-1][j]==2){
                    pointsGained+=2;
                }
                if(i<rows-1 && arena[i+1][j]==2){
                      pointsGained+=2;
                }
                if(j>0 && arena[i][j-1]==2){
                
                 pointsGained+=2;
                }
                if(j<cols-1 && arena[i][j+1]==2){
                     
                    pointsGained+=2;
                }
                
             }
            }

        }
        if(pointsGained>=10){
          
            System.out.println("Points Gained: "+pointsGained+" . Your team has survived the game.");
        }

         System.out.println("Points Gained: "+pointsGained+" . Your team is out.");
