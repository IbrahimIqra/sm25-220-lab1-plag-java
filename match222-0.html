 public static Integer rowRotation( Integer examWeek, String[][] matrix ){
    
    int rows = matrix.length;
    int columns = matrix[0].length;
    
    for (int x = 0; x < examWeek; x++) {
      String[] lastRow = matrix[rows - 1];
      for (int i = rows - 1; i > 0; i--) {
        matrix[i] = matrix[i - 1];
      }
      matrix[0] = lastRow;
    }
    
    Arr.print2D(matrix);
    
    for (int i = 0; i < rows; i++) {
      for (int j = 0; j < columns; j++) {
        if (matrix[i][j].equals("AA")) {
          return i;
        }
      }
    }
    
    return null;
  }

public static void mostWater(Integer[] height) {
    int maximumArea = 0, left = 0, right = height.length - 1;
    
    while (left < right) {
      int width = right - left;
      int minimumHeight = Math.min(height[left], height[right]);
      int area = width * minimumHeight;


      maximumArea = Math.max(maximumArea, area);
      
      if (height[left] < height[right]) {
        left++;
      } else {
        right--;
      }
    }
    
    System.out.println(maximumArea);
  }

 public static Integer[][] compressMatrix( Integer[][] matrix ){
    
    int row = matrix.length;
    int column = matrix[0].length;
    int r = row / 2;
    int c = column / 2;
    Integer[][] compressed = new Integer[r][c];
    
    int rw = 0, cl = 0;


    
    for (int i = 0; i < row; i += 2) {
      for (int j = 0; j < column; j += 2) {
        int sum = matrix[i][j] + matrix[i + 1][j] + matrix[i][j + 1] + matrix[i + 1][j + 1];
        compressed[rw][cl] = sum;
      }
    }
    
    return compressed; 
  }

public static void playGame( Integer[][] arena ){
    int row = arena.length;
    int column = arena[0].length;
    int sum = 0;
    for (int i = 0; i < row; i++){
      for (int j = 0; j < column; j++){
        if (arena[i][j] != 0 && arena[i][j] % 50 == 0){
          if (i > 0 && arena[i - 1][j] == 2){
            sum+=2;
          }
          if (i < row - 1 && arena[i + 1][j] == 2){
            sum+=2;
          }
          if (j > 0 && arena[i][j - 1] == 2){
            sum+=2;
          }
          if (j < column - 1 && arena[i][j + 1] == 2){
            sum+=2;
          }
        }
      }
    }
    if (sum >= 10) {
      System.out.println("Points Gained: " + sum + ". Your team has survived the game.");
    } else {
      System.out.println("Points Gained: " + sum + ". Your team is out.");
    }
  }
