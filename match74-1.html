public static void mostWater(Integer[] height) {
        int left_pillar = 0;
        int right_pillar = height.length - 1;
        int maxArea = 0;

        while (left_pillar < right_pillar) {
            int h = 0;
            int w = 0;
            int area = 0;
            if (height[left_pillar] < height[right_pillar])
                h = height[left_pillar];
            else
                h = height[right_pillar];
            w = right_pillar - left_pillar;


            area = h * w;
            if (area > maxArea) {
                maxArea = area;
            }
            if (height[left_pillar] < height[right_pillar])
                left_pillar++;
            else
                right_pillar--;
        }
        System.out.println(maxArea);
    }

   public static Integer rowRotation(Integer examWeek, String[][] matrix) {

        int examRow = 0;
        int rows = matrix.length;
        int cols = matrix[0].length;



        int shift = examWeek - 1;

        for (int i = 0; i < shift; i++) {
            String[] lastRow = matrix[rows - 1];

            for (int j = rows - 1; j > 0; j--) {
                matrix[j] = matrix[j - 1];
            }
            matrix[0] = lastRow;
        }
        System.out.println("exam week seat status");
        Arr.print2D(matrix);


        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if (matrix[i][j].equals("AA")) {
                    examRow = i + 1;
                }
            }
        }
        return examRow;
    }

    public static Integer[][] compressMatrix(Integer[][] matrix) {
        int rows = matrix.length;
        int cols = matrix[0].length;
        Integer[][] res = new Integer[rows / 2][cols / 2];

        for (int i = 0; i < rows; i += 2) {
            for (int j = 0; j < cols; j += 2) {

                int sum = matrix[i][j] + matrix[i][j + 1] +
                        matrix[i + 1][j] + matrix[i + 1][j + 1];
                res[i / 2][j / 2] = sum;
            }
        }

        return res;
    }

    public static void playGame(Integer[][] arena) {
        int points = 0;

        for (int i = 0; i < arena.length; i++) {
            for (int j = 0; j < arena[0].length; j++) {
                if (arena[i][j] != 0 && arena[i][j] % 50 == 0 && arena[i][j] != 1) {
                    int rowPos = i;
                    int colPos = j;



                    if (rowPos - 1 >= 0 && arena[rowPos - 1][colPos] == 2) {
                        points += 2; // up
                    }
                    if (rowPos + 1 < arena.length && arena[rowPos + 1][colPos] == 2) {


                        points += 2; // down
                    }
                    if (colPos - 1 >= 0 && arena[rowPos][colPos - 1] == 2) {
                        points += 2; // left
                    }
                    if (colPos + 1 < arena[0].length && arena[rowPos][colPos + 1] == 2) {


                        points += 2; // right
                    }
                }
            }
        }

        if (points >= 10) {
            System.out.println("Points Gained: " + points + ". Your team has survived the game.");
        } else {
            System.out.println("Points Gained: " + points + ". Your team is out.");
        }
    }
