    public static void mostWater(Integer[] height){

        int left = 0;
        int right = height.length-1;
        int Area = 0;

        while(left < right){
            int h = Math.min(height[left],height[right]);
            int w = right-left;
            int area = h*w;


            Area = Math.max(Area,area);
    
            if(height[left]<height[right]){
                left++;
            }else{
                right--;
            }
        }
        System.out.println(Area);



}

    public static Integer rowRotation(Integer examWeek,String[][]matrix){
        int row=matrix.length;
        int col=matrix[0].length;

        for(int i=0;i<examWeek-1;i++){
            String[] lastRow = matrix[row - 1];
            
            for(int r = row-1;r>0;r--){
                matrix[r] = matrix[r-1];
            }
            matrix[0] = lastRow;
    }

    for(int i = 0; i < row; i++){
        System.out.print("| ");
        for(int j = 0; j < col; j++){
            System.out.print(matrix[i][j] + " | ");
        }
        System.out.println();
    }

    for(int i = 0; i < row; i++){
        for (int j = 0; j < col; j++){
            if (matrix[i][j].equals("AA")){
                return i + 1;
            }
        }
    }
        return -1;
    }

    public static Integer[][] compressMatrix( Integer[][] matrix ){
        int row = matrix.length;
        int col = matrix[0].length;

    Integer[][] compressed = new Integer[row/2][col/2];

    for(int i = 0; i < row; i += 2){
        for(int j = 0; j < col; j += 2){
            int sum = matrix[i][j] + matrix[i][j+1] +
                      matrix[i+1][j] + matrix[i+1][j+1];
            compressed[i/2][j/2] = sum;
        }
    }

    return compressed;
    }

    public static void playGame( Integer[][] arena ){
        int row = arena.length;
        int col = arena[0].length;
        int Points = 0;

    for (int i = 0;i < row;i++){
    for (int j = 0;j < col;j++){
        if (arena[i][j] % 50 == 0 && arena[i][j] != 0){
            if(i - 1 >= 0){
                if(arena[i - 1][j] == 2){
                    Points += 2;
                }
            }
            if(i + 1 < row){
                if(arena[i + 1][j] == 2){
                    Points += 2;
                }
            }
            if(j - 1 >= 0){
                if(arena[i][j - 1] == 2){
                    Points += 2;
                }
            }
            if(j + 1 < col){


                if(arena[i][j + 1] == 2){
                    Points += 2;
                }
            }
        }
    }
}
    System.out.print("Points Gained: " +Points+ ". ");
    if(Points >= 10){
        System.out.println("Your team has survived the game.");
    }else{
        System.out.println("Your team is out.");
    }

    }
