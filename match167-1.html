class AssgnTask1{


    public static void mostWater( Integer[] height ){

      

       int start_point = 0;
       int end_point = height.length-1;
       int area = 0;

       while(start_point < end_point){

         int hello;

         if(height[start_point]<height[end_point]){
          hello=height[start_point];
         }

         else{
          hello=height[end_point];
         }

         int yello = end_point-start_point;
         int area1 = hello*yello;

         if(area1>area){
          area=area1;
         }

         if(height[start_point]<height[end_point]){
          start_point++;
         }

         else{
          end_point--;
         }

       }



       System.out.println(area);


    }

class AssgnTask2{

   
    public static Integer rowRotation( Integer examWeek, String[][] matrix ){

        int row = matrix.length;
        int column = matrix[0].length;

        for (int a = 0; a < examWeek - 1; a++) {

            String last_row[] = new String[column];

        
            for (int b = 0; b < column; b++) {
                last_row[b] = matrix[row - 1][b];
            }

    
            for (int c = row - 1; c > 0; c--) {

                for (int d = 0; d < column; d++) {
                    matrix[c][d] = matrix[c - 1][d];
                }

            }

        
            for (int i = 0; i < column; i++) {
                matrix[0][i] = last_row[i];
            }

        }

    
        for (int j = 0; j < row; j++) {

            System.out.print("| ");



            for (int p = 0; p < column; p++) {
                System.out.print(matrix[j][p] + " | ");
            }

            System.out.println();

        }

        for (int k = 0; k < row; k++) {

            for (int m = 0; m < column; m++) {

                if (matrix[k][m].equals("AA")) { 
                    return k + 1;
                }

            }
        }

       


        return null;
    }

class AssgnTask3{

        public static Integer[][] compressMatrix( Integer[][] matrix ){

        int row_o = matrix.length;
        int column_o = matrix[0].length;

        int row_n = row_o / 2;
        int column_n = column_o / 2;
        Integer[][] result = new Integer[row_n][column_n];

        int row_c = 0;

        for (int a = 0; a < row_o; a = a + 1 + 1) {

            int column_c = 0;

            for (int b = 0; b < column_o; b = b + 1 + 1) {

                int mat1 = matrix[a][b];
                int mat2 = matrix[a][b + 1];
                int mat3 = matrix[a + 1][b];
                int mat4 = matrix[a + 1][b + 1];
                int mat_sum = mat1 + mat2 + mat3 + mat4;
                result[row_c][column_c] = mat_sum;

                column_c = column_c + 1;
            }



            row_c =row_c + 1;

        }

        return result;
        

    }

class AssgnTask4{

   
    public static void playGame( Integer[][] arena ){

        int pts_a = 0;
        int row_a = arena.length;
        int column_a = arena[0].length;

        for (int a = 0; a < row_a; a++) {

            for (int b = 0; b < column_a; b++) {

                int vl_a = arena[a][b];

                if (vl_a != 0 && vl_a % 50 == 0) {

                    int ct_a = 0;

                    if (a > 0 && arena[a - 1][b] == 2) {
                        ct_a++;
                    }
                    if (a < row_a - 1 && arena[a + 1][b] == 2) {
                        ct_a++;
                    }
                    if (b > 0 && arena[a][b - 1] == 2) {
                        ct_a++;
                    }
                    if (b < column_a - 1 && arena[a][b + 1] == 2) {
                        ct_a++;
                    }

                    pts_a += ct_a * 2;
                }
            }
        }



        System.out.print("Points Gained: " + pts_a + ". ");

        if (pts_a >= 10) {
            System.out.println("Your team has survived the game.");
        }
        else {
            System.out.println("Your team is out.");
        }
    
    }
