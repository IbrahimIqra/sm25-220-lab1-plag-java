public static void main(String[]args){
Scanner sc=new Scanner(System.in);

int a=sc.nextInt();

System.out.println("Give the value of height:");
for(int i=0;i<a;i++){
height[i]=sc.nextInt();
}
int h=0;
int maxwater=0;
int y=0;
int x=height.length-1;
while(y<x){


if(height[y]<height[x]){
h=height[y];
}
else{
h=height[x];
}
int w=x-y;
int area=w*h;
if(area>maxwater){
maxwater=area;
}
if(height[y]<height[x]){
y++;
}
else{
x--;
}
}
}
}

public static int row_rotation(int exam_weak,String[][]seat_status){
int row=seat_status.length;
exam_weak=exam_weak%row;
for( int i=0;i<exam_weak;i++){
String[] lastrow = seat_status[seat_status.length - 1];


for(int j=seat_status.length-1;j>0;j--){
seat_status[j]=seat_status[j-1];
}
seat_status[0]=lastrow;
}
int aarow=-1;
for(int i=0;i<row;i++){
for(int j=0;j<seat_status[0].length;j++){
System.out.print(seat_status[i][j]+ " ");
if(seat_status[i][j].equals("AA")){
aarow=i;
}
}
System.out.println();
}
System.out.println("Your friend AA will be on row: "+aarow);
return aarow;
}
}

import java.util.Scanner;
public class firstlabtask3{


public static int[][] compress_matrix(int[][]arr){
int row=arr.length;
int col=arr[0].length;
int Row=row/2;
int Col=col/2;
int [][] result=new int[Row][Col];
for(int i=0;i<row;i+=2){
for( int j=0;j<col;j+=2){
int sum=arr[i][j]+arr[i+1][j]+arr[i][j+1]+arr[i+1][j+1];


result[i/2][j/2]=sum;
}
}
return result;
}
public static void main(String []args){
Scanner sc=new Scanner(System.in);
System.out.println("Enter the value of row");
int a=sc.nextInt();
System.out.println("Enter the value of column");
int b=sc.nextInt();
if(a%2!=0||b%2!=0){
System.out.println("Rows and columns must be even numbers for compression");
return;
}
int[][] matrix=new int[a][b];
for (int i=0;i<a;i++){
for (int j=0;j<b;j++){
matrix[i][j]=sc.nextInt();
}
}
int[][]compressed=compress_matrix(matrix);
for(int i=0;i<compressed.length;i++){
for( int j=0;j<compressed[i].length;j++){
System.out.print(compressed[i][j]+" ");
}
System.out.println();
}
}
}

import java.util.Scanner;
public class firstlabtask4{
public static void checkalive(int [][]array){
int row=array.length;
int cols = array[0].length;
int totalPoints = 0;


for (int i = 0; i < row; i++) {
for (int j = 0; j < cols; j++) {
if (array[i][j] % 50 == 0 && array[i][j] != 0) {
if ((i - 1) >= 0 && array[i - 1][j] == 2) {
totalPoints += 2;
}
if ((i + 1) < row && array[i + 1][j] == 2) {
totalPoints += 2;
}
if ((j - 1) >= 0 && array[i][j - 1] == 2) {
totalPoints += 2;
}
if ((j + 1) < cols && array[i][j + 1] == 2) {
totalPoints += 2;
}
}
}
}
System.out.print("Points Gained: " + totalPoints + ". ");
if (totalPoints >= 10) {
System.out.println("Your team has survived the game.");
} else {
System.out.println("Your team is out.");
}
}

public static void main(String[] args) {
Scanner sc = new Scanner(System.in);
System.out.print("Enter number of rows: ");
int r = sc.nextInt();
System.out.print("Enter number of columns: ");
int c = sc.nextInt();
int[][] arena = new int[r][c];
System.out.println("Enter matrix values row by row:");
for (int i = 0; i < r; i++) {
for (int j = 0; j < c; j++) {
arena[i][j] = sc.nextInt();
}
}
checkalive(arena);
}
