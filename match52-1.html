//Assignment Task 01: Container with Most Water


public static void mostWater(Integer[] height) {
    int i = 0; //start
    int j = height.length-1; //end
    int maxArea = 0;
    
    while (i < j){
      int length = j-i;
      int minHeight = Math.min(height[i], height[j]);
      int area = length * minHeight;
      
      if (area > maxArea){
        maxArea = area;
      }
      
      if (height[i] <= height[j]){
        i++;
      }
      else{
        j--;
      }
    }
    System.out.println(maxArea);  
  }


//Assignment Task 02: Row Rotation Policy
public static int rowRotation(int examWeek, String[][] matrix){
    int rows = matrix.length;
    int cols = matrix[0].length;
    


    for (int week = 1; week < examWeek; week++){
      String[] temp = new String[cols];  
      for (int j = 0; j < cols; j++){
        temp[j] = matrix[rows-1][j];  //Storing last row in an array
      }
      
      for (int i = rows-1; i > 0; i--){
        for (int j = 0; j < cols; j++){
          matrix[i][j] = matrix[i-1][j];   //Move rows downwards except last row
        }
      }
      
      for (int j = 0; j < cols; j++){
        matrix[0][j] = temp[j];  //Last row shifted to 1st row
      }
    }
    
    for (int i = 0; i < rows; i++){
        System.out.print("| ");
        for (int j = 0; j < cols; j++){
            System.out.print(matrix[i][j] + "  | ");


        }
        System.out.println();
    }
    
    for (int i = 0;i < rows; i++){
      for (int j = 0; j < cols; j++){
        if (matrix[i][j].equals("AA")){
          return i+1;
        }
      }
    }
    return -1 ;
  }


//Assignment Task 03: Matrix Compression


public static Integer[][] compressMatrix( Integer[][] matrix ){
    int rows = matrix.length;
    int cols = matrix[0].length;
    Integer[][] arr = new Integer[rows/2][cols/2];
    int x = 0;
    int y = 0;
    
    for (int i = 0; i < rows; i+=2) {
      for (int j = 0; j < cols; j+=2){
        arr[x][y] = 0;


        arr[x][y] += matrix[i][j] + matrix[i][j+1] + matrix[i+1][j] + matrix[i+1][j+1];  //index 00 = 00 + 01 + 10 + 11 and so on
        y += 1;  //index 01 and 11
      }
      x += 1; 
      y = 0;  //index 10
    }
    return arr;
  }



//Assignment Task 04: Game Arena


public static void playGame( Integer[][] arena ){
    int rows = arena.length;
    int cols = arena[0].length;
    int points = 0;
    
    for (int i = 0; i < rows; i++){
      for (int j = 0; j < cols; j++){
        if (arena[i][j] != null && arena[i][j] % 50 == 0 && arena[i][j] != 0){


          if (i > 0 && arena[i-1][j] != null && arena[i-1][j] == 2){
            points += 2;  //Move up , Find 2
          }
          if (i < rows-1 && arena[i+1][j] != null && arena[i+1][j] == 2){
            points += 2;  //Down
          }


          if (j > 0 && arena[i][j-1] != null && arena[i][j-1] == 2){
            points += 2;  //Left
          }
          if (j < cols-1 && arena[i][j+1] != null && arena[i][j+1] == 2){
            points += 2;  //Right
          }
        }
      }
    }
    
    if (points < 10){
      System.out.println("Points Gained: " + points + ". Your team is out.");
    }
    else{
      System.out.println("Points Gained: " + points + ". Your team has survived the game.");
    }
  }
