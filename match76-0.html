    public static int mostWater(int[] arr) {
        int left = 0;
        int right = arr.length - 1;
        int maxArea = 0;

        while (left < right) {
            int height;
            if(arr[left] < arr[right]) {
                height = arr[left];
            } else {
                height = arr[right];
            }

            int width = right - left;
            int area = height * width;

            if(area > maxArea) {
                maxArea = area;
            }

            if (arr[left] < arr[right]) {
                left++;
            } else {
                right--;
            }
        }

        return maxArea;
    }

    public static void row_rotation(int exam_week, String seat_status[][]) {
        while(exam_week-- > 1) {
            String[] lastRow = seat_status[seat_status.length - 1];
            for (int row = seat_status.length - 1; row > 0; row--) {
                seat_status[row] = seat_status[row - 1];
            }
            seat_status[0] = lastRow;
        }



        for (int i = 0; i < seat_status.length; i++) {
            System.out.print("| ");
            for (int j = 0; j < seat_status[i].length; j++) {
                System.out.print(seat_status[i][j] + " | ");
            }
            System.out.println();
        }

        for (int i = 0; i < seat_status.length; i++) {
            for (int j = 0; j < seat_status[i].length; j++) {
                if (seat_status[i][j].equals("AA")) {
                    System.out.println("Your friend AA will be on row " + (i + 1));
                    return;
                }
            }
        }
    }

    public static void compress_matrix(int[][] arr) {
        int[][] newArr = new int[arr.length / 2][arr[0].length / 2];

        for (int row = 0, i = 0; row < arr.length; row += 2, i++) {
            for (int col = 0, j = 0; col < arr[0].length; col += 2, j++) {


                newArr[i][j] = arr[row][col] + arr[row][col + 1] +
                        arr[row + 1][col] + arr[row + 1][col + 1];
            }
        }

        for (int row = 0; row < newArr.length; row++) {
            for (int col = 0; col < newArr[0].length; col++) {
                System.out.print(newArr[row][col] + " ");
            }
            System.out.println();
        }
    }

    public static void teamSurvives(int[][] arr) {
        int points = 0;

        for(int row = 0; row < arr.length; row++) {
            for(int col = 0; col < arr[0].length; col++) {
                if(arr[row][col] != 0 && arr[row][col] % 50 == 0) {
                    // upp
                    if (row > 0 && arr[row - 1][col] == 2) {
                        points += 2;
                    }
                    // for down
                    if (row < arr.length - 1 && arr[row + 1][col] == 2) {
                        points += 2;
                    }
                    // leftiee
                    if (col > 0 && arr[row][col - 1] == 2) {
                        points += 2;
                    }
                    // rightt
                    if (col < arr[0].length - 1 && arr[row][col + 1] == 2) {
                        points += 2;
                    }
                }
            }
        }
        if(points >= 10) {
            System.out.println("Points Gained: " + points + ". Your team has survived the game.");
        }
        else {
            System.out.println("Points Gained: " + points + ". Your team is out. ");
        }
    }
