﻿//Assignment Task 01: Container with Most Water
class AssgnTask1{
  public static void mostWater(Integer[] height) {
    int left = 0;
    int maxArea = 0;


    int right = height.length - 1;
    


    while (left < right) {    
        int width = right - left;
        int ht;
        if (height[left] < height[right]) {       
          ht = height[left];
        }else {
          ht = height[right];
        }


        int current_container = ht * width;
        if (current_container > maxArea) {
            maxArea = current_container;
        }


        if (height[left] < height[right]) {
            left++;
        } else {
            right--;
        }
    }


    System.out.println(maxArea);
}


// Assignment Task 02: Row Rotation Policy
class AssgnTask2 {


    public static Integer rowRotation(Integer examWeek, String[][] matrix) {
        int totalRows = matrix.length;
        int totalCols = matrix[0].length;
        int shift = (examWeek-1)%totalRows;




        for (int s = 0; s < shift; s++) {
            String[] lastRow = matrix[totalRows - 1];
            for (int i = totalRows - 1; i > 0; i--) {
                matrix[i] = matrix[i - 1];
            }
            matrix[0] = lastRow;
        }


        for (int i = 0; i < totalRows; i++) {
            System.out.print("| ");
            for (int j = 0; j < totalCols; j++) {
                System.out.print(matrix[i][j] + " | ");
            }
            System.out.println();
        }


        for (int i = 0; i < totalRows; i++) {
            for (int j = 0; j < totalCols; j++) {
                if ((matrix[i][j]=="AA")) {
                    return i + 1;
                }
            }
        }


        return -1; 
    }

class AssgnTask3 {




    public static Integer[][] compressMatrix(Integer[][] matrix) {
        int originalRowCount = matrix.length;
        int originalColCount = matrix[0].length;


        int compressedRowCount = originalRowCount / 2;
        int compressedColCount = originalColCount / 2;


        Integer[][] compressedMatrix = new Integer[compressedRowCount][compressedColCount];


        for (int i = 0, compressedRow = 0; i< originalRowCount; i+= 2, compressedRow++) {
            for (int j= 0, compressedCol = 0; j< originalColCount; j += 2, compressedCol++) {
                int sum = matrix[i][j] 
                             + matrix[i + 1][j] 
                             + matrix[i][j + 1] 
                             + matrix[i + 1][j + 1];
                compressedMatrix[compressedRow][compressedCol] = sum;
            }
        }


        return compressedMatrix;
    }




    
//Assignment Task 04: Game Arena
class AssgnTask4{




    public static void playGame(Integer[][] arena) {
        int rows = arena.length;
        int cols = arena[0].length;
        int totalPoints = 0;


        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {


                if (arena[i][j] != null && arena[i][j] != 0 && arena[i][j] % 50 == 0) {
                    
                    if (i > 0 && arena[i - 1][j] == 2) totalPoints += 2;
                    
                    if (i < rows - 1 && arena[i + 1][j] == 2) totalPoints += 2;
                   
                    if (j > 0 && arena[i][j - 1] == 2) totalPoints += 2;
                    
                    if (j < cols - 1 && arena[i][j + 1] == 2) totalPoints += 2;
                }
            }
        }


        System.out.print("Points Gained: " + totalPoints + ". ");
        if (totalPoints >= 10) {
            System.out.println("Your team has survived the game.");
        } else {
            System.out.println("Your team is out.");
        }
    }
