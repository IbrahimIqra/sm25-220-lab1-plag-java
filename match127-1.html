//Assignment Task 01: Container with Most Water
 class AssgnTask1{
  public static int area;
  public static int width;
  public  static int temp1;

public static void mostWater( Integer[] height ){
  for(int i = 0; i < height.length;i++){
    for(int j = i+1;j<height.length;j++){
      width=j-i;
      if(height[i]>height[j]){
        temp1=height[j];
      }
         else{
           temp1=height[i];
         }
         int temparea= width*temp1;
         if(temparea>area){
           
           area=temparea;
         }
         
         }
      
      }
    System.out.println(area);
    }
}

//Assignment Task 03: Matrix Compression
class AssgnTask3{



    public static Integer[][] compressMatrix( Integer[][] matrix ){
      int r1= matrix.length;
      int c1 = matrix[0].length;
      Integer[][]compressed = new Integer[r1/2][c1/2];
      for(int i =0; i<r1;i+=2){
       for(int j =0; j<c1;j+=2){
       int sum = matrix[i][j]+
               matrix[i][j+1]+
            matrix[i+1][j]+
           matrix[i+1][j+1];
       compressed[i/2][j/2]=sum;
      }
     }
      return compressed;
    }

  public static void playGame( Integer[][] arena ){
    int r1= arena.length;
    int c1 = arena[0].length;
    int total=0;
    
    for(int i =0; i <r1;i++){
      for(int j =0; j <c1;j++){
        if(arena[i][j]%50==0 && arena[i][j]!=0){
          if(i-1>=0 && arena[i-1][j]==2){
            total+=2;
          }
          if(i+1<r1&& arena[i+1][j]==2){
            total+=2;
          }
          if(j-1>=0 && arena[i][j-1]==2){
            total+=2;
          }


          if(j+1<c1 && arena[i][j+1]==2){
            total+=2;
          }
          
          
          
        }
        
      }
