public static void mostWater( Integer[] height ){

        int l = 0;
        int r = height.length-1;
        int maxH = 0;
        while (l<r){
            int minH = Math.min(height[l], height[r]);
            int wid = r-l;   
            int area = minH*wid;   
            maxH = Math.max(maxH,area);

        if (height[l] > height[r]) {
                r--;
            }  
            
            else {   
                l++;
            }
        }
        System.out.println(maxH);

    }

public static Integer rowRotation( Integer examWeek, String[][] matrix ){
        int rowN = matrix.length;
        int colN = matrix[0].length;
        int rotation = examWeek-1;


        for (int i=0; i<rotation; i++){
            String[] TempArr = matrix[rowN-1];
            for (int j=rowN-1; j>0; j--){

                matrix[j]=matrix[j-1];
            }
            matrix[0]=TempArr;



        }
        int intRow = 0;

        for (int i = 0; i < rowN; i++) {
            for (int j = 0; j < colN; j++) {
                System.out.print("| " + matrix[i][j] + " ");
                if (matrix[i][j].equals("AA")) {
                    intRow = i;
                }
            }
            System.out.println("|");
        }
        return intRow+1;
    }

    public static Integer[][] compressMatrix( Integer[][] matrix ){
        int rowN = matrix.length;
        int colN = matrix[0].length;
        Integer [][] compMatrix = new Integer[matrix.length/2][matrix[0].length/2];



        for(int i=0; i<rowN; i+=2){
            for (int j=0; j<colN; j+=2){
                int tempSum = matrix[i][j]+matrix[i+1][j]+matrix[i][j+1]+matrix[i+1][j+1];


                compMatrix [i/2][j/2] = tempSum;
            }
        }

        return compMatrix;
    }

public static void playGame( Integer[][] arena ){
        int points = 0;
        for (int i = 0; i < arena.length; i++){


            for(int j = 0; j < arena[0].length; j++){
                if(arena[i][j]%50==0 && arena[i][j]!=0){
                    if(i>0 && arena[i-1][j]==2){
                        points+=2;
                        //System.out.println("up " + arena[i][j] + "=" + points); //for Debug
                    }
                    if(j<(arena[0].length-1) && arena[i][j+1]==2){
                        points+=2;
                        //System.out.println("Right " + arena[i][j] + "=" + points); //for Debug
                    }


                    if(i<(arena.length-1) && arena[i+1][j]==2){
                        points+=2;
                        //System.out.println("Down " + arena[i][j] + "=" + points); //for Debug
                    }
                    if(j>0 && arena[i][j-1]==2){
                        points+=2;
                        //System.out.println("Left " + arena[i][j] + "=" + points); //for Debug
                    }
                }

            }

        }
        if (points < 10){
             System.out.println( "Points Gained: "+points+". Your team is out.");
        }
        else {
             System.out.println( "Points Gained: "+points+". Your team has survived the game.");
        }
    }
