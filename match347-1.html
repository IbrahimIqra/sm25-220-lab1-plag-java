    public static void mostWater(Integer[] height) {
        int maxArea = 0;
        for (int i = 0; i < height.length; i++) {
            for (int j = i + 1; j < height.length; j++) {
                int area = Math.min(height[i], height[j]) * (j - i);
                maxArea = Math.max(maxArea, area);
            }
        }
        System.out.println(maxArea);
    }

public static Integer rowRotation(Integer examWeek, String[][] m) {
    int r = m.length;
    int c = m[0].length;
    int rotations = (examWeek - 1) % r;




    for (int k = 0; k < rotations; k++) {
        for (int i = 0; i < c; i++) {
            String temp = m[r - 1][i];
            for (int j = r - 1; j > 0; j--) {
                m[j][i] = m[j - 1][i];
            }
            m[0][i] = temp;
        }
    }


    for (int i = 0; i < r; i++) {
        System.out.print("| ");
        for (int j = 0; j < c; j++) {
            System.out.print(m[i][j] + " | ");
        }
        System.out.println();
    }


    int x = -1;


    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++) {
            if (m[i][j].equals("AA")) {
                x = i + 1;
                break;
            }
        }
    }


    return x;
}

public static Integer[][] compressMatrix( Integer[][] matrix ){


    int r = matrix.length;
    int c = matrix[0].length;


    Integer[][] compressedMatrix = new Integer[r / 2][c / 2];


    for (int i = 0; i < r; i += 2) {
        for (int j = 0; j < c; j += 2) {


            int sum = matrix[i][j] + matrix[i][j + 1] + matrix[i + 1][j] + matrix[i + 1][j + 1];
            compressedMatrix[i / 2][j / 2] = sum;
        }
    }


    return compressedMatrix;
}

public static void playGame(Integer[][] arena) {
    int r = arena.length;
    int c = arena[0].length;
    int tp = 0;


    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++) {
            if (arena[i][j] % 50 == 0 && arena[i][j] != 0) {
                int p = 0;


                p += (i > 0 && arena[i - 1][j] == 2) ? 2 : 0;
                p += (i < r - 1 && arena[i + 1][j] == 2) ? 2 : 0;
                p += (j > 0 && arena[i][j - 1] == 2) ? 2 : 0;
                p += (j < c - 1 && arena[i][j + 1] == 2) ? 2 : 0;


                tp += p;
            }
        }
    }


    if (tp >= 10) {
        System.out.println("Points Gained: " + tp + ". Your team has survived the game.");
    } else {
        System.out.println("Points Gained: " + tp + ". Your team is out.");
    }
}
