public static void mostWater( Integer[] height ){
   
    int maximum_water=0;
    int temp_max;
    for(int i=0;i<height.length-1;i++){
      for(int j=height.length-1;j>i;j--){
        if(height[i]>height[j]){
          temp_max=height[j]*(j-i);
        }
        else{
          temp_max=height[i]*(j-i);
        }
        if(temp_max>maximum_water){
          maximum_water=temp_max;
        }
      }
    }
    System.out.println(maximum_water);
  }

public static Integer[][] compressMatrix( Integer[][] matrix ){
       int r= matrix.length;
        int c = matrix[0].length;
        int r1 = r / 2;
        int c1 = c / 2;
        Integer[][] arr = new Integer[r1][c1];


        for (int i = 0; i<r1; i++) {
            for (int j = 0; j<c1; j++) {
                arr[i][j] = matrix[i*2][j*2] + matrix[i*2+1][j*2] + matrix[i*2][j*2+1] + matrix[i*2+1][j*2+1];    
            }
        }
        return arr;
    }

public static void playGame( Integer[][] arena ){
int r = arena.length;
        int c = arena[0].length;
        int sum = 0;
        for (int i = 0; i<r; i++) {
            for (int j = 0; j<c; j++) {
                if (arena[i][j]%50==0 && arena[i][j]>0) {
                    if (j>0 && arena[i][j-1]==2) {
                        sum += 2;
                    }
                    if (i>0 && arena[i-1][j]==2) {


                        sum += 2;
                    }


                    if (i<r-1 && arena[i+1][j]==2) {
                        sum += 2;
                    }
                    if (j<c-1 && arena[i][j+1]==2) {
                        sum += 2;
                    }
                }
            }
        }
        System.out.print("Points gained: "+sum);
        if(sum<10){
          System.out.print("Your team is out");
        }
        else{
          System.out.print("Your team has survived the game");
        }
    }
