public static void mostWater( Integer[] height ){

        int leftWall = 0, rightWall = height.length - 1;


        int maxArea = 0;
        while (leftWall < rightWall) {
            int width = rightWall - leftWall;
            int minHeight;
            if (height[leftWall] < height[rightWall]) {
                minHeight = height[leftWall];
            } else {
                minHeight = height[rightWall];
            }
            int area = width * minHeight;
            if (area > maxArea) {
                maxArea = area;
            }
            if (height[leftWall] < height[rightWall]) {
                leftWall++;
            } else {
                rightWall--;
            }
        }
        System.out.println(maxArea);

    }

public static Integer[][] compressMatrix( Integer[][] matrix ){

        //For this task you'll need to create new 2D array
        
        //TO DO

        //remove the line below and return the newly created Compressed 2D matrix
        int rows = matrix.length;
        int cols = matrix[0].length;
        int r = rows / 2;
        int c = cols / 2;
        Integer[][] compressedMatrix = { {0, 0}, {0, 0} };
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if( i < r && j < c ){
                    compressedMatrix[0][0] += matrix[i][j];
                }
                if ( i < r && j >= c ){
                    compressedMatrix[0][1] += matrix[i][j];
                }
                if ( i >= r && j < c ){
                    compressedMatrix[1][0] += matrix[i][j];
                }
                if ( i >= r && j >= c ){
                    compressedMatrix[1][1] += matrix[i][j];
                }
            }
        }
        
        return compressedMatrix;
    }

public static void playGame(Integer[][] arena) {

        // For this task you don't need to create any new 2D array
        // just print the result inside the function

        // TO DO

        int rows = arena.length;
        int cols = arena[0].length;
        int points = 0;
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {


                if (arena[i][j] % 50 == 0 && arena[i][j] != 0) {
                    if (i > 0 && arena[i - 1][j] == 2) {
                        points+=2;
                    }
                    if (i < rows - 1 && arena[i + 1][j] == 2) {
                        points+=2;
                    }
                    if (j > 0 && arena[i][j - 1] == 2) {
                        points+=2;
                    }
                    if (j < cols - 1 && arena[i][j + 1] == 2) {
                        points+=2;
                    }
                }
            }
        }
        System.out.print("Points Gained: " + points + ". ");
        if (points >= 10) {
            System.out.print("Your team has survived the game.");
        } 
        else {
            System.out.print("Your team is out.");
        }
    }
