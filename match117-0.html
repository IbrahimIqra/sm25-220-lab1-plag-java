class AssgnTask1 {


    public static int mostWater(Integer[] height) {
        int h = height.length;
        int maxArea = 0, left = 0, right = h - 1;
        
        while (left < right) {
            maxArea = Math.max(maxArea, (right - left) * Math.min(height[left], height[right]));
            
            if (height[left] < height[right]) {
                left++;
            } else {
                right--; 
            }
        }


        return maxArea;
    }

class AssgnTask2 {


    
    public static Integer rowRotation(Integer examWeek, String[][] matrix) {


        int n = matrix.length;
        int k = (examWeek - 1) % n;


     


        int i = 0, j = n - 1;
        while (i < j) {
            String[] temp = matrix[i];
            matrix[i] = matrix[j];
            matrix[j] = temp;
            i++;
            j--;
        }


        i = 0;
        j = k - 1;
        while (i < j) {
            String[] temp = matrix[i];
            matrix[i] = matrix[j];
            matrix[j] = temp;
            i++;
            j--;
        }


        i = k;


        j = n - 1;
        while (i < j) {
            String[] temp = matrix[i];
            matrix[i] = matrix[j];
            matrix[j] = temp;
            i++;
            j--;
        }


        for (int row = 0; row < matrix.length; row++) {
            System.out.print("| ");
            for (int col = 0; col < matrix[row].length; col++) {
                System.out.print(matrix[row][col] + " | ");
            }
            System.out.println();
        }


        for (int row = 0; row < matrix.length; row++) {
            for (int col = 0; col < matrix[row].length; col++) {
                if (matrix[row][col].equals("AA")) {
                    return row + 1; // Return 1-based row number
                }
            }
        }


        return -1; 
    }

class AssgnTask3{


   


   public static Integer[][] compressMatrix( Integer[][] matrix ){


 
    int row = matrix.length;
    int col = matrix[0].length;
    int r = row / 2;
    int c = col / 2;


    Integer[][] compressed = new Integer[r][c];




    for (int i = 0; i < row; i += 2) {
        for (int j = 0; j < col; j += 2) {
            int sum = matrix[i][j] + matrix[i + 1][j] + matrix[i][j + 1] + matrix[i + 1][j + 1];
            compressed[i / 2][j / 2] = sum;
        }
    }


    return compressed; 
}

class AssgnTask4{


    


    public static void playGame( Integer[][] arena ){


    int row = arena.length;
    int col = arena[0].length;
    int sum = 0;


    for (int i = 0; i < row; i++) {
      for (int j = 0; j < col; j++) {
        if (arena[i][j] != 0 && arena[i][j] % 50 == 0) {
          if (i > 0 && arena[i - 1][j] == 2) {
            sum++; 
          }
          if (i < row - 1 && arena[i + 1][j] == 2) {
            sum++;
          }
          if (j > 0 && arena[i][j - 1] == 2) {
            sum++;
          }
          if (j < col - 1 && arena[i][j + 1] == 2) {
            sum++;
          }
        }
      }
    }


    int totalPoints = sum * 2; 




    if (totalPoints >= 10) {
      System.out.println("Points Gained: " + totalPoints + ". Your team has survived the game.");
    } else {
      System.out.println("Points Gained: " + totalPoints + ". Your team is out.");
    }
  }
