public static void mostWater(Integer[] arr) {
    int maximum = 0;
    for (int i = 0; i < arr.length; i++) {
        for (int j=i + 1; j<arr.length; j++) {
          
            int y=0;
            


            if (arr[i] < arr[j]) {
                y = arr[i];
            } else {
                y = arr[j];
            }
            int s = j - i;
            
            int Full_area = y * s;
            
            if (Full_area > maximum) {
                maximum = Full_area;
            }
        }
    }
    System.out.println(maximum);
}

public static int[][] compressArr(int[][] arr) {
    int r = arr.length;
    int c = arr[0].length;
    int nRow = r / 2;
    int nCol = c / 2;
    int[][] cmp = new int[nRow][nCol];

    for (int x = 0; x < r; x = x + 2) {
        for (int y = 0; y < c; y = y + 2) {
            int total = 0;
            
            total += arr[x][y];
            total += arr[x][y + 1];
            total += arr[x + 1][y];
            total += arr[x + 1][y]; 
            
            cmp[x / nRow][y / nCol] = total;
        }
    }
    return cmp;
}

public static Integer rowRotation(Integer examWeek, String[][] seatStatus) {
   
      int rowss = seatStatus.length;
      
    int colss = seatStatus[0].length;

    
    int numRotate = (examWeek-1)%rowss;

    
    for (int rotate = 0; rotate < numRotate; rotate++) {
        String[] lastR = seatStatus[rowss - 1];
        
        for (int r = rowss - 1; r > 0; r--) {
            seatStatus[r] = seatStatus[r - 1];
        }
        seatStatus[0] = lastR;
    }

    
    for (int r = 0; r < rowss; r++) {
        System.out.print(" | ");
        for (int c = 0; c < colss; c++) {
            System.out.print(seatStatus[r][c] + " | ");
        }
        System.out.println();
    }

    
    int DoubleA_Row = -1;
    for (int r = 0; r < rowss; r++) {
        for (int c = 0; c < colss; c++) {
            if ("AA".equals(seatStatus[r][c])) {
                DoubleA_Row = r + 1; 
                break;
            }
        }
        if (DoubleA_Row != -1) 
          break;
    }

    return DoubleA_Row;
}

public static String playGame(int[][] arena) {
    int row1 = arena.length;
    int col1 = arena[0].length;
    int total = 0;

    for (int r = 0; r < row1; r++) {
        for (int c = 0; c < col1; c++) {
            if (arena[r][c] != 0 && arena[r][c] % 50 == 0) {
                
                if (r - 1 >= 0 && arena[r - 1][c] == 2) {
                    total = total + 1;
                }
                


                if (c - 1 >= 0 && arena[r][c - 1] == 2) {
                    total++;
                }
                
                if (c + 1 < col1 && arena[r][c + 1] == 2) {
                    total++;
                }
                
                if (r + 1 < row1 && arena[r + 1][c] == 2) {
                    total++;
                }
            }
        }
    }

    int finalGrade = total * 2;
    if (finalGrade < 10) {
        return "Points Gained: " + finalGrade + ". Your team is out.";
    } else {
        return "Points Gained: " + finalGrade + ". Your team has survived the game.";
    }
}
