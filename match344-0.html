public static void mostWater(Integer[] height){


        int left = 0;
        int right = height.length - 1;
        int maxArea = 0;


        while(left < right){
            int width = right - left;
            int h;


            if(height[left] < height[right]){
                h = height[left];
                left++; 
            } else {
                h = height[right];
                right--; 
            }


            int area = width * h;


            if(area > maxArea){
                maxArea = area;
            }
        }


        
        System.out.println(maxArea);
    }

public static Integer rowRotation(Integer examWeek, String[][] matrix) {


    int totalRows = matrix.length;
    


    int rotations = (examWeek - 1) % totalRows;


    for (int i = 0; i < rotations; i++) {
        
        String[] lastRow = matrix[totalRows - 1];


        
        for (int r = totalRows - 1; r > 0; r--) {
            matrix[r] = matrix[r - 1];
        }


        
        matrix[0] = lastRow;
    }


    
    Arr.print2D(matrix);


    


    for (int i = 0; i < totalRows; i++) {
        for (int j = 0; j < matrix[i].length; j++) {
            if (matrix[i][j].equals("AA")) {
                return i + 1;  
            }
        }
    }


    return -1;  
}

public static Integer[][] compressMatrix( Integer[][] matrix ) {


    
    int originalRows = matrix.length;
    int originalCols = matrix[0].length;


    
    int newRows = originalRows / 2;
    int newCols = originalCols / 2;


    
    Integer[][] compressed = new Integer[newRows][newCols];


    
    for (int i = 0; i < newRows; i++) {
        for (int j = 0; j < newCols; j++) {
            
            int row = i * 2;
            int col = j * 2;


            
            int sum = matrix[row][col] + matrix[row][col + 1] + matrix[row + 1][col] + matrix[row + 1][col + 1];


            
            compressed[i][j] = sum;
        }
    }


    
    return compressed;
}

public static void playGame(Integer[][] arena) {

    int totalPoints = 0;


    int rows = arena.length;
    int cols = arena[0].length;


    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {


            
            if (arena[i][j] % 50 == 0 && arena[i][j] != 0) {


                
                if (i > 0 && arena[i - 1][j] == 2) {
                    totalPoints += 2;
                }


                
                if (i < rows - 1 && arena[i + 1][j] == 2) {
                    totalPoints += 2;
                }


                
                if (j > 0 && arena[i][j - 1] == 2) {
                    totalPoints += 2;
                }


                
                if (j < cols - 1 && arena[i][j + 1] == 2) {
                    totalPoints += 2;
                }
            }
        }
    }


    


    if (totalPoints >= 10) {
        System.out.println("Points Gained: " + totalPoints + ". Your team has survived the game.");
    } else {
        System.out.println("Points Gained: " + totalPoints + ". Your team is out.");
    }
}
