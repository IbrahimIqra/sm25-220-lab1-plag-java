public static void mostWater( Integer[] height ){

        int area = 0;

        for(int i=0; i<height.length; i++){
            for(int j=i+1; j<height.length; j++){
                int min_height = 0;
                int width = j-i;
                int temp_area = 0;
                if(height[i]>height[j]){
                    min_height = height[j];
                }
                else{
                    min_height = height[i];
                }
                temp_area = min_height*width;
                if(temp_area>area){
                    area = temp_area;
                }
            }
        }
        System.out.println(area);

    }

 public static Integer rowRotation( Integer examWeek, String[][] matrix ){

        int row = matrix.length;
        int col = matrix[0].length;



        for(int i=1; i<=examWeek; i++){

            String [] temp = matrix[row-1];

            for(int j=row-1; j>0; j--){
                matrix[j] = matrix[j-1];
            }
            matrix[0] = temp ;

        }



        for(Integer i=0; i<row; i++){
            for(int j=0;j<col; j++){
                if(matrix[i][j].equals("AA")){
                    return i+1;
                }
            }
        }
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){

    Integer [][] new_arr = new Integer [2][2];
    int row = matrix.length;


    int col = matrix[0].length;

    
    for(int i=0;i<row;i+=2){
        for(int j=0; j<col; j+=2){
            int temp = matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+matrix[i+1][j+1];
            new_arr[i/2][j/2] = temp;
        }
    }
    return new_arr;

    }

public static void playGame( Integer[][] arena ){

        int sum = 0;
        int row = arena.length;
        int col = arena[0].length;

        for(int i=0;i<row;i++){
            for(int j=0; j<col; j++){
                if(arena[i][j]%50==0 && arena[i][j]!=0){
                    if(j>0 && arena[i][j-1]==2){


                        sum+=2;
                    }
                    if(j<col-1 && arena[i][j+1]==2){
                        sum+=2;
                    }
                    if(i>0 && arena[i-1][j]==2){


                        sum+=2;
                    }
                    if(i<row-1 && arena[i+1][j]==2){
                        sum+=2;
                    }
                }
            }
        }
        if(sum>=10){
        System.out.print("Points Gained: "+sum);
        System.out.print(". Your team has survived the game");
        }
        else{
            System.out.print("Points Gained: "+sum);
        System.out.print(". Your team is out");
        }
    }
