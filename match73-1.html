public static void mostWater( Integer[] height ){
int lt,maxA; 
     maxA = 0;
     lt = 0;
    int rt = height.length - 1;




    while (lt < rt) {
         int currentH;
       
        if (height[lt] < height[rt]) {
            currentH = height[lt];
        } else {
            currentH = height[rt];
        }


        int currentWid = rt - lt;
        int currentArr = currentH * currentWid;


         if (currentArr > maxA) {
                maxA = currentArr;
            }


            if (height[lt] < height[rt]) {
                lt += 1;
            } else {
                rt -= 1;
            }
        }
    System.out.println(maxA);
}

 public static Integer[][] compressMatrix( Integer[][] matrix ){


    int row = matrix.length;
    int colmn = matrix[0].length;


    Integer[][] temp = new Integer[row / 2][colmn / 2];


for (int a = 0; a < row; a = a + 2) {


 for (int b = 0; b < colmn; b = b + 2) {
int w = matrix[a][b];
int x = matrix[a][b + 1];
int y = matrix[a + 1][b];
 int z = matrix[a + 1][b + 1];


 int totl = w + x + y + z;


temp[a / 2][b / 2] = totl;
}
 }


 return temp;
}
