public static void mostWater( Integer[] height ){

      int Maxwater=0;
      int leftP=0;
      int rightP=height.length-1;
      
      while(leftP<rightP){
        int width=rightP-leftP;


        int ht=Math.min(height[leftP],height[rightP]);
        int water=width*ht;
        
        if(water>Maxwater){
          Maxwater=water;
        }
        
        if(height[leftP]<height[rightP]){
          leftP++;
        }else{
          rightP--;
        }
      }
      System.out.println(Maxwater);
}

public static Integer rowRotation( Integer examWeek, String[][] matrix ){
      
   int rows=matrix.length;
   int cols=matrix[0].length;
   int shift=examWeek-1;
   
   for(int i=0;i<shift;i++){
     String[] temp=matrix[rows-1];
     for(int j=rows-1;j>0;j--){
       matrix[j]=matrix[j-1];
     }
     matrix[0]=temp;
   }
   for(int i=0;i<rows;i++){
     for(int j=0;j<cols;j++){
       System.out.print(" | "+ matrix[i][j] );
     }
     System.out.println(" | ");
   }
   int a=0;


   for(int i=0;i<rows;i++){
     for(int j=0;j<cols;j++){
       if(matrix[i][j].equals("AA")){
         a=i+1;
         break;
       }
     }
   }
   return a;
  
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){
  
      Integer[][] CM=new Integer[2][2];
      for(int i=0;i<matrix.length;i+=2){
        for(int j=0;j<matrix[0].length;j+=2){


          CM[i/2][j/2]=matrix[i][j]+matrix[i+1][j]+matrix[i][j+1]+matrix[i+1][j+1];
        }
      }
      return CM;
    }

public static void playGame( Integer[][] arena ){

    int row=0;
    int col=0;


    int count=0;
    
    for(int i=0;i<arena.length;i++){
    for(int j=0;j<arena[0].length;j++){
          if(arena[i][j]%50==0 && arena[i][j]!=0){
          row=i;
          col=j;
          
          if(row-1>=0 && arena[row-1][col]==2){
            count++;
          }
           if(row+1<arena.length && arena[row+1][col]==2){
            count++;
          }
           if(col-1>=0 && arena[row][col-1]==2){
            count++;
          }
            if(col+1<arena.length && arena[row][col+2]==2){
            count++;
          }
          }
    }
    }
    int score=count*2;
    


    if(score>=10){
      System.out.println("Points gained: " + score +". Your team has survived the game. ");
    }
    else{
      System.out.println("Points gained: " + score +". Your team is out. ");
    }
    }
