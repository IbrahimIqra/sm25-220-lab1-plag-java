public static void mostWater( Integer[] height ){


    int L = 0;
    int R = height.length - 1;
    int MA = 0;




    while(L < R) {
        int H;
        if (height[L] < height[R]) {
            H = height[L];
        } else {
            H = height[R];
        }
        int W = R - L;
        int A = H * W;


        if (A > MA) {
            MA = A;
        }
        if(height[L] < height[R]) {
            L++;
        } else {
            R--;
        }
    }


    System.out.print(MA);
}

public static Integer[][] compressMatrix( Integer[][] matrix ){


    int R1 = matrix.length;
    int C1 = matrix[0].length;


    int R2 = R1 / 2;
    int C2 = C1 / 2;


    Integer[][] Comp = new Integer[R2][C2];


    for(int i = 0; i < R1; i += 2) {
        for(int j = 0; j < C1; j += 2) {
            int sum = matrix[i][j] + matrix[i][j+1]
                    + matrix[i+1][j] + matrix[i+1][j+1];
            Comp[i/2][j/2] = sum;
        }
    }


    return Comp;
}
