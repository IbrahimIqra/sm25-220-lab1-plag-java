public static void mostWater( Integer[] height ){


   int n=height.length;
   int[] a=new int[n];
   int[] p=new int[n];


   for(int i=0;i<n;i++){
       a[i]=height[i];
       p[i]=i;
   }


   for(int i=0;i<n-1;i++){
       for(int j=i+1;j<n;j++){
           if(a[i]<a[j]){
               int t=a[i]; a[i]=a[j]; a[j]=t;
               t=p[i]; p[i]=p[j]; p[j]=t;
           }
       }
   }


   int l=p[0], r=p[0], ans=0;


   for(int i=1;i<n;i++){
       if(p[i]<l) l=p[i];
       if(p[i]>r) r=p[i];


       int d1=p[i]-l;
       if(d1<0) d1=-d1;


       int d2=p[i]-r;
       if(d2<0) d2=-d2;


       int w=d1;
       if(d2>w) w=d2;


       int area=a[i]*w;
       if(area>ans) ans=area;
   }


   System.out.print(ans);
}

public static Integer rowRotation(Integer examWeek,String[][] matrix){
   int r=matrix.length;
   int c=matrix[0].length;
   int move=(examWeek+1)%r;


   for(int t=0;t<move;t++){
       String[] first=matrix[0];
       for(int i=0;i<r-1;i++){
           matrix[i]=matrix[i+1];
       }


       matrix[r-1]=first;
   }


   for(int i=0;i<r;i++){
       System.out.print("| ");
       for(int j=0;j<c;j++){


           System.out.print(matrix[i][j]+" | ");
       }
       System.out.println();
   }


   for(int i=0;i<r;i++){
       for(int j=0;j<c;j++){
           if(matrix[i][j].equals("AA")){
               return i+1;
           }
       }
   }


   return -1;
}

public static Integer[][] compressMatrix(Integer[][] matrix){


   int r=matrix.length,c=matrix[0].length;
   Integer[][] res=new Integer[r/2][c/2];
   for(int i=0;i<r;i+=2){
       for(int j=0;j<c;j+=2){
           int s=matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+matrix[i+1][j+1];
           res[i/2][j/2]=s;
       }
   }
   return res;
}

public static void playGame( Integer[][] arena ){


   int row=arena.length;
   int col=arena[0].length;
   int total=0;


   for(int i=0;i<row;i++){
       for(int j=0;j<col;j++){
           if(arena[i][j]%50==0 && arena[i][j]!=0){
               int pts=0;


               if(i-1>=0 && arena[i-1][j]==2) pts++;


               if(i+1<row && arena[i+1][j]==2) pts++;


               if(j-1>=0 && arena[i][j-1]==2) pts++;


               if(j+1<col && arena[i][j+1]==2) pts++;
               total+=pts*2;
           }
       }
   }


   System.out.print("Points Gained: "+total+". ");
   if(total>=10){
       System.out.println("Your team has survived the game.");
   }else{
       System.out.println("Your team is out.");
   }
}
