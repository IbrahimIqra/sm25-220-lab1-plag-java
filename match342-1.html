public static void mostWater(Integer[] height){
    int right = height.length - 1;
    int left = 0;
    int maximum = 0;
    while(left < right){
      int h = 0;


      if (height[left] < height[right]){
        h = height[left];
      } 
      else{
        h = height[right];
      }
      int width = right - left;
      int area = h * width;
      if(area > maximum){
        maximum = area;
      }
      if(height[left] < height[right]){
        left += 1;
      } 
      else{
        right -= 1;
      }
    }    
    System.out.println(maximum);
  }

public static Integer rowRotation( Integer examWeek, String[][] matrix ){
    int row = matrix.length;


    int rotate = examWeek % row;
    for(int i = 0; i < rotate; i++) {
      String [] temp = matrix[row - 1];
      for (int j = row - 1; j > 0; j--) {
        matrix[j] = matrix[j - 1];
      }
      matrix[0] = temp;
    }
    for (int i = 0; i < row; i++) {
      System.out.print("| ");
      for (int j = 0; j < matrix[i].length; j++) {
        System.out.print(matrix[i][j] + "  | ");
      }
      System.out.println();
    }
    for (int i = 0; i < row; i++) {
      for (int j = 0; j < matrix[i].length; j++) {
        if (matrix[i][j].equals("AA")) {
          return i + 1;
        }
      }
    }
    return -1;
  }

public static Integer[][] compressMatrix( Integer[][] matrix ){
    int row = matrix.length;
    int col = matrix[0].length;
    int new_row = row / 2;
    int new_col = col / 2;


    Integer[][] compress = new Integer[new_row][new_col];
    for(int i = 0; i < row; i += 2){
      for(int j = 0; j < col; j += 2){


        int sum = matrix[i][j] + matrix[i][j+1] + matrix[i+1][j] + matrix[i+1][j+1];
        compress[i / 2][j / 2] = sum;
      }
    }
    return compress;
  }

public static void playGame( Integer[][] arena ){
    int col = arena[0].length;
    int row = arena.length;
    int point = 0;


    for(int i = 0; i < row; i++){
      for(int j = 0; j < col; j++){
        if(arena[i][j] % 50 == 0 && arena[i][j] != 0){
          if(j + 1 < col && arena[i][j + 1] == 2){
            point += 2;
          }
          if(j - 1 >= 0 && arena[i][j - 1] == 2){
            point += 2;
          }


          if(i - 1 >= 0 && arena[i - 1][j] == 2){
            point += 2;
          }
          if(i + 1 < row && arena[i + 1][j] == 2){
            point += 2;
          }
        }
      }
    }
    if(point >= 10){
      System.out.println("Points Gained: " + point + ". Your team has survived the game.");
    } 
    else{
      System.out.println("Points Gained: " + point + ". Your team is out.");
    }
  }
