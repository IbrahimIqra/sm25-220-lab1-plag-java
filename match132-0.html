public class AssgnTask1 {
    public static int calculateMaxWaterVolume(Integer[] height) {
        int maxVolume = 0;
        for (int i = 0; i < height.length; i++) {
            for (int j = i + 1; j < height.length; j++) {
                int width = j - i;




                int waterLevel;
                if (height[i] < height[j]) {
                    waterLevel = height[i];
                } else {
                    waterLevel = height[j];
                }


                int volume = width * waterLevel;


                if (volume > maxVolume) {
                    maxVolume = volume;
                }
            }
        }


        return maxVolume;
    }

class AssgnTask3{


    public static Integer[][] compressMatrix(Integer[][] mat) {
        int row = mat.length;
        int col = mat[0].length;
        int r1 = row / 2;
        int c1 = col / 2;




        Integer[][] finalArr = new Integer[r1][c1];


        for (int i = 0; i < row; i += 2) {
            for (int j = 0; j < col; j += 2) {
                int sum = mat[i][j] + mat[i+1][j] + mat[i][j+1] + mat[i+1][j+1];


                finalArr[i / 2][j / 2] = sum;
            }
        }
        return finalArr;
}

class AssgnTask4{


public static void playGame(Integer[][] arena) {
    int row = arena.length;
    int col = arena[0].length;
    int counter = 0;


    for (int i = 0; i < row; i++) {
        for (int j = 0; j < col; j++) {
            if (arena[i][j] % 50 == 0 && arena[i][j] != 0 && arena[i][j] != 1) {
                if (i > 0 && arena[i - 1][j] == 2) {
                    counter += 2;
                }
                if (i < row - 1 && arena[i + 1][j] == 2) {
                    counter += 2;
                }
                if (j > 0 && arena[i][j - 1] == 2) {
                    counter += 2;
                }


                if (j < col - 1 && arena[i][j + 1] == 2) {
                    counter += 2;
                }
            }
        }
    }


    if (counter >= 10) {
        System.out.println("Points Gained: " + counter + ". Your team has survived the game.");
    } else {
        System.out.println("Points Gained: " + counter + ". Your team is out.");
    }
}
