//Task1
 public static void mostWater( Integer[] height ){
        int length = height.length, area = 0, maxArea = 0;

        for(int i = 0; i < length; i++){
            for(int j = i + 1; j < length; j++){
                int cHeight;
                int width = j - i;
                if(height[i]< height[j]){
                    cHeight = height[i];
                }
                else{
                    cHeight = height[j]; 
                }
                
                area = cHeight * width;

                if(area > maxArea){
                    maxArea = area;
                }
            }

        }
        System.out.println(maxArea);
    }

 public static Integer rowRotation( Integer examWeek, String[][] matrix ){
        int row = matrix.length, column = matrix[0].length, move = examWeek - 1 , checker = 0; 
        String friend = "AA";



        for(int i = 0; i < move; i++){
            String[] last = matrix[row - 1];

            for(int r = row - 1; r > 0; r--){
                matrix[r] = matrix[r - 1];
            }
            matrix[0] = last;
        }
        String[][] temp = matrix;
        for(int r =0; r < row; r++){
            for(int c = 0; c < column; c++){
                System.out.print(temp[r][c] + " ");
                if (temp[r][c].equals(friend)) {
                    checker += r + 1;
                }
            }
            System.out.println();

        }

       
        return checker;
    }

 public static Integer[][] compressMatrix( Integer[][] matrix ){


        int row = matrix.length, column = matrix[0].length, sum = 0;
        Integer[][] compArr = new Integer[row/2][column/2];
        
        for(int r = 0; r < row; r+=2){


            for(int c = 0; c < column; c+=2){
                sum= 0;
                sum += matrix[r][c] + matrix[r+1][c] + matrix[r][c+1] + matrix[r+1][c+1];
                compArr[r/2][c/2] = sum; 
            }
        }
        return compArr;
    }

 public static void playGame( Integer[][] arena ){


         int row = arena.length, column = arena[0].length, count = 0;
        
        for(int r = 0; r < row; r++){
            for(int c = 0; c < column; c++){
                if(arena[r][c] % 50 == 0 && arena[r][c] > 0){
                    if(r - 1 >=0 && arena[r - 1][c] == 2 ){
                        count++;
                    }
                    if(c - 1 >=0 && arena[r][c - 1] == 2 ){
                        count++;
                    }
                    if(r + 1 < row && arena[r + 1][c] == 2 ){
                        count++;
                    }
                    if(c + 1 < column && arena[r][c + 1] == 2 ){
                        count++;
                    }
                }
            }
        }


        count *= 2;
        if(count < 10){
            System.out.println("Points gained: "+count+". Your team is out");
        }
        else{
            System.out.println("Points gained: "+count+". Your team has survived the game");
        }
    }
