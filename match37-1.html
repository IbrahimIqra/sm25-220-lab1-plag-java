class AssgnTask1{


    public static void mostWater(Integer[] height) {
    int L = 0;
    int R = height.length - 1;
    int maxArea = 0;

    while (L < R) {
        int minHeight = Math.min(height[L], height[R]);
        int width = R - L;
        int area = minHeight * width;


        maxArea = Math.max(maxArea, area);

        if (height[L] < height[R]) {
            L++;
        } else {
            R--;
        }
    }

    System.out.println(maxArea);
}

class AssgnTask3{


    public static Integer[][] compressMatrix( Integer[][] matrix ){

        //For this task you'll need to create new 2D array
        int rows=matrix.length;
        int cols=matrix[0].length;
        int comprows=rows/2;
        int compcols=cols/2;
        Integer[][] compressed=new Integer[comprows][compcols];

        for(int i=0;i<comprows;i++){
            for(int j=0;j<compcols;j++){
                int sum=0;
                sum+=matrix[2*i][2*j];
                sum+=matrix[2*i][2*j+1];
                sum+=matrix[2*i+1][2*j];
                sum+=matrix[2*i+1][2*j+1];
                compressed[i][j] = sum;
            }
        }

        return compressed;
    }

    }
}

class AssgnTask4{


    public static void playGame( Integer[][] arena ){
        int cols=arena.length;
        int rows=arena[0].length;
        int totalsum=0;

        for(int i=0;i<rows;i++){
            for(int j=0;j<cols;j++){


                if(arena[i][j]%50==0 && arena[i][j]!=0){
                    if(i>0 && arena[i-1][j]==2){
                        totalsum+=2;
                    }
                    if(i<rows-1 && arena[i+1][j]==2){
                        totalsum+=2;
                    }
                    if(j>0 && arena[i][j-1]==2){
                        totalsum+=2;
                    }
                    if(j<cols-1 && arena[i][j+1]==2){
                        totalsum+=2;
                    }
                }
            }
        }
        if(totalsum>=10){
            System.out.println("Points Gained: "+ totalsum+". Your team has survived the game.");
        }
        else{
            System.out.println("Points Gained: "+totalsum+". Your team is out");
        }
    }
