public static void mostWater(Integer[] height){
        int mA=0;
        int l=0;
        int r=height.length-1;
        
        while(l<r){
            int curH=Math.min(height[l], height[r]);
            int curW=r-l;
            int curA=curH*curW;
            
            if (curA>mA){
                mA = curA;
            }
            
            if (height[l]<height[r]){
                l++;
            }else{
                r--;
            }
        }
        System.out.print(mA);
    }
}
