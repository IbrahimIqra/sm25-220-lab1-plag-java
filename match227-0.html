public static void mostWater( Integer[] height ){

       int leftIdx = 0;
       int rightIdx = height.length-1; //Index is one less than length
       int areaMax = 0;
       int width=0;
       int shortHeight;
       int area;
       while(rightIdx>leftIdx){
         width=rightIdx-leftIdx;
         if(height[leftIdx]<height[rightIdx]){
           shortHeight = height[leftIdx];
         }
         else{
           shortHeight = height[rightIdx];
         }
         area = shortHeight*width;
         if(area>areaMax){
           areaMax=area;
         }
         if(height[rightIdx]>height[leftIdx]){
           leftIdx++;
         }
         else{
           rightIdx--;
         }
      }
        System.out.println(areaMax);
    }

      int sum;


      Integer [][] compressed = new Integer[r/2][c/2];
      for(int i = 0; i<r; i+=2){
        for(int j =0; j<c; j+=2){


          sum = matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+matrix[i+1][j+1];
