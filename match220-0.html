public static void mostWater( Integer[] height ){

        int l_side =0;
        int r_side =height.length-1;
        int max_area =0;

        while (l_side <r_side){

            int p;

            if(height[l_side] <height[r_side]){

                p =height[l_side];
            }
            else{
                p =height[r_side];
            }

            int k =r_side -l_side;
            int a =p *k;

            if (a >max_area){
                max_area =a;
            }

            if(height[l_side] <height[r_side]){
                l_side++;
            }
            else{
                r_side--;
            }
        }
        System.out.print(max_area);
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){

        //For this task you'll need to create new 2D array
        
        int og_r =matrix.length;
        int og_c =matrix[0].length;
        
        int nr =og_r /2;
        int nc =og_c /2;
        
        Integer [][] compressed =new Integer[nr][nc];
        
        for(int i=0; i<og_r; i=i+2){
          
          for(int j=0; j<og_c; j=j+2){
           


            int total_s =matrix[i][j] +matrix[i][j+1] +matrix[i+1][j] +matrix[i+1][j+1];
            
            compressed[i/2][j/2] =total_s;
          }
        }
        return compressed;
    }

        int[] y_ax ={0, 0, -1, 1};

          for(int j=0; j<c; j++){
           
            Integer v =arena[i][j];
