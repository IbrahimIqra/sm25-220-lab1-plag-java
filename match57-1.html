public static Integer rowRotation( Integer examWeek, String[][] matrix ){


        //For this task you don"t need to create new arrays
        //After rotation the Matrix should be printed inside the method
        //Only the integer row number is to be returned


        //TO DO


        int row_length = matrix.length;
        int col_length = matrix[0].length;




        for(int k = 0; k < examWeek-1; k++){


           
            // savinng last row
            String[] temp = new String[col_length];


            for(int j = 0; j < col_length; j++){
                temp[j] = matrix[row_length - 1][j];
            }


            for(int i = row_length - 1; i > 0; i--){


                for(int j = 0; j < col_length; j++){


                    matrix[i][j] = matrix[i - 1][j];
                }
            }


            for(int j = 0; j < col_length; j++){
                matrix[0][j] = temp[j];
            }
        }




       
        // Printing array
        for(int i = 0; i < row_length; i++){
            System.out.print(" | ");




            for(int j = 0; j < col_length; j++){
                System.out.print(matrix[i][j] + " | ");
            }


            System.out.println();
        }


        // Finding AA
        for(int i = 0; i < row_length; i++){


            for(int j = 0; j < col_length; j++){
                if (matrix[i][j].equals("AA")){
                    return i + 1;
                }
            }
        }






        return 0;
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){


        //For this task you'll need to create new 2D array
       
        //TO DO


        int rowNum = matrix.length;
        int colNum = matrix[0].length;


        Integer [][] np = new Integer[2][2];


        //counting from top left corner
        for(int i = 0; i < rowNum; i=i+rowNum/2){


            for(int j = 0; j<colNum; j=j+colNum/2){


                int sum = matrix[i][j] + matrix[i][j+1] + matrix[i+1][j] + matrix[i+1][j+1];
                       
                np[i/2][j/2] = sum;
            }




        }








        return np;
    }

public static void playGame( Integer[][] arena ){


        //For this task you don't need to create any new 2D array
        //just print the result inside the function
       
        //TO DO
        int rowNum = arena.length;
        int colNum = arena[0].length;


        int totalScore = 0;


        for(int i =0; i < rowNum; i++){


            for(int j= 0; j<colNum; j++){
                if (arena[i][j] % 50 == 0 && arena[i][j] != 0){
                    int score =0;


                    if (i-1 >= 0 && arena[i-1][j] == 2){


                        score +=2;
                    }
                   
                    if (i+1 < rowNum && arena[i+1][j] == 2){
                        score +=2;
                    }
                   
                    if (j-1 >= 0 && arena[i][j-1] == 2){
                        score +=2;
                    }
                   
                    if (j+1 < colNum && arena[i][j+1] == 2){
                        score +=2;


                    }
                    totalScore += score;


                }
            }
        }


        System.out.print("Points Gained: " + totalScore + ". ");


        if(totalScore >= 10){
            System.out.println("Your team has survived the game.");
        }


        else{
            System.out.println("Your team is out.");
        }


    }
