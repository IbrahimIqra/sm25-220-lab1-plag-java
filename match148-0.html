public static void mostWater( Integer[] height ){
      int i = 0;
      int j = height.length -1;
      int finalArea = 0;
      while(i<j){
        int h = 0;
        if (height[i]<height[j]){
          h = height[i];
        }
        else{
          h = height[j];
        }
        int width = j - i;


        int currentArea = h*width;
        if (currentArea>finalArea){
          finalArea = currentArea;
        }
        if (height[i] <height[j]){
          i++;
          
        }
        else{
          j--;
        }
      }
      System.out.print(finalArea);
    }

  public static Integer[][] compressMatrix( Integer[][] matrix ){
    int r = matrix.length;
    int c = matrix[0].length;
    int newRows = r/2;
    int newCols = c/2;
    Integer[][] compressed_matrix = new Integer[newRows][newCols];
    for(int i = 0; i<newRows; i++){
      for (int j = 0; j<newCols; j++){
        int rows = i*2;
        int cols = j*2;
        compressed_matrix[i][j] = matrix[rows][cols]+matrix[rows][cols+1]+matrix[rows+1][cols]+matrix[rows+1][cols+1];
      }
    }
    return compressed_matrix;
    
  }

  public static void playGame( Integer[][] arena ){
      int r = arena.length;
      int c = arena[0].length;
      int points = 0;
      for (int i = 0; i<r; i++){
        for(int j = 0; j<c; j++){


          if(arena[i][j]!=0 && arena[i][j]%50==0){
            if(i>0 && arena[i-1][j]==2){
              points+=2;
              
            }
            if(i<r-1 && arena[i+1][j] ==2){
              points+=2;
              
            }
            if (j>0 && arena[i][j-1]==2){
              points+=2;
            }
            if(j<c-1 && arena[i][j+1]==2){
              points+=2;            
          }
        }
      }
    }
      if (points>=10){
        System.out.print("Points Ganed: "+points+". Your team has survived the game.\n");
      }
      else{
        System.out.print("Points Ganed: "+points+". Your team is out.\n");
      }
