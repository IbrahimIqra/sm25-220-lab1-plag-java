TASK01class AssgnTask1{
    public static void mostWater( Integer[] height ){
         int max=0;
         int temp=0;
         for(int i=0; i<height.length; i++){
            for(int j=i; j<height.length; j++){
                if(height[i]<height[j]){
                   temp= height[i]* (j-i);}
                else{
                    temp= height[j]*(j-i);
                }
                if(temp>max){
                    max = temp;
                }

            }}
            System.out.println(max);
         }

                }
                System.out.println();
            }
   
        for (int i = 0; i < rows; i++) {

class AssgnTask3{


    public static Integer[][] compressMatrix( Integer[][] matrix ){
         

        int r = matrix.length;
        int c = matrix[0].length;
        Integer[][]result = new Integer[r/2][c/2];


        for (int i=0; i<r; i += 2) {
            for (int j= 0; j<c; j+= 2) {
                result[i/2][j/2] = matrix[i][j]+ matrix[i + 1][j]+ matrix[i][j+1]+ matrix[i+1][j+1];
            }
        }
        return result;
    }

class AssgnTask4{
    public static void playGame( Integer[][] arena ){
        

        int p = 0;
        int r = arena.length;
        int c = arena[0].length;


        for (int i = 0; i < r; i++) {
            for (int j = 0; j < c; j++) {
                if (arena[i][j]%50 == 0 && arena[i][j] != 0) {
                    if (i> 0 && arena[i-1][j]== 2) p += 2;
                    if (i< r-1&& arena[i+1][j]==2) p += 2;
                    if (j> 0 && arena[i][j-1] == 2) p += 2;
                    if (j< c-1 && arena[i][j+1] == 2) p += 2;
                }
            }
        }
        if(p>=10){
           System.out.println("Points Gained: " + p + ". " + "Your team has survived the game.");
        }
        else{
            System.out.println("Points Gained: " + p + ". " +  "Your team is out.");}
        }
