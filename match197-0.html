public static void mostWater(Integer[] height) {
        int amax = 0;
        int l = 0;
        int r = height.length - 1;



        while (l < r) {
            int h = Math.min(height[l], height[r]);
            int w = r - l;
            int a = h * w;
            amax = Math.max(amax, a);


            if (height[l] < height[r]) {
                l++;
            } else {
                r--;
            }
        }

        System.out.print(amax);
    }

public static Integer rowRotation( Integer examWeek, String[][] matrix ){

        
        int a = matrix.length;
        int b = (examWeek - 1) % a;




        for (int i = 0; i < b; i++) {

            String[] lrow = matrix[a - 1];


            for (int j = a - 1; j > 0; j--) {
                matrix[j] = matrix[j - 1];
            }


            matrix[0] = lrow;
        }


        for (int k = 0; k < a; k++) {
            System.out.print("| ");
            for (int l = 0; l < matrix[k].length; l++) {
                System.out.printf("%-3s", matrix[k][l]);


                System.out.print(" | ");
            }
            System.out.println();
        }


        for (int i = 0; i < a; i++) {
            for (int j = 0; j < matrix[i].length; j++) {
                if ("AA".equals(matrix[i][j])) {
                    return i + 1; 
                }
            }
        }

        return -1; 
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){

        
         int a = matrix.length;
        int b = matrix[0].length;


        int aa = a / 2;
        int bb = b / 2;


        Integer[][] compressed = new Integer[aa][bb];


        for (int i = 0; i < aa; i++) {
            for (int j = 0; j < bb; j++) {

                int r = i * 2;
                int c = j * 2;




                int sum1 = matrix[r][c] + matrix[r][c + 1]
                        + matrix[r + 1][c] + matrix[r + 1][c + 1];


                compressed[i][j] = sum1;
            }
        }

        return compressed;
    }

public static void playGame( Integer[][] arena ){

        int a = arena.length;
        int b = arena[0].length;
        int total = 0;

        for (int i = 0; i < a; i++) {
            for (int j = 0; j < b; j++) {



                if (arena[i][j] >= 50 && arena[i][j] % 50 == 0) {

                    if (i > 0 && arena[i - 1][j] == 2) {
                        total += 2;
                    }

                    if (i < a - 1 && arena[i + 1][j] == 2) {
                        total += 2;
                    }

                    if (j > 0 && arena[i][j - 1] == 2) {
                        total += 2;
                    }

                    if (j < b - 1 && arena[i][j + 1] == 2) {
                        total += 2;
                    }
                }
            }
        }
