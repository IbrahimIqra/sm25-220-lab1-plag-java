public static void mostWater(Integer[]array){
        int left = 0;
        int right = array.length-1;
        int maxarea = 0;


        while(left < right){
            int lol = 0;
            if (array[left] < array[right]){
                lol = array[left];
                left++;
            }
            else{
                lol = array[right];
                right--;
            }
            int wid = right - left + 1;
            int area = lol * wid;
            if (area > maxarea){
                maxarea = area;
            }
        }


        System.out.println(maxarea);
    }
