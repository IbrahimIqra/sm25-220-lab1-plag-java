public static void mostWater( Integer[] height ){
    int leftin=0;
    int rightin= height.length-1;
    int maxArea=0;
    while(leftin < rightin){
       
      int high=Math.min(height[leftin],height[rightin]);
      int surfacedis=rightin-leftin;
      int area=high*surfacedis;
      maxArea=Math.max(maxArea,area);
      if(height[leftin]<height[rightin]){
         
          leftin++;
      }
      else {
          rightin++;
      }
    }
   System.out.println(maxArea);
  

}
