//Assignment Task 01: Container with Most Water
class AssgnTask1{
public static void mostWater( Integer[] height ){
int maxArea = 0;
int left_line=0;
int right_line=height.length-1;

while (left_line < right_line) {
int curWidth= right_line - left_line;
int curHeight=Math.min(height[left_line], height[right_line]);
int curArea=curHeight*curWidth;
if (curArea > maxArea) 
{ maxArea= curArea;
 }
if (height[left_line] < height[right_line]) 
{left_line++;
} 
else 
{right_line--;
}
}
System.out.println(maxArea);
}

int row = matrix.length;

for (int i = 0; i < row; i++) {

 row_aa = i;
 col_aa = j;
 break;
        }
    }
if (row_aa != -1) break;
}

int newRow = (row_aa + (examWeek - 1)) % row;



for (int shift = 0; shift < examWeek - 1; shift++) {
 String[] temp = matrix[row - 1];
 for (int i = row - 1; i > 0; i--) {
 matrix[i] = matrix[i - 1];
  }

    }



//Assignment Task 03: Matrix Compression
public static Integer[][] compressMatrix( Integer[][] matrix ){
int row = matrix.length;
int col = matrix[0].length;


Integer[][] smaller_matrix = new Integer[row / 2][col / 2];
for (int i = 0; i < row; i += 2) {
for (int j = 0; j < col; j += 2) {
int sum = matrix[i][j] + matrix[i][j + 1] + matrix[i + 1][j] + matrix[i + 1][j + 1];
smaller_matrix[i / 2][j / 2] = sum;
    }
}
return smaller_matrix;
}




//Assignment Task 04: Game Arena


public static void playGame( Integer[][] arena ){
           int total_point = 0;
int row = arena.length;
int col = arena[0].length;

for (int i = 0; i < row; i++) {
for (int j = 0; j < col; j++) {
if (arena[i][j] % 50 == 0 && arena[i][j] != 0) {
if (i > 0 && arena[i - 1][j] == 2) {


total_point += 2;
}
if (i < row - 1 && arena[i + 1][j] == 2) {
total_point += 2;
}
if (j > 0 && arena[i][j - 1] == 2) {


total_point += 2;
}
if (j < col - 1 && arena[i][j + 1] == 2) {
total_point += 2;
}
}
}
}

if (total_point >= 10) {
