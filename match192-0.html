int maxWater = 0;
        int left = 0;
        int right = height.length - 1;


        while (left < right) {
            // Calculate the current area
            int heightOfContainer = Math.min(height[left], height[right]);
            int width = right - left;
            int currentWater = heightOfContainer * width;


            // Update maxWater if we found a bigger container
            maxWater = Math.max(maxWater, currentWater);


            // Move the pointer pointing to the shorter line
            if (height[left] < height[right]) {
                left++;
            } else {
                right--;
            }
        }


        return maxWater;
    }

            String[] lastRow = seat_status[rows - 1];


            // Shift rows downward
            for (int i = rows - 1; i > 0; i--) {
                seat_status[i] = seat_status[i - 1];
            }
            seat_status[0] = lastRow; // Move last row to the top
        }


        // Find "AA" and return the row number (1-based)
        for (int i = 0; i < rows; i++) {
            for (String seat : seat_status[i]) {
                if (seat.equals("AA")) {

                }
            }
        }


        return -1; // "AA" not found
    }
