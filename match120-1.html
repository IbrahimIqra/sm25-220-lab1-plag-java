public static void mostWater( Integer[] height ){
        int i = 0 ;
        int j = height.length-1 ;
        int max = 0 ;

        while (i < j) {
            int h = 0 ;
        if (height[i] < height[j]) {
            h = height[i] ;
        } 
        else {


            h = height[j] ;
        }
        int l = j - i ;
        int area = h * l ;
        if (area > max) {
            max = area ;
        }
        if (height[i] < height[j]) {
            i++ ;
        } 
        else {
            j-- ;
        }
        }
        System.out.print(max) ;
    }

public static Integer rowRotation( Integer examWeek, String[][] matrix ){
        int row = matrix.length ;
        int column = matrix[0].length ;


        for (int i = 0 ; i < examWeek ; i++) {
            String[] newArr = matrix[row - 1] ;
            for (int j = row - 1 ; j > 0 ; j--) {
                matrix[j] = matrix[j - 1] ;
            }
            matrix[0] = newArr ;
        }
        for (int i = 0 ; i < row ; i++) {
            System.out.print("| ") ;
            for (int j = 0 ; j < column ; j++) {
                System.out.print(matrix[i][j] + " | ") ;
            }
            System.out.println() ;
        }
        for (int i = 0; i < row; i++) {
            for (int j = 0; j < column; j++) {
                if (matrix[i][j] == "AA") {
                    return i + 1 ;
                }
            }
        }
        return null ;
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){
        int row = matrix.length ;
        int column = matrix[0].length ;
        Integer[][] newArr = new Integer[2][2] ;


        for (int i = 0; i < row ; i += 2) {
            for (int j = 0; j < column ; j += 2) {
                int a = matrix[i][j] ;
                int b = matrix[i][j + 1] ;
                int c = matrix[i + 1][j] ;
                int d = matrix[i + 1][j + 1] ;
                int sum = (a + b + c + d) ;
                if (i == 0 && j == 0) {
                newArr[0][0] = sum ;
                }
                else if (i == 0 && j == 2) {
                    newArr[0][1] = sum ;
                }
                else if (i == 2 && j == 0) {
                    newArr[1][0] = sum ;
                }
                else if (i == 2 && j == 2) {
                newArr[1][1] = sum ;
                }
                return newArr ;
            }

public static void playGame( Integer[][] arena ){
        int row = arena.length ;
        int column = arena[0].length ;
        int total = 0 ;
        for (int i = 0 ; i < row ; i++) {
            for (int j = 0 ; j < column ; j++) {
                if (arena[i][j] != 0 && arena[i][j] % 50 == 0) {
                    int points = 0 ;
                    if (i - 1 >= 0) {
                        if (arena[i - 1][j] == 2) {
                            points += 2 ;
                        }
                    }
                    if (i + 1 < row) {
                        if (arena[i + 1][j] == 2) {
                            points += 2 ;
                        }
                    }
                    if (j - 1 >= 0) {
                        if (arena[i][j - 1] == 2) {
                            points += 2 ;
                        }
                    }
                    if (j + 1 < column) {
                        if (arena[i][j + 1] == 2) {
                            points += 2 ;
                        }
                    }
                    total += points ;
                }
            }
        }


        if (total >= 10) {
            System.out.println("Points Gained: " + total + ". Your team has survived the game.") ;
        }
        else {
            System.out.println("Points Gained: " + total + ". Your team is out.") ;
        }
    }
