public static Integer[][] compressMatrix( Integer[][] matrix ){

        int row =matrix.length;
        int col =matrix[0].length;

        int halfRow= (row/2);
        int halfCol= (col/2);

        int [][] compressed= new int [halfRow][halfCol];
        int countRow=0;
        int countCol=0;

        for(int i=0; i<row; i+=2){
            for(int j=0; j<col; j+=2){

                int sum=matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+matrix[i+1][j+1];
                
                compressed[countRow][countCol++]=sum;
            }
            countCol=0;
            countRow++;
            
        }
        return compressed;

   }

public static void playgame(int[][] arena) {

        int row = arena.length;
        int col = arena[0].length;
        int sum = 0;



        for (int i = 0; i < row; i++) {
            for (int j = 0; j < col; j++) {

                if (arena[i][j] != 0 && arena[i][j] % 50 == 0) {

                    if (i > 0 && arena[i - 1][j] == 2) {
                        sum += 2;
                    }
                    if (j > 0 && arena[i][j - 1] == 2) {
                        sum += 2;
                    }
                    if (i < row - 1 && arena[i + 1][j] == 2) {
                        sum += 2;
                    }
                    if (j < col - 1 && arena[i][j + 1] == 2) {
                        sum += 2;
                    }

                }
            }
        }

        if (sum >= 10) {
                System.out.print("Points Gained: " + sum + ". Your team has survived the game");

        } else {
                System.out.println("Points Gained: " + sum + ". Your team has survived the game");

        }
   }
