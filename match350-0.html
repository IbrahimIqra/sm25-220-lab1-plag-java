// Assignment 1
public static void mostWater( Integer[] height ){
  int mArea=0;
  int L=0;


  int R=height.length-1;
  while(L<R){
    int H=0;
    if(height[L]<height[R]){
      H=height[L];
    }
    else{
      H=height[R];
    }
    int wdt=R-L;
    int area=H*wdt;
    if(area>mArea){
      mArea=area;
    }
    if(height[L]<height[R]){
      L++;
    }
    else{
      R--;
    }
  }
  System.out.println(mArea);
}

  public static Integer rowRotation( Integer examWeek, String[][] matrix ){
    int row=matrix.length;
    int col=matrix[0].length;


    int rotate=(examWeek-1)%row;
    
    for(int k=0;k<rotate;k++){
      String[]l_Row=matrix[row-1];
      for(int i=row-1;i>0;i--){
        matrix[i]=matrix[i-1];
      }
      matrix[0]=l_Row;


    }
    int aA_row=0;
    for(int i=0;i<row;i++){
      for(int j=0;j<col;j++){
        if("AA".equals(matrix[i][j])){
          aA_row=i+1;
        }
      }
    }


    for(int i=0;i<row;i++){
      System.out.print(" | ");
      for(int j=0;j<col;j++){
        System.out.print(matrix[i][j]+" | ");
      }
      System.out.println();
    }


    return aA_row;
  }

  public static Integer[][] compressMatrix( Integer[][] matrix ){
    int row=matrix.length;
    int col=matrix[0].length;
    Integer[][]matrix2 = new Integer[row/2][col/2];


    
    for(int i=0;i<row;i+=2){
      for(int j=0;j<col; j+=2){
        int s= matrix[i][j]+matrix[i+1][j]+matrix[i][j+1]+matrix[i+1][j+1];


        matrix2[i/2][j/2]=s;
      }
    }
    return matrix2;
  }

  public static void playGame( Integer[][] arena ){
    int row=arena.length;
    int col=arena[0].length;
    int need_point=10;
    int point=0;
    
    for(int i=0;i<row;i++){
      for(int j=0;j<col;j++){
        int num=arena[i][j];
        if(num!=0 && num%50==0){
          if(i>0 && arena[i-1][j]==2){
            point++;
          }
          if(i<row-1 && arena[i+1][j]==2){
            point++;
          }
          if(j>0 && arena[i][j-1]==2){
            point++;
          }
          if(j<col-1 && arena[i][j+1]==2){
            point++;
          }
        }
      }
      point=point*2;
      if(point<need_point){
        System.out.println("Points Gained: "+point+". Your team is out.");
      }
      else{
        System.out.println("Points Gained: "+point+". Your team has survived the game.");
      }
    }
