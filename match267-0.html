public static int mostWater( Integer[] height ){
    int left = 0;
    int right = height.length -1;
    int max = 0;
    
    while(left< right){
      int hei;
      if(height[left]< height[right]){
        hei = height[left];
      }
      else{


         hei = height[right];
      }
      
      int wid = right - left;
      int area = hei * wid;
      
      if(area> max){
        max = area;
      }
      
      if(height[left] < height[right]){
        left++;
      }
      else{
        right--;
      }
    }
    
    return max;
 
   }

 public static Integer[][] compressMatrix( Integer[][] matrix ){
      
      int r = matrix.length;
      int c = matrix[0].length;
      
      Integer[][]  arr2 = new Integer[r/2][c/2];
      
      for(int i= 0; i<r; i+=2){
        for(int j= 0; j<c; j+=2){
          int sum = matrix[i][j] +matrix[i][j+1] + matrix[i+1][j]+ matrix[i+1][j+1];
          arr2 [i/2][j/2] = sum;
        }


        
      }
      return arr2;
    }

public static void playGame( Integer[][] arena ){
    
    int sum =0;
    for(int i= 0; i<arena.length; i++){
      for(int j=0; j<arena[i].length; j++){
        if(arena[i][j]!=0 && arena[i][j]%50==0 ){


          if(i>0 && arena[i-1][j]==2){
            sum+= 2;
          }
           if(i< arena.length-1 && arena[i+1][j]==2){
            sum+= 2;
          }
           if(j>0 && arena[i][j-1]==2){
            sum+= 2;
          }
           if(j<arena[i].length-1 && arena[i][j+1]==2){
            sum+= 2;
          }
        }
      }
    }
    
    if(sum>=10){
      System.out.println("Points Gained: " +sum +". Your team has survived the game.");
    }
    else{
      System.out.println( "Points Gained: " +sum +". Your team is out.");
    }
  }
