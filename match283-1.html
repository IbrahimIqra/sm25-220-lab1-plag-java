 public static void mostWater( Integer[] height ) {

        //TO DO
        int indexfromleft = 0;
        int indexfromright = height.length - 1;
        int WaterArea = 0;

        while (indexfromleft < indexfromright) {
            int minimumHeight = 0;


            if (height[indexfromright] < height[indexfromleft]) {
                minimumHeight = height[indexfromright];
            } else {
                minimumHeight = height[indexfromleft];
            }

            int length = indexfromright - indexfromleft;
            int normalArea = minimumHeight * length;

            if (normalArea > WaterArea) {
                WaterArea = normalArea;
            }

            if (height[indexfromright] < height[indexfromleft]) {
                indexfromright--;
            } else {
                indexfromleft++;
            }
        }
        System.out.println(WaterArea);
       
    }
