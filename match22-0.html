    public static void mostWater( Integer[] height ){
        Integer [] a = new Integer[height.length*height.length];


        int count=0;
        for(int i=0; i<height.length; i++){
            for(int j=0; j<height.length; j++){
                int width = 0;
                if(i>j){
                    width=i-j;
                } else if ( j>i) {
                    width = j-i;
                }
                a [count++] = Math.min(height[i], height[j])*width;
            }
        }


        int max = -1;
        for(int i=0; i<a.length; i++){
            max = Math.max(max, a[i]);
        }
        System.out.println(max);

        }

public static Integer rowRotation(Integer examWeek, String[][] matrix) {
    int rowC = matrix.length;
    int colC = matrix[0].length;


    for (int week = 0; week < examWeek-1; week++) {
        String[] LastRow = new String[colC];
        for (int j = 0; j < colC; j++) {
            LastRow[j] = matrix[rowC - 1][j];


        }
        for (int i = rowC - 1; i > 0; i--) {
            for (int j = 0; j < colC; j++) {
                matrix[i][j] = matrix[i - 1][j];
            }
        }
        for (int j = 0; j < colC; j++) {
            matrix[0][j] = LastRow[j];
        }
    }

    Arr.print2D(matrix);

    int rowNum = 0;
    boolean flag = false;


    for (int i = 0; i < rowC; i++) {
        for (int j = 0; j < colC; j++) {
            if (matrix[i][j].equals("AA")) {
                rowNum = i+1;
                flag = true;
                break;
            }
        }
        if (flag) {
            break;
        };
    }



    return rowNum;

}

    public static void playGame( Integer[][] arena ){
        int totalPoints = 0;
        for ( int row = 0; row < arena.length; row++ ){

            for ( int col = 0; col < arena[row].length; col++ ){
                if ( arena[row][col] %50==0 ){
                    if ( row+1< arena.length && arena[row+1][col] == 2  ){
                        totalPoints+=2;
                    }
                    if (row-1> arena.length && arena[row+1][col] == 2  ){
                        totalPoints+=2;
                    }
                    if(col+1< arena[row].length && arena[row][col+1] == 2  ){
                        totalPoints+=2;
                    }
                    if(col-1> arena[row].length && arena[row][col-1] == 2  ){
                        totalPoints+=2;
                    }

                }

            }

        }


        if(totalPoints>=10){
            System.out.println("Points Gained: "+totalPoints+". Your team has survived the game.\n");
        }
        else{
            System.out.println( "Points Gained: "+totalPoints +". Your team is out.\n");
        }
    }
