Assignment Task 01: Container with Most Water


public static void mostWater( Integer[] height ){  
    int lE = 0;
    int rI = height.length - 1;
    int max = 0;  
    while (lE < rI) {
      int h;


      if (height[lE] < height[rI]) {
        h = height[lE];
      } else {
        h = height[rI];
      }     
      int w = rI - lE;
      int area = h * w;      
      if (area > max) {
        max = area;
      }     
      if (height[lE] < height[rI]) {
        lE++;
      } 
      else {
        rI--;
      }
    }   
    System.out.println(max);
  }

Assignment Task 02: Row Rotation Policy
public static Integer rowRotation( Integer examWeek, String[][] matrix ){
    int rot = examWeek - 1;
    for (int r = 0; r < rot; r++) {
      String[] temp = new String[matrix[0].length];
      for (int j = 0; j < matrix[0].length; j++) {
        temp[j] = matrix[matrix.length - 1][j];
      }
      for (int i = matrix.length - 1; i > 0; i--) {
        for (int j = 0; j < matrix[0].length; j++) {
          matrix[i][j] = matrix[i - 1][j];
        }
      }
      for (int j = 0; j < matrix[0].length; j++) {


        matrix[0][j] = temp[j];
      }
    }
    for (int i = 0; i < matrix.length; i++) {
      System.out.print("| ");
      for (int j = 0; j < matrix[0].length; j++) {
        System.out.print(matrix[i][j]);
        if (matrix[i][j].length() == 1) System.out.print("  | ");
        else System.out.print(" | ");
      }
      System.out.println();
    }
    for (int i = 0; i < matrix.length; i++) {
      for (int j = 0; j < matrix[0].length; j++) {
        if (matrix[i][j].equals("AA")) {
          return i + 1;
        }
      }
    }
    return -1;
  }

Assignment Task 03: Matrix Compression


public static Integer[][] compressMatrix( Integer[][] matrix ){
    int row = matrix.length;
    int col = matrix[0].length;
    int nr = row / 2;
    int nc = col / 2;


    Integer[][] res = new Integer[nr][nc];
    for (int i = 0; i < row; i += 2) {
      for (int j = 0; j < col; j += 2) {
        int sum = matrix[i][j] + matrix[i][j + 1] + matrix[i + 1][j] + matrix[i + 1][j + 1]; 
        res[i / 2][j / 2] = sum;
      }
    }
    return res;
  }

Assignment Task 04: Game Arena


  public static void playGame( Integer[][] arena ) {
    int row = arena.length;
    int col = arena[0].length;
    int tp = 0;
    for (int i = 0; i < row; i++) {
      for (int j = 0; j < col; j++) {
        if (arena[i][j] % 50 == 0 && arena[i][j] != 0) {
          int c = 0;
          if (i > 0 && arena[i - 1][j] == 2) {
            c++;
          }
          if (i < row - 1 && arena[i + 1][j] == 2) {
            c++;
          }
          if (j > 0 && arena[i][j - 1] == 2) {
            c++;
          }
          if (j < col - 1 && arena[i][j + 1] == 2) {
            c++;
          }
          tp += c * 2;
        }
      }
    }


    System.out.print("Points Gained: "+ tp +". ");
    if (tp >= 10) {
      System.out.println("Your team has survived the game.");
    } else {
      System.out.println("Your team is out.");
    }
  }
