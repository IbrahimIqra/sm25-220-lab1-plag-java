public static void mostWater( Integer[] a ){


   int lft =0;
   int r = a.length-1;
   int Area =0;
   while(lft < r){
       int h;
       if(a[lft] < a[r]){
           h=a [lft];
       }else{
           h=a[r];
       }
       int w= r-lft;
       int area= h*w;


       if(area>Area){
           Area= area;
       }
       if(a[lft] < a[r]){
           lft++;
       }else{
           r--;
       }
   }
   System.out.print(Area);
}

public static Integer rowRotation( Integer examWeek, String[][] matrix ){

   //Only the integer row number is to be returned


   int row= matrix.length;
   int col= matrix[0].length;
   int circle= (examWeek-1)%row;
   for(int i= 0; i<row; i++){
       int circled_row= (i + row-circle)%row;


       System.out.print("| ");
       for(int j= 0; j<col; j++) {
           System.out.print(matrix[circled_row][j]+"  | ");
       }
       System.out.println();
   }
   int aaRow = -1;
   for(int i=0; i<row; i++){
       int cir_row = (i + row - circle)%row;
       for(int j= 0; j<col; j++){
           if("AA".equals(matrix[cir_row][j])){
               aaRow= i+1;
               break;
           }
       }
       if(aaRow!= -1) break;
   }
   return aaRow;
}

public static Integer[][] compressMatrix( Integer[][] matrix ){


       int row= matrix.length;
       int col= matrix[0].length;
       int R= row/2;
       int C= col/2;
       Integer[][] compressedMatrix= new Integer[R][C];
       for(int i= 0; i<R; i++){
           for(int j= 0; j<C; j++){
               int add= matrix[2*i][2*j] + matrix[2*i][2*j+ 1]+matrix[2*i+1][2*j]+ matrix[2*i+1][2*j+1];
               compressedMatrix[i][j] = add;
           }
       }
       return compressedMatrix;
}

public static void playGame( Integer[][] arena ){


   //For this task you don't need to create any new 2D array
   //just print the result inside the function
   int row= arena.length;
   int col= arena[0].length;
   int sum= 0;


   int[] xx= {-1, 1, 0, 0};
   int[] yy= {0, 0, -1, 1};


   for(int i= 0; i<row; i++){
       for(int j= 0; j<col; j++){
           if(arena[i][j] != null && arena[i][j]%50 ==0 && arena[i][j] != 0) {


               int points= 0;
               for(int a=0; a<4; a++){
                   int ii= i + xx[a];
                   int nj= j + yy[a];
                   if(ii >= 0 && ii < row && nj >= 0 && nj < col){
                       if(arena[ii][nj] != null && arena[ii][nj] == 2){
                           points += 2;
                       }
                   }
               }
               sum += points;
           }
       }
   }


   if (sum >= 10) {
       System.out.println("Points Gained: " + sum + ". Your team has survived the game.");
   } else {
       System.out.println("Points Gained: " + sum + ". Your team is out.");
   }
}
