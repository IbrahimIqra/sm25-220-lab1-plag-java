public static void mostWater( Integer[] height ){
        int start = 0;
        int end = height.length - 1; 
        int max_area = 0;
        while (start < end){
            int h;
            if (height[start] < height[end]){
                h = height[start];
            }
            else{
                h = height [end];
            }
            int w = end - start;
            int area = h * w;
            
            if (area > max_area){


                max_area = area;
            }

            if (height[start] < height[end]){
                start ++;
            }
            else{
                end --;
            }
        }
        
        System.out.println(max_area);
}

public static Integer rowRotation( Integer examWeek, String[][] matrix ){
        int rotate = examWeek - 1;  
        for(int r = 0; r < rotate; r ++){


            String lastRow[]= matrix[matrix.length - 1];
            for (int i = matrix.length - 1; i > 0; i --){
                matrix[i] = matrix[i - 1];
            }
            matrix[0] = lastRow;
        }

        for (int i = 0; i < matrix.length; i ++){
            System.out.print ("| ");
            for (int j = 0; j < matrix[i].length; j ++){
                System.out.print(matrix[i][j] + " | ");
            }
            System.out.println();
        }

        int rowNumber = -1;
        for (int i = 0; i < matrix.length; i ++){
            for (int j = 0; j < matrix[i].length; j ++){
                if (matrix[i][j].equals("AA")){
                    rowNumber = i + 1;
                    break;
                }
            }
        }
        return rowNumber;
}

public static Integer[][] compressMatrix( Integer[][] matrix ){
        int row = matrix.length;
        int col = matrix[0].length;
        int new_row = row / 2;
        int new_col = col / 2;


        Integer new_matrix[][] = new Integer[new_row][new_col]; 

        for (int i = 0; i < row; i += 2){
            for (int j = 0; j < col; j += 2){
                int sum = matrix[i][j] + matrix[i][j + 1] + matrix[i + 1][j] + matrix[i + 1][j + 1] ;
                new_matrix[i / 2][j / 2] = sum;
            }
        }
        return new_matrix;
}

public static void playGame( Integer[][] arena ){
        int t_count = 0;

        for (int i = 0; i < arena.length; i ++){
            for (int j = 0; j < arena[0].length; j ++){


                if (arena[i][j] != 0 && arena[i][j] % 50 == 0){
                    int count = 0;
                    if (i - 1 >= 0 && arena[i - 1][j] == 2){
                        count += 1;
                    }
                    if (i + 1 < arena.length && arena[i + 1][j] == 2){
                        count += 1;
                    }
                    if (j - 1 >= 0 && arena[i][j - 1] == 2){
                        count += 1;
                    }
                    if (j + 1 < arena[0].length && arena[i][j + 1] == 2){
                        count += 1;
                    }
                    t_count += count;
                }
            }
        }
        int t_points = 2 * t_count;

        if (t_points >= 10){
            System.out.println("Points Gained: " + t_points + ". Your team has survived the game.");
        }
        else{
            System.out.println("Points Gained: " + t_points + ". Your team is out.");
        }
}
