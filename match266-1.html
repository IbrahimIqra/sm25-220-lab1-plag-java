 public static void mostWater(Integer[] height) {
    int i = 0;
    int j = height.length - 1;
    int maxWater = 0;

    while (i < j) {
        
        int h = Math.min(height[i], height[j]);
        int width = j - i;
        int area = h * width;
        System.out.println(area);

        
        if (area > maxWater) {
            maxWater = area;
        }

        
        if (height[i] < height[j]) {
            i++;
        } else {
            j--;
        }
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){

       int row = matrix.length;
       int col=matrix[0].length;
       int newrow=row/2;
       int newcol=col/2;
       Integer[][] mat=new Integer[newrow][newcol];


       for(int i=0;i<row;i+=2){
           int sum=matrix[i][i]+matrix[i][i+1]+matrix[i+1][i]+matrix[i+1][i+1];
           mat[i/2][i/2]=sum;
       }
       return mat;
    }

public static void playGame( Integer[][] arena ){

       int row=arena.length;
       int col=arena[0].length;
       int totalpoints=0;


       int[] dx={-1,1,0,0};
       int[] dy={0,0,-1,1};
       for(int i=0;i<row;i++){
           for(int j=0;j<col;j++){
               int value=arena[i][j];
               if(value%50==0&&value!=0){


                   for(int k=0;k<4;k++){
                       int ni=i+dx[k];
                       int nj=j+dy[k];
                       if (ni >= 0 && ni < row && nj >= 0 && nj < col) {
                       if(arena[ni][nj]==2){
                           totalpoints++;
                       }
                    
                   }
               }
           }
       }
       }
       int points=totalpoints*2;
       if(points>=10){
           System.out.println("Points Gained: "+points+". Your team has survived the game.");
       }
       else{
           System.out.println("Points Gained: "+points+". Your team is out.");
       }
    
    }
