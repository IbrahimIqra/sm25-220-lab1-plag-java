    public static void mostWater( Integer[] height ){
        
        int leftBoundary = 0;
        int rightBoundary = height.length - 1;
        int maximumArea = 0;

        for (; leftBoundary < rightBoundary;){ // While loop alternative
            


            int shortestHeight; 
            
            /* Need to compare the two boundaries and find the one with shorter height
               and use it as the height for the area because otherwise the water will spill over */
            
            if (height[leftBoundary] < height[rightBoundary]) {
                shortestHeight = height[leftBoundary];
            } else {
                shortestHeight = height[rightBoundary];
            }
            
            // Here I also could have used Math.min()

            int width = rightBoundary - leftBoundary;
            int area = shortestHeight * width;
            
            if (area > maximumArea) {
                maximumArea = area;
            }
            
            /* We have to move the boundaries inward after each iteration. 
               However, there is a catch we have to keep the larger boundary and eleminate the shorter one.
               then move inwards from there. Because to find an even bigger area the shortest boundary must beceome larger */
            
            if (height[leftBoundary] < height[rightBoundary]) {
                leftBoundary++;
            } else {
                rightBoundary--;
            }
        }
        System.out.println(maximumArea);
    }

    public static Integer[][] compressMatrix( Integer[][] matrix ){
        
        int rows = matrix.length;
        int cols = matrix[0].length;
        // since every block is 2x2 and the new matrix will have half the rows and half the columns
        int newRows = rows / 2; 
        int newCols = cols / 2;
        
        Integer[][] compressed = new Integer[newRows][newCols];
        
        for (int i = 0; i < newRows; i++) {
            for (int j = 0; j < newCols; j++) {
                // I have to calculate the top-left index of the current 2x2 block
                int r = 2 * i; // compressed matrix is smaller and have half the rows and half the columns
                int c = 2 * j;


                compressed[i][j] = matrix[r][c] + matrix[r][c + 1] + matrix[r + 1][c] + matrix[r + 1][c + 1];
            }
        }
        return compressed;
    }

    public static void playGame( Integer[][] arena ){
        
        int rows = arena.length;
        int cols = arena[0].length;
        int totalPoints = 0;
        
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if (arena[i][j] != 0 && arena[i][j] % 50 == 0) { 



                    if (i - 1 >= 0 && arena[i - 1][j] == 2) {
                        totalPoints += 2;
                    }
                    if (i + 1 < rows && arena[i + 1][j] == 2) {
                        totalPoints += 2;
                    }
                    if (j - 1 >= 0 && arena[i][j - 1] == 2) {
                        totalPoints += 2;
                    }
                    if (j + 1 < cols && arena[i][j + 1] == 2) {
                        totalPoints += 2;
                    }

                }
            }
        }
        // The inner conditions check for UP, DOWN, LEFT, RIGHT
        // Player cell must be non-zero and multiple of 50
        // Each if statement checks boundaries first so no way out of the matrix
        // I was thinking about doing this problem with direction arrays, will try it later!
        
        System.out.print("Points Gained: " + totalPoints + ". ");
        if (totalPoints >= 10) {
            System.out.println("Your team has survived the game.");
        } else {
            System.out.println("Your team is out.");
        }
    }
