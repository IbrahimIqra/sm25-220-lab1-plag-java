 public static void mostWater( Integer[] height ){
    int i = 0;
    int j = height.length-1;
    int max = 0;
    while (i<j){
      int h;
      if (height[i]<height[j]){
        h = height[i];
      }
      else{
        h = height[j];
      }
      int w = j-i;
      int area = h*w;
      if (area>max){
        max = area;
      }
      if (height[i]<height[j]){
        i++;
      }
      else{
        j--;
      }
    }
    System.out.print(max);
  }

 public static Integer rowRotation( Integer examWeek, String[][] matrix ){
    int rows = matrix.length;
    int cols = matrix[0].length;
    int rotations = (examWeek - 1) % rows;
    for (int r = 0; r < rotations; r++) {
      String[] temp = matrix[rows - 1];
      for (int i = rows - 1; i > 0; i--) {
        matrix[i] = matrix[i - 1];
      }
      matrix[0] = temp;
    }


    for (int i = 0; i < rows; i++) {
      System.out.print("| ");
      for (int j = 0; j < cols; j++) {
        System.out.print(matrix[i][j] + " | ");
      }
      System.out.println();
    }
    for (int i = 0; i < rows; i++) {
      for (int j = 0; j < cols; j++) {
        if ("AA".equals(matrix[i][j])) {
          return i + 1;
        }
      }
    }
    return -1;
  }

    public static Integer[][] compressMatrix( Integer[][] matrix ){
      int rows = matrix.length;
      int cols = matrix[0].length;
      Integer[][] compressed = new Integer[rows/2][cols/2];
      for (int i=0; i<rows; i+=2){
        for (int j=0; j<cols; j+=2){


          int sum = matrix[i][j] + matrix[i][j+1] + matrix[i+1][j] + matrix[i+1][j+1];
          compressed[i/2][j/2] = sum;
        }
      }
      return compressed;
    }

public static void playGame(Integer[][] arena) {
    int r = arena.length;
    int c = arena[0].length;
    int totalPoints = 0;
    for (int i=0; i<r; i++){
      for (int j=0; j<c; j++){


        if (arena[i][j]!=null && arena[i][j]%50==0 && arena[i][j]!=0){
          int points=0;
          if (i-1>=0){
            if (arena[i-1][j]==2){
              points++;
            }
          }
          if (i+1<r){
            if (arena[i+1][j]==2){
              points++;
            }
          }
          if (j-1>=0){
            if (arena[i][j-1]==2){
              points++;
            }
          }
          if (j+1<c){


            if (arena[i][j+1]==2){
              points++;
            }
          }
          totalPoints+=points*2;
        }
      }
    }
    if (totalPoints >= 10){
      System.out.println("Points Gained: " + totalPoints + ". Your team has survived the game.");
    } 
    else{
      System.out.println("Points Gained: " + totalPoints + ". Your team is out.");
    }
  }
