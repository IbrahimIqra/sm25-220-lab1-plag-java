 public static void mostWater( Integer[] height ){


        Integer maxArea=0;
        Integer pointer1=0;
        Integer pointer2=height.length-1;
        
        while (pointer1<pointer2)
        {
          maxArea=Math.max(maxArea,(pointer2-pointer1)*Math.min(height[pointer1],height[pointer2]));
          
          if (height[pointer1]<height[pointer2])
          {
            pointer1++;
          }
          else
          {
            pointer2--;
          }
        }
        System.out.println(maxArea);
    }

public static Integer rowRotation( Integer examWeek, String[][] matrix ){


        int row= matrix.length;
        int rotation=(examWeek%row)-1;
        int i=0;


        int last_row=matrix.length-1;
        while(i<last_row)
        {
          String[]temp=matrix[i];
          matrix[i]=matrix[last_row];
          matrix[last_row]=temp;
          i++;
          last_row--;
        }
        i=0;
        last_row=rotation-1;
        while(i<last_row)
        {
          String[]temp=matrix[i];
          matrix[i]=matrix[last_row];
          matrix[last_row]=temp;
          i++;
          last_row--;
        }
        i=rotation+1;
        last_row=matrix.length-1;
        while(i<last_row)
        {
          String[]temp=matrix[i];
          matrix[i]=matrix[last_row];
          matrix[last_row]=temp;
          i++;
          last_row--;
        } 
        
        
        int rows=0;
        for (int x=0;x<matrix.length;x++){
            for (int y=0;y < matrix[0].length;y++){
                 
                if (matrix[x][y].equals("AA")){
                    rows = x+1 ;
                }
                if (y == matrix[0].length-1){
                    System.out.print("| "+matrix[x][y]+" |");    
                }
                else{
                    System.out.print("|  "+matrix[x][y]+"  ");


                }
                
            }
            System.out.println();
        }


        return rows;
    }

 public static Integer[][] compressMatrix( Integer[][] matrix ){
        
        Integer row=matrix.length;
        Integer col=matrix[0].length;
        Integer r=row/2;
        Integer c=col/2;
        Integer[][] compressed = new Integer[r][c];
        Integer r1=0;
        Integer c1=0;
        for (int i=0; i<row; i+=2)
        {
          for ( int j=0;j<col;j+=2)
          {


            int sum=matrix[i][j]+matrix[i+1][j]+matrix[i][j+1]+matrix[i+1][j+1];
            compressed[r1][c1]=sum;
            c1++;
          }
           r1++;
           c1=0;
        }
        return compressed;
    }

public static void playGame( Integer[][] arena ){
      int row =arena.length;
      int col =arena[0].length;
      int total =0;
      for (int x=0;x<row;x++)
      {
        for (int y=0;y<col;y++)
        {


          if (arena[x][y]!=0 && arena[x][y]%50 ==0)
          {
            if (x>0 && arena[x-1][y]==2)
            {
              total+=2;
            }
            if (x<row-1 && arena[x+1][y]==2)
            {
              total+=2;
            }
            if (y>0 && arena[x][y-1]==2)
            {
              total+=2;
            }
            if (y<col-1 && arena[x][y+1]==2)
            {
              total+=2;
            }
          }
        }
      }
      if (total>=10)
      {
        System.out.println("Points Gained: " + total + ". Your team has survived the game.");
      } 
      else 
      {
        System.out.println("Points Gained: " + total + ". Your team is out.");
      }
    }
