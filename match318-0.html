public static void mostWater(Integer[]height){
    int s1=0;
    int s2=height.length-1;
    int max=0;
    int min=0;
    while(s1<s2){
      int w=s2-s1;
      if(height[s2]>height[s1]){
         min=height[s1];
      } 
      else{
         min=height[s2];
      }
     int ar=w*min;
      if(ar>max){


         max=ar;
      }
      if(height[s1]<height[s2]){
         s1++;
      } 
      else{
         s2--;
      }
    }
   System.out.println(max);
}

public static Integer[][] compressMatrix( Integer[][] matrix ){
    int rw=matrix.length;
    int co=matrix[0].length;
    Integer[][]newrw = new Integer[rw/2][co/2];
    int i=0;
    while(i<rw){
      int j=0;
        while(j<co){


          int temp=matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+matrix[i+1][j+1];
          newrw[i/2][j/2]=temp;
          j+=2;
        }
        i+=2;
    }
    return newrw;
}
