public static void mostWater( Integer[] height ){

        int left;
        int right = height.length - 1; //9-1=8
        int maxArea;

        while(left < right){ // 0 < 8
            int width = right - left; //8 - 0 = 8
            int h;
            if(height[left] < height[right]){ //1 < 7
                h = height[left]; //1
                ++left; //2
            } else{
                h = height[right];
                --right;
            }
            int area = width*h; //8*1 = 8
            if(area > maxArea){//8 > 0


                maxArea = area; //8
            }
        }
        System.out.println(maxArea);
    }
}

public static Integer[][] compressMatrix( Integer[][] matrix ){
        int row = matrix.length;
        int col = matrix[0].length;

        int compRow = row/2;
        int compCol = col/2; //half of main



        Integer[][] compressed = new Integer[compRow][compCol]; // new array for compressed

        for (int i = 0; i < row; i+=2){ // for every 2
            for (int j = 0; j < col; j+=2){

                int sum = matrix[i][j] + matrix[i+1][j] + matrix[i][j+1]  + matrix[i+1][j+1];



                compressed[i/2][j/2] = sum; // storing
            }
        }
        return compressed;
    }
