    public static void mostWater( Integer[] height ){

        //TO DO
      int start = 0;
      int end = height.length - 1;
      int max = 0;
      
      while((end-start) != 0){


        int h = Math.min(height[start], height[end]);
        int w = end - start;
        int area = h*w;


        if(area > max) max = area;
        if(height[start] < height[end]) start++;
        else end--;
      }
      System.out.println(max);
    }

    public static Integer[][] compressMatrix( Integer[][] matrix ){
       
        //TO DO
      int rows = matrix.length;
      int cols = matrix[0].length;
      
      Integer[][] answer = new Integer[rows/2][cols/2];
      
      for(int r=0; r<rows; r+=2){
        int sum = 0;
        for(int c=0; c<cols; c+=2){


          sum = matrix[r][c] + matrix[r][c+1] + matrix[r+1][c] + matrix[r+1][c+1];
          answer[r/2][c/2] = sum;
        }
      }
      return answer;
    }

    public static void playGame( Integer[][] arena ){
      
        //TO DO
      int rows = arena.length;
      int cols = arena[0].length;
      int point = 0;
      
      for(int r=0; r<rows; r++){
        for(int c=0; c<cols; c++){
          if(arena[r][c]%50==0 && arena[r][c]!=0){
            if((r-1)>=0){
              if(arena[r-1][c]==2) {
                point += 2;
              }
            }
            if((r+1)<rows){
              if(arena[r+1][c]==2){
                point += 2;
              }
            }
            if((c-1)>=0){
              if(arena[r][c-1]==2) {
                point += 2;
              }
            }
            if((c+1)<cols){
              if(arena[r][c+1]==2){
                point += 2;
              }
            }
          }
        }
      }
      System.out.print("Points Gained: "+point);
      if(point>=10){
        System.out.println(". Your team has survived the game.");
      }
      else{
        System.out.println(". Your team is out.");
      }
    }
