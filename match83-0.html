public class Task01{ 
    public static void mostWater(Integer[] height) { 
        int left = 0; 
        int right = height.length - 1; 
        int maxArea = 0; 
        while (left < right) { 
            int heightContainer = Math.min(height[left], height[right]); 
            int width = right - left; 
            int area = heightContainer * width; 


            maxArea = Math.max(maxArea, area); 
 
            if (height[left] < height[right]) { 
                left++; 
            } else { 
                right--; 
            } 
        } 
        System.out.println(maxArea); 
    } 
}

 public class Task02{ 
    public static Integer rowRotation(Integer examWeek, String[][] matrix) { 
        int rowCount = matrix.length; 
        int rotation = (examWeek - 1) % rowCount; 
        for (int r = 0; r < rotation; r++) { 
            String[] last_Row = matrix[rowCount - 1]; 


            for (int i = rowCount - 1; i > 0; i--) { 
                matrix[i] = matrix[i - 1]; 
            } 
            matrix[0] = last_Row; 
        } 
        for (int i = 0; i < matrix.length; i++) { 
            System.out.print("| "); 
            for (int j = 0; j < matrix[i].length; j++) { 
                System.out.print(matrix[i][j] + "  | "); 
            } 
            System.out.println(); 


        } 
        for (int i = 0; i < matrix.length; i++) { 
            for (int j = 0; j < matrix[i].length; j++) { 
                if ("AA".equals(matrix[i][j])) { 
                    return i + 1; 
                } 
            } 
        } 
        return -1; 
    } 
}

public class Task03{ 


    public static Integer[][] compressMatrix(Integer[][] matrix) { 
        int rows = matrix.length / 2; 
        int cols = matrix[0].length / 2; 
        Integer[][] compressed_Matrix = new Integer[rows][cols]; 
        for (int i = 0; i < rows; i++) { 
            for (int j = 0; j < cols; j++) { 
                int sum = 0; 
                sum += matrix[i * 2][j * 2]; 
                sum += matrix[i * 2][j * 2 + 1]; 
                sum += matrix[i * 2 + 1][j * 2]; 


                sum += matrix[i * 2 + 1][j * 2 + 1]; 
                compressed_Matrix[i][j] = sum; 
            } 
        } 
        return compressed_Matrix; 
    } 
} 

public class Task04{ 


    public static void playGame(Integer[][] matrix) { 
        int rows = matrix.length; 
        int cols = matrix[0].length; 
        int total_Points = 0; 
        for (int i = 0; i < rows; i++) { 
            for (int j = 0; j < cols; j++) { 
                int val = matrix[i][j]; 
                if (val > 0 && val % 50 == 0) { 
                    int points = 0; 
                    if (i - 1 >= 0 && matrix[i - 1][j] == 2) points += 2; 
                    if (i + 1 < rows && matrix[i + 1][j] == 2) points += 2; 
                    if (j - 1 >= 0 && matrix[i][j - 1] == 2) points += 2; 
                    if (j + 1 < cols && matrix[i][j + 1] == 2) points += 2; 
                    total_Points += points; 
                } 
            } 
        } 


        if (total_Points >= 10) { 
            System.out.println("Points Gained: " + total_Points + ". Your team has survived the game."); 
        } else { 
            System.out.println("Points Gained: " + total_Points + ". Your team is out."); 
 
        } 
    } 
}
