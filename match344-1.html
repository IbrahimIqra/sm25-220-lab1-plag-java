    public static void mostWater( Integer[] height ) {

        if (height == null || height.length < 2){
            System.out.println(0);
            return;
        }

        int l = 0;                  
        int r = height.length - 1;  
        int max_Wat = 0;            
        
        while (l<r){
            int c_H;

            if (height[l]<height[r]){
                c_H = height[l];
            }

            else{
                c_H = height[r];
            }

            int w = r-l;

            int c_A = c_H*w;

            if (c_A>max_Wat){
                max_Wat = c_A;
            }

            if (height[l]<height[r]){
                l++; 
            }
            else{
                r--; 
            }
        }

        System.out.println(max_Wat);

    }

    public static Integer rowRotation( Integer examWeek, String[][] matrix ) {

        int n_rot = examWeek - 1;                   


        int n_row = matrix.length;                       

        for (int i=0; i<n_rot; i++){
            String[] l_r = matrix[n_row-1];      

            for (int j=n_row-2; j>=0; j--){
                matrix[j + 1] = matrix[j]; 
            }

            matrix[0] = l_r;

        }

       
        System.out.println("Rotated Seat Status:");
        
        for (int i=0; i<matrix.length; i++){
            String[] r = matrix[i];
            System.out.print("| ");
            
            for (int j=0; j<r.length; j++){
                String cell = r[j];
                System.out.print(cell);
                
                if (cell.length() == 1){
                    System.out.print("  | ");
                } 
                else{
                    System.out.print(" | ");
                }
            }
            
            System.out.println();

        }

        
        int f_row = -1;         
        


        for (int i=0; i<n_row; i++){
            
            for (int j=0; j<matrix[i].length; j++){
          
                if (matrix[i][j].equals("AA")){
                    f_row = i+1; 
                    break; 
                }
            }
            
            if (f_row != -1){
                break; 
            }
        }

        return f_row;

    }

    public static Integer[][] compressMatrix( Integer[][] matrix ){

        int rows = matrix.length;
        int cols = matrix[0].length;

        int n_row = rows/2;
        int n_col = cols/2;

        Integer[][] comp_mat = new Integer[n_row][n_col];        

        for (int i=0; i<n_row; i++){

            for (int j=0; j<n_col; j++){
                int r = i*2;
                int c = j*2;

                int sum = matrix[r][c] + matrix[r][c + 1] + matrix[r + 1][c] + matrix[r + 1][c + 1];

                comp_mat[i][j] = sum;

            }
        }

        return comp_mat;

    }

    public static void playGame( Integer[][] arena ){

        int t_p = 0;         
        int r = arena.length;
        int c = arena[0].length;

        for (int i=0; i<r; i++){

            for (int j=0; j<c; j++){

                if (arena[i][j]% 50 == 0 && arena[i][j] != 0){

                    if (i>0 && arena[i - 1][j] == 2){
                        t_p = t_p+2;
                    }

                    if (i<r-1 && arena[i + 1][j] == 2){
                        t_p = t_p+2;
                    }

                    if (j>0 && arena[i][j - 1] == 2){
                        t_p = t_p+2;
                    }

                    if (j<c-1 && arena[i][j + 1] == 2){
                        t_p = t_p+2;
                    }
                }
            }
        }



        if (t_p>=10){
            System.out.println("Points Gained: " + t_p + ". Your team has survived the game.");
        }
        else{
            System.out.println("Points Gained: " + t_p + ". Your team is out.");
        }
    }
