    public static void mostWater(Integer[] height) {
        int max_water = 0;
        int left_pointer = 0;
        int right_pointer = height.length - 1;

        while (left_pointer < right_pointer) {
            int width = right_pointer - left_pointer;
            int h = Math.min(height[left_pointer], height[right_pointer]);
            int current_water = width * h;


            max_water = Math.max(max_water, current_water);

            if (height[left_pointer] < height[right_pointer]) {
                left_pointer++;
            } else {
                right_pointer--;
            }
        }

        System.out.println(max_water);   
 }

public static Integer rowRotation(Integer examWeek, String[][] matrix) {
    
    examWeek=examWeek-1;//as we have to rotate the week given and then -1



    int row = matrix.length;
    for (int i = 0; i < examWeek; i++) {
        String[] lastr = matrix[row - 1]; 
        for (int j = row - 1; j > 0; j--) {
            matrix[j] = matrix[j - 1];
        }
        matrix[0] = lastr;
    }
    for (int i = 0; i < row; i++) {
        for (int j = 0; j < matrix[0].length; j++) {
            if (matrix[i][j].equals("AA")) {
                return i + 1;
            }
        }
    }
    return -1;
}
