 public static Integer[] mergeSortedArray( Integer[] arr1, Integer[] arr2 ){

         int n1 = arr1.length;
        int n2 = arr2.length;
        Integer[] merged = new Integer[n1 + n2];

        int i = 0, j = 0, k = 0;

                for (; i < n1 && j < n2; k++) {
            if (arr1[i] < arr2[j]) {
                merged[k] = arr1[i];
                i++;
            } else {
                merged[k] = arr2[j];
                j++;
            }
        }

    public static Integer[][] compressMatrix( Integer[][] matrix ){
        int rows = matrix.length;
        int cols =matrix[0].length;

        int r= rows/2;
        int c= cols/2;
        Integer[][]compressed = new Integer[r][c];
        int cl=0;
        int rw=0;
        for(int i=0; i< rows; i+=2){
           for(int j=0; j< cols;j+=2){


           int sum = matrix[i][j]+matrix[i+1][j]+matrix[i][j+1]+matrix[i+1][j+1];
           compressed [rw][cl]= sum;
          cl++;
           }
          rw++;


          cl=0; 
        }

               return compressed ;
    }

    public static void playGame( Integer[][] arena ){
      int row = arena.length;
      int col= arena[0].length;
      int sum =0;
      for(int i=0; i<row;i++) {
        for(int j=0;j<col;j++){


            if (arena[i][j]!=0 && arena[i][j]%50==0){
                if(i>0 && arena[i-1][j]==2){
                    sum+=2;
                
                }
                if(i<row-1 &&arena[i+1][j]==2){
                   sum+=2;   
                }
                 if(j>0 && arena[i][j-1]==2){
                   sum+=2;
                 }
                 if(j<col-1 &&arena[i][j+1]==2){
                    sum+=2;
                 }
            }
        }
      }
      if(sum>=10){
        System.out.println("Points Gained: " +sum+ ". Your team has survived the game.");
