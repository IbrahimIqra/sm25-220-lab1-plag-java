public static void mostWater(Integer[] height) {
    int L =0;
    int R =height.length-1;
    int area1=0;

    while (L<R) {
        int h;
        if(height[L]<height[R]) {

            h=height[L];

        } else {

            h=height[R];
        }

        int w=R-L;

        int area2=h*w;

        if (area2>area1) {
            area1=area2;
        }

        if (height[L]<height[R]) {

            L++;

        } else {
            
            R--;

        }
    }

    System.out.print(area1);
}

public static Integer rowRotation( Integer examWeek, String[][] matrix ) {


    int row =matrix.length;

    int col=matrix[0].length;


    int rot =(examWeek-1)%row;


    String[][] rotmat = new String[row][col];
    for (int x=0;x<row;x++) {

        int oldidx=(row-rot+x)%row;

        for (int y=0;y<col;y++) {

            rotmat[x][y]=matrix[oldidx][y];
        }
    }


    for (int a =0;a<row;a++) {



        System.out.print("|");

        for (int b=0;b<col;b++) {
            System.out.print(rotmat[a][b]+ " ");
            if (b==col-1)

                System.out.print(" |");
        }
        
        System.out.println();
    }


    for (int r = 0;r<row;r++) {
        for (int c =0;c<col;c++) {
            if ("AA".equals(rotmat[r][c])) {
                
                return r+ 1;
            }
        }


    }
    return examWeek;
}

public static Integer[][] compressMatrix(Integer[][] matrix) {

    int row= matrix.length;
    int col= matrix[0].length;


    int comprow=row/2;
    int compcol=col/2;

    Integer[][] comp = new Integer[comprow][compcol];


    for (int i =0; i< row;i=i+2) {
        for (int j=0;j<col;j=j+2) {
            int sum=0;
            
            sum = sum+matrix[i][j];
            
            sum = sum+matrix[i][j+1];
            sum =sum+matrix[i+1][j];
            sum =sum+matrix[i+1][j+1];

            
            comp[i/2][j/2]=sum;
        }
    }
    return comp;

}

public static void playGame(Integer[][] arena) {

    int total =0;
    int row =arena.length;
    int col =arena[0].length;



    int[] dir1 ={-1,1,0,0}; //for dir
    int[] dir2 ={0,0,-1,1};

    for (int x = 0; x<row;x++) {

        for (int y =0;y<col;y++) {

            int loc = arena[x][y];



            if (loc!=0&&loc%50==0&&loc!=0) {


                int count=0;


                for (int i =0; i<4;i++) {
                    int dir3 = x+dir1[i];

                    int dir4 = y+dir2[i];

                    if (dir3 >=0&&dir3<row&&dir4>=0&&dir4< col) {
                        
                        if (arena[dir3][dir4]!=null&&arena[dir3][dir4] ==2) {
                            
                            count++;}
                        
                    }
                }

                total = total+(count*2);
            }

        }
        
        

    }
    

    if (total>=10) {
        System.out.print("Point gained" + " "+ total+ " your team has survived.");
    } else {
        System.out.print("Point gained" + " "+ total+ " your team is out.");
    }
}
