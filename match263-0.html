public static void MaxWater(Integer[] height) {
    int left = 0;
    int right = height.length - 1;
    int maxArea = 0 ;

    while (left < right) {
        int h = min(height[left],height[right]);
        int w = right - left;
        int area = h * w;


        maxArea= max(maxArea, area) ;
        if (height[left] < height[right]) {
            left++ ;
        } else {
            right--;
        }
    }

    System.out.println(" Maximum water : " + maxArea);
}

public static void playGame(Integer[][] arena) {
    int r = arena.length;
    int c = arena[0].length;
    int totalPoints = 0;

    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++) {
            if (arena[i][j] != 0 && arena[i][j] % 50 == 0) {


                int adjacentTwos = 0;

                if(i > 0 && arena[i - 1][j] == 2) {
                    adjacentTwos++;
                }
             
                if (i < r- 1 && arena[i + 1][j] == 2) {
                    adjacentTwos++;
                }
               
                if (j > 0 && arena[i][j - 1] == 2) {
                    adjacentTwos++;
                }
               
                if (j < c - 1 && arena[i][j + 1] == 2) {
                    adjacentTwos++;
                }

                totalPoints += adjacentTwos * 2;
            }
        }
    }

    System.out.print("Points Gained : " + totalPoints + " . ");
    if (totalPoints >= 10 ) {
        System.out.println (" You team has survived the game .");
