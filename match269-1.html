public static void mostWater( Integer[] height ){
      int i=0; 
      int j= height.length-1;
      int maxArea=0;
      
      while (i<j){
        int Area= Math.min( height[i], height[j]) * (j-i);
        if( Area>maxArea){
          maxArea= Area;
        }
        if( height[i]< height[j]){
          i=i+1;
        }
        else{
          j=j-1;
        }
      }
      System.out.print(maxArea);
    }

 public static Integer rowRotation( Integer examWeek, String[][] matrix ){
      int row= matrix.length;
      int col= matrix[0].length;
      String lastRow[]= new String[col];
      
      for( int i=0; i<examWeek;i++){


        for( int j=0; j< col; j++){
          lastRow[j] = matrix[row-1][j];
        }
        for( int x= row-2; x>=0; x--){
          for( int y=0; y<col;y++){
            matrix[x+1][y]= matrix[x][y];
          }
        }
        for( int j=0; j< col; j++){
          matrix[0][j]= lastRow[j];
        }
        }
        int newRow= 0;
        for( int x=0; x<row; x++){
          for ( int y= 0; y<col; y++){
            if( matrix[x][y].equals ("AA")){
              newRow=x;
            }
          }
        }
        
        return newRow-1;
       
    }

 public static Integer[][] compressMatrix( Integer[][] matrix ){
      int row= matrix.length;
      int col= matrix[0].length;
      Integer[][]newMatrix = new Integer [row/2][col/2];
      for( int i=0; i<row; i+=2){
        for( int j=0; j<col; j+=2){
          int sum= matrix[i][j] + matrix[i][j+1] + matrix[i+1][j] + matrix[i+1][j+1];
          newMatrix[i/2][j/2]= sum;
        }
      }
      return newMatrix;
    }
