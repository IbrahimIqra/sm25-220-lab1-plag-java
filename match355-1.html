public static void mostWater( Integer[] height ){ 
    int l=0;
    int minH=0;
    int r=height.length-1;


    int max= 0;
while(l<r){
        int width=r-l;
        if(height[l]>height[r]){
        minH=height[r];
        }
        else{
          minH=height[l];
        }
        int a=width*minH;
if(a>max){
            max=a;
        }
        if(height[l]<height[r]){
            l++;
        } else{
            r--;
        }
    }
System.out.println(max);
}

    public static Integer[][] compressMatrix( Integer[][] matrix ){
      Integer row=matrix.length;
      Integer col=matrix[0].length;
      Integer cr=row/2;
      Integer cc=col/2;
      Integer[][] c=new Integer[cr][cc];
      
      for(int k=0;k<row;k+=2){
        for(int i=0;i<col;i+=2){


           int t=matrix[k][i] + matrix[k][i + 1] + matrix[k+1][i]+matrix[k+1][i+1];
           c[k/2][i/2]=t;
        }
      }
        return c;
    }
