public static void mostWater( Integer[] height ){


        int max = 0;
        int left = 0;
        int right = height.length-1;
        while (left<right) {
            int width = right-left;
            int minHeight = 0;


            if (height[left]<height[right]) {
                minHeight = height[left];
            }
            else {
                minHeight = height[right];
            }
            int area = width*minHeight;
            if (area>max) {
                max = area;
            }
            if (height[left]<height[right]) {
                left++;
            }
            else {
                right--;
            }
        }
        System.out.println(max);


 }

public static Integer rowRotation( Integer examWeek, String[][] matrix ){




        Integer rotationCount = examWeek - 1;
        Integer rowCount = matrix.length;


        Integer examRow = -1;


        for (int i = 0; i < rotationCount; i++) {
            String[] tempRow = matrix[rowCount-1];
            for (int j = rowCount-1; j > 0; j--) {
                matrix[j] = matrix[j-1];
            }
            matrix[0] = tempRow;
        }


        for (int i = 0; i < rowCount; i++) {
            for (int j = 0; j < matrix[i].length; j++) {
                if (matrix[i][j].equals("AA")) {
                    examRow = i;
                    break;
                }
            }
            if (examRow!=-1) break;
        }


        Arr.print2D(matrix);


        return examRow+1;
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){
        
        Integer rows = matrix.length;
        Integer cols = matrix[0].length;


        Integer compRows = rows/2;


        Integer compCols = cols/2;


        Integer[][] compressed = new Integer[compRows][compCols];


        for (int i = 0; i < rows; i+=2) {
            for (int j = 0; j < cols; j+=2) {


                int sum = matrix[i][j] + matrix[i + 1][j] + matrix[i][j + 1] + matrix[i + 1][j + 1];
                compressed[i / 2][j / 2] = sum;
            }
        }


        return compressed;
    }

public static void playGame( Integer[][] arena ){


        
        int rows = arena.length;
        int cols = arena[0].length;
        int total = 0;


        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                Integer val = arena[i][j];


                if (val != null && val % 50 == 0 && val != 0) {


                    if (i > 0 && arena[i - 1][j] != null && arena[i - 1][j] == 2) total += 2;
                    if (i < rows - 1 && arena[i + 1][j] != null && arena[i + 1][j] == 2) total += 2;
                    if (j > 0 && arena[i][j - 1] != null && arena[i][j - 1] == 2) total += 2;


                    if (j < cols - 1 && arena[i][j + 1] != null && arena[i][j + 1] == 2) total += 2;
                }
            }
        }


        System.out.print("Points Gained: " + total + ". ");
        if (total >= 10) {
        System.out.println("Your team has survived the game.");
        } else {
            System.out.println("Your team is out.");
        }
    }
