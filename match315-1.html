 public static void mostWater( Integer[] height ){
        int maxcontain = 0;
        int left = 0;
        int right = height.length - 1;

        while (right > left) {
            int distance = right - left;
            int currentHeight = Math.min(height[left], height[right]);// get the minimum height of the two lines
            int area = distance * currentHeight; // calculate the area of the container


            maxcontain = Math.max(maxcontain, area);

            if (height[left] < height[right]) {
                left++;
            } else {
                right--;
            }
        }
        System.out.print(maxcontain);
    }

    public static Integer[][] compressMatrix( Integer[][] matrix ){
        int rows = matrix.length;
        int cols = matrix[0].length;
        Integer[][] compressedMatrix = new Integer[rows / 2][cols / 2];
        for (int i = 0; i < rows; i += 2) {
            for (int j = 0; j < cols; j += 2) {


                compressedMatrix[i / 2][j / 2] = matrix[i][j] + matrix[i][j + 1] + matrix[i + 1][j] + matrix[i + 1][j + 1];
            }
        }
        return compressedMatrix;

    }
