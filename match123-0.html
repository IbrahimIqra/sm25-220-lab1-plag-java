    public static void mostWater( Integer[] arr ){
        int totArea = 0;
        int left = 0;
        int right = arr.length - 1;

        while (left < right) {


            int width = right - left;
            int minHeight;
            if (arr[left] < arr[right]) {
                minHeight = arr[left];
            } else {
                minHeight = arr[right];
            }
            int area = width * minHeight;
           if (area>totArea) {
            totArea = area;
           }
            if (arr[left] < arr[right]) {
                left++;
            } else {
                right--;
            }
        }

        System.out.print(totArea);

    }
