  public static void mostWater( Integer[] height ){


        int maxArea = 0;
        int i = 0;
        int j = height.length-1;
        
        while (i<j){
            int area = 0;
            if(height[i]<=height[j]){               
                area = height[i]*(j-i);
                i++;                                       
            }
            else if(height[j]<=height[i]){
                 area = height[j]*(j-i);
                 j--;            
            }
            if(area==0 || area>maxArea){
                maxArea = area;            
            }                        
        }              


        System.out.println(maxArea);
    }

 public static Integer rowRotation( Integer examWeek, String[][] matrix ){
        
        int rotate=examWeek-1;
        int row=matrix.length;
        int col=matrix[0].length;
               
       for(int k=0;k<rotate;k++){
            for(int i=row-1;i>0;i--){
                for(int j=0;j<col;j++){
                    String temp=matrix[i][j];

                    matrix[i-1][j]=temp;                                    
                }                        
            } 
        }

  public static Integer[][] compressMatrix(Integer[][] matrix) {
        int row = matrix.length;
        int col = matrix[0].length;
        
        int halfRow = row/2;
        int halfCol = col/2;
        Integer [][] newArr = new Integer [halfRow][halfCol];
        int i=0;
        int k=0;
        
        while(i<row){
            int t =0;
            for(int j=0;j<col;j+=2){


                int sum=matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+matrix[i+1][j+1];
                newArr[k][t++]=sum;                        
            }
            ++k;
            i+=2;               
        }
        return newArr;
    }

 public static void playGame(Integer[][] arena) {
        int row=arena.length;
        int col=arena[0].length;
        int points=0;
        
        for(int i=0;i<row;i++){
            for(int j=0;j<col;j++){


                if(arena[i][j]%50==0 && arena[i][j] !=0){                
                    if(i>0 && arena[i-1][j]==2){
                        points+=2;
                    }
                    if(i<row-1 && arena[i+1][j]==2){
                        points+=2;
                    }
                    if(j>0 && arena[i][j-1]==2){
                        points+=2;
                    }    
                    if(j<col-1 && arena[i][j+1]==2){
                        points+=2;
                    }                                
                }
            }            
        }                       
        System.out.print("Points Gained: "+points+". ");        
        if(points>=10){
            System.out.println("Your team has survived the game.");
        }
        else{
            System.out.println("Your team is out.");
        }
    }
