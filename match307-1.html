    public static void mostWater(Integer[] height) {
        int mswt = 0;
        int newhgt = 0;
        for (int s = 0; s < height.length; s++) {
            for (int t = s + 1; t < height.length; t++) {
                if (height[s] < height[t]) {
                    newhgt = height[s];
                } else {
                    newhgt = height[t];
                }
                int newwdh = t - s;
                int max_areaa = newhgt * newwdh;
                if (max_areaa > mswt) {
                    mswt = max_areaa;
                }
            }
        }
        System.out.println(mswt);
    }

    public static Integer rowRotation(Integer examWeek, String[][] matrix) {
        int rotationTotal = examWeek - 1;
        int r = matrix.length;
        int c = matrix[0].length;



        for (int s = 0; s < rotationTotal; s++) {
            String[] temp = matrix[r - 1];
            for (int t = r - 1; t > 0; t--) {
                matrix[t] = matrix[t - 1];
            }
            matrix[0] = temp;
        }
        for (int o = 0; o < r; o++) {
            for (int p = 0; p < c; p++) {
                if (matrix[o][p] == ("AA")) {
                    return o + 1;
                }
            }
        }
        return -1;
    }

  public static Integer[][] compressMatrix(Integer[][] matrix) {
    int rw = matrix.length;
    int clmn = matrix[0].length;
    Integer[][] compress = new Integer[rw / 2][clmn / 2];
    // For this task you'll need to create new 2D array
    for (int s = 0; s < rw; s += 2) {
      for (int t = 0; t < clmn; t += 2) {
        int summation = matrix[s][t] + matrix[s][t + 1] + matrix[s + 1][t] + matrix[s + 1][t + 1];
        compress[s / 2][t / 2] = summation; 
      }
    }

    return compress;
  }
