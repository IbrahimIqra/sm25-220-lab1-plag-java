 public static void mostWater( Integer[] height ){
      int c= 0;


        Integer[]arr=new Integer[height.length*height.length];
               for(int i=0;i<height.length;i++){
                 
                 for(int j=0;j<height.length; j++){
                int w= 0;
                
                if ( j>i){
                    w= j-i;
                }
                else if(i>j){
                  w =i- j;}
                arr[c] = w*Math.min(height[i],height[j]);
                c++;
                }}
               


        int m= -1;
        
        for(int k=0;k<arr.length; k++){
            m= Math.max(m,arr[k]);
        }
        
        System.out.println(m);
        }
