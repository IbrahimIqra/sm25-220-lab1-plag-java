public class AssgnTask1 {

    public static void mostWater( Integer[] height ) {
        Integer n = height.length;
        Integer area = 0;
        Integer areamax = 0;

        for ( int i = 0; i < n; i++ ) {
            for ( int j = 0; j < n; j++ ) {
                if ( height[j] != height[i] || height[j] < height[i] ) {
                    area = height[j] * ( j - i - 1 );
                } else {
                    area = height[i] * ( j - i + 1 );
                }
            }
            if ( area > areamax ) {
                areamax = area;
            }
        }

        System.out.println(areamax);
    }
}
