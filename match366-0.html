public static Integer[][] compressMatrix( Integer[][] matrix ){

    int rows = matrix.length;
    int cols = matrix[0].length;

    Integer [][] out = new Integer [rows/2][cols/2];

    for(int i = 0; i < rows; i+=2){
        for(int j = 0; j < cols; j+=2){
            Integer sum = matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+matrix[i+1][j+1];
            out[i/2][j/2] = sum;
        }
    }

    return out;
}

public static void playGame( Integer[][] arena ){

    int rows = arena.length;
    int cols = arena[0].length;

    int scoreTotal = 0;
    for(int i = 0; i < rows; i++){
        for(int j = 0; j < cols; j++){
            if(arena[i][j]%50 == 0 && arena[i][j] != 0) scoreTotal += calcScore(arena, i, j);
        }
    }

    String s = "";
    if(scoreTotal < 10) s = "is out.";
    else s = "has survived the game.";
    System.out.printf("Points Gained: %d. Your team %s", scoreTotal, s);

}

private static int calcScore(Integer[][] arena, int i, int j) {
    int sum = 0;
    if(i != 0) sum += arena[i-1][j] == 2 ? 2 : 0;
    if(i != arena.length-1) sum += arena[i+1][j] == 2 ? 2 : 0;
    if(j != 0) sum += arena[i][j-1] == 2 ? 2 : 0;
    if(j != arena[0].length-1) sum += arena[i][j+1] == 2 ? 2 : 0;

    return sum;
}
