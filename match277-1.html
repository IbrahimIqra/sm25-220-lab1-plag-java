public static void mostWater( Integer[] height ){
    int arr_len = height.length;
    int max_area = 0;
    for(int i=0; i<arr_len; i++){
        for(int j=i; j<arr_len; j++){
            int h;


            if(height[i]<height[j]){
                h = height[i];
            }
            else {
                h = height[j];
            }
            int l = j - i;
            int area = h * l;
            if(area > max_area){
                max_area = area;
            }
        }
    }
    System.out.println(max_area);
}

public static Integer[][] compressMatrix( Integer[][] matrix ){
    int row_length = matrix.length;
    int col_length = matrix[0].length;
    Integer[][] matrix2 = new Integer[row_length/2][col_length/2];
    for(int i=0; i<row_length; i+=2){
        for(int j=0; j<col_length; j+=2){
            int sum = matrix[i][j] + matrix[i+1][j] + matrix[i][j+1] + matrix[i+1][j+1];


            matrix2[i/2][j/2] = sum;
        }
    }
    return matrix2;
}

public static void playGame( Integer[][] arena ){
    int point = 0;
    int row_length = arena.length;
    int col_length = arena[0].length;
    for(int i=0;i<row_length; i++){
        for(int j=0; j<col_length; j++){
            if(arena[i][j] >= 50 && arena[i][j] % 50 == 0){
                //calculate point


                if(i<row_length-1 && arena[i+1][j] == 2){
                    point += 2;
                }
                if(i>0 && arena[i-1][j] == 2){
                    point += 2;
                }
                if(j<col_length-1 && arena[i][j+1] == 2){
                    point += 2;
                }
                if(j>0 && arena[i][j-1] == 2){
                    point += 2;
                }
            }
        }
    }
    if(point >= 10){
        System.out.printf("Points Gained: %d. Your team has survived the game.\n", point);
    }
    else {
        System.out.printf("Points Gained: %d. Your team is out.\n", point);
    }
}
