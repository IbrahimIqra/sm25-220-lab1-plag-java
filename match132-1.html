   public static Integer rowRotation(Integer examWeek, String[][] matrix) {
    int n = matrix.length;
    int k = (examWeek % n) - 1; 

    for (int start = 0, end = n - 1; start < end; start++, end--) {
        String[] temp = matrix[start];
        matrix[start] = matrix[end];
        matrix[end] = temp;
    }

 
    for (int start = 0, end = k - 1; start < end; start++, end--) {
        String[] temp = matrix[start];
        matrix[start] = matrix[end];
        matrix[end] = temp;
    }


    for (int start = k + 1, end = n - 1; start < end; start++, end--) {


        String[] temp = matrix[start];
        matrix[start] = matrix[end];
        matrix[end] = temp;
    }

    for (int row = 0; row < n; row++) {
        for (int col = 0; col < matrix[row].length; col++) {
            if ("AA".equals(matrix[row][col])) {
                return row + 1;
            }
        }
    }

    return null;
}

public static Integer[][] compressMatrix(Integer[][] matr) {
        int rows = matr.length;
        int cols = matr[0].length;

        Integer[][] resultMatrix = new Integer[rows / 2][cols / 2];


           int rw = 0, cl = 0;
           
        for (int i = 0; i < rows; i += 2) {
            for (int j = 0; j < cols; j += 2) {
                int sum = matr[i][j] + matr[i][j + 1]


                        + matr[i + 1][j] + matr[i + 1][j + 1];
                resultMatrix[i / 2][j / 2] = sum;
                cl++;
            }
            rw++;
        }

        return resultMatrix;
    }

 public static void playGame(Integer[][] arena) {
    int totalRows = arena.length;
    int totalCols = arena[0].length;
    int points = 0;

    for (int r = 0; r < totalRows; r++) {
        for (int c = 0; c < totalCols; c++) {
            Integer cell = arena[r][c];

            if (cell != 0 && cell % 50 == 0) {
                if (r - 1 >= 0 && arena[r - 1][c] == 2) {
                    points += 2;
                }
                if (r + 1 < totalRows && arena[r + 1][c] == 2) {
                    points += 2;
                }
                if (c - 1 >= 0 && arena[r][c - 1] == 2) {
                    points += 2;
                }


                if (c + 1 < totalCols && arena[r][c + 1] == 2) {
                    points += 2;
                }
            }
        }
    }

    if (points >= 10) {
        System.out.println("Points Gained: " + points + ". Your team has survived the game.");
    } else {
        System.out.println("Points Gained: " + points + ". Your team is out.");
    }
}
