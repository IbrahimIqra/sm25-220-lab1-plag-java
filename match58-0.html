  public static void mostWater( Integer[] height ){
        int left = 0;
        int right = height.length-1;
        int max = 0;
        while(left<right){
            int width = right-left;
            int length = 0;


            if(height[left]<height[right]){
                length = height[left];
            }
            else{
                length = height[right];
            }
            int area = width*length;


            if(area>max){
                max = area;
            }
            if(height[left]<height[right]){
                left++;
            }
            else if(height[left]>height[right]){
                right--;
            }
            else{
                left++;
                right--;
            }
        }
       System.out.println(max);




        }

 public static Integer rowRotation( Integer examWeek, String[][] matrix ){
        int row = matrix.length;
        int column = matrix[0].length;


        int rotation = (examWeek-1)%row;


       for(int i=0; i<rotation; i++){
        String[] lastRow = matrix[row-1];
        for(int j = row-1; j>0; j--){
            matrix[j] = matrix[j-1];
        }
        matrix[0] = lastRow;
       }




       for(int i=0; i<row; i++){
        for(int j=0; j<column; j++){
            if(matrix[i][j].equals("AA")){
                return i+1;
            }
        }
       }

return-1;
}

  public static Integer[][] compressMatrix( Integer[][] matrix ){
        if (matrix == null || matrix.length == 0) {
           return null;
       }
           
           int rows = matrix.length;
           int columns= matrix[0].length;
           


           Integer array [][] = new Integer [rows/2][columns/2];
           
           for(int i=0; i<rows; i+=2) {
                   
                   for(int j=0; j<columns; j+=2) {
                           
                           array[i/2][j/2] = matrix [i][j] + matrix [i][j+1] + matrix[i+1][j] + matrix[i+1][j+1];
                   }
           }
     return array;
}

 public static void playGame( Integer[][] arena ){
        int row = arena.length;
        int column = arena[0].length;
        int total = 0;


        for(int i=0; i<row; i++){
            for(int j=0; j<column; j++){
                if(arena[i][j]%50==0 && arena[i][j]!=0){
                    int count = 0;
                    if(i>0 && arena[i-1][j]==2){
                        count++;
                    }
                    if(i<row-1 && arena[i+1][j]==2){
                        count++;
                    }
                    if(j>0 && arena[i][j-1]==2){
                        count++;
                    }
                    if(j<column-1 && arena[i][j+1]==2){
                        count++;
                    }


                    total+=count*2;
                }
            }
        }


        if(total<10){
            System.out.println("Points Gained: "+ total +" .Your team is out.");
        }
        else{
            System.out.println("Points Gained: "+ total +" .Your team has survived the game");
        }


}
