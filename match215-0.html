public static void mostWater( Integer[] height ){


       //TO DO
       int i=0;
       int j=height.length-1;
       int max=0;
       while (i<j){
           int h;
           if(height[i]<height[j]){
               h=height[i];
           }
           else{
               h=height[j];
           }
           int wide =j-i;


           int area= h*wide;
          
           if(area>max){
               max=area;
           }
           if(height[i]<height[j]){
               i++;
           }
           else{
               j--;
           }
       }


       System.out.print(max);


   }
