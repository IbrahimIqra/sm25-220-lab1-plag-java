public static Integer[][] compressMatrix(Integer[][] array) {
    int rowCount = array.length;
    int colCount = array[0].length;
    int newRow = rowCount / 2;
    int newCol = colCount / 2;
    Integer[][] result = new Integer[newRow][newCol];

    for (int i = 0; i < newRow; i++) {
        for (int j = 0; j < newCol; j++) {
            int sum = 0;
            sum += array[i * 2][j * 2];
            sum += array[i * 2][j * 2 + 1];
            sum += array[i * 2 + 1][j * 2];
            sum += array[i * 2 + 1][j * 2 + 1];
            result[i][j] = sum;
        }
    }

    return result;
}

public static void playGame(Integer[][] arena) {
    int totalPoints = 0;
    int rows = arena.length;
    int cols = arena[0].length;



    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            if (arena[i][j] % 50 == 0 && arena[i][j] != 0) {


                if (i - 1 >= 0 && arena[i - 1][j] == 2) totalPoints++;
                if (i + 1 < rows && arena[i + 1][j] == 2) totalPoints++;
                if (j - 1 >= 0 && arena[i][j - 1] == 2) totalPoints++;
                if (j + 1 < cols && arena[i][j + 1] == 2) totalPoints++;
            }
        }
    }

    System.out.print("Points Gained: " + totalPoints + ". ");
    if (totalPoints >= 10) {
        System.out.print("Your team has survived the game.");
    } else {
        System.out.print("Your team is out.");
    }
}
