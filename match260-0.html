    public static void mostWater( Integer[] height ){
        //TO DO
        int largerArea = 0;
        int l = height.length ;
        for(int i=0 ; i<l ; i++){
            for(int j=i+1 ; j<l ; j++){
                int w = j-i ;
                int smallerHeight ;
                if(height[i]<height[j]){
                    smallerHeight = height[i] ;
                }
                else{


                    smallerHeight = height[j] ;
                }
                int area = w*smallerHeight ;
                if(area>largerArea){
                    largerArea = area ;
                }
            }
        }
        System.out.println(largerArea) ;
    }

    public static Integer[][] compressMatrix( Integer[][] matrix ){

        //For this task you'll need to create new 2D array
        
        //TO DO
        int r = matrix.length ;
        int c = matrix[0].length ;
        int rowNum = r/2 ;
        int colNum = c/2 ;
        int sum = 0 ;
        Integer[][] result = new Integer[rowNum][colNum] ;
        for(int i=0 ; i<rowNum ; i++){
            for(int j=0 ; j<colNum ; j++){
            int r2 = 2*i ;
            int c2 = 2*j ;
            sum = matrix[r2][c2]+matrix[r2+1][c2]+matrix[r2][c2+1]+matrix[r2+1][c2+1] ;


            result[i][j] = sum ;
            }
        }

        //remove the line below and return the newly created Compressed 2D matrix
        return result ;
    }

    public static void playGame( Integer[][] arena ){
        
        //For this task you don't need to create any new 2D array
        //just print the result inside the function
        
        //TO DO
        int r = arena.length ;
        int c = arena[0].length ;
        int count = 0 ;
        for(int i=0 ; i<r ; i++){
            for(int j=0 ; j<c ; j++){
                int point = arena[i][j] ;
                if(point>0 && point%50==0){
                    int gained = 0 ;


                    if(j-1>=0 && arena[i][j-1]==2){
                        gained++ ;
                    }
                    if(j+1<r && arena[i][j+1]==2){
                        gained++ ;


                    }
                    if(i-1>=0 && arena[i-1][j]==2){
                        gained++ ;
                    }
                    if(i+1<r && arena[i+1][j]==2){
                        gained++ ;
                    }
                    count+=gained ;
                }  
            }
        }
        int total = 2*count ;


        if(total>=10){
            System.out.println("Points Gained: " + total + ". Your team has survived the game.") ;
        }
        else{
            System.out.println("Points Gained: " + total + ". Your team is out.") ;
        }
    }
