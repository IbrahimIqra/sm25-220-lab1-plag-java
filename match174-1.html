public static void mostWater(Integer[] height) {
    int maxArea = 0;
    int i = 0;
    int j = height.length - 1;

    while (i < j) {
        int minHeight = Math.min(height[i], height[j]);
        int width = j - i;
        int area = minHeight * width;

        if (area > maxArea) {
            maxArea = area;
        }

        // Move the smaller height inward
        if (height[i] < height[j]) {
            i++;
        } else {
            j--;
        }
    }

    System.out.println(maxArea);
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){

    int r = matrix.length;
      int c = matrix[0].length;
    int compR = r / 2;
    int compC = c/ 2;



    Integer[][] compressed = new Integer[compR][compC];

    for (int i = 0; i < r; i += 2) {
        for (int j = 0; j <c; j += 2) {
            
            int sum = matrix[i][j] + matrix[i][j + 1]+ matrix[i + 1][j]   + matrix[i + 1][j + 1];
            
            compressed[i/2][j/2] = sum;
        }
    }

    return compressed;
}

 public static void playGame( Integer[][] arena ){
        int r= arena.length;
        int c= arena[0].length;

        int totalTwo = 0;

        for (int i=0; i< r; i++) {
            for (int j=0; j<c; j++) {
                int v = arena[i][j];



                if (v>0 && v%50 == 0) {
                    int numOfTwo = 0;

                    if (i-1 >= 0 && arena[i-1][j] == 2) {
                        numOfTwo++;
                    }
                    if (i+1 <r && arena[i+1][j] == 2) {
                        numOfTwo++;
                    }
                    if (j-1 >= 0 && arena[i][j-1] == 2) {
                        numOfTwo++;
                    }
                    if (j+1< c && arena[i][j+1] == 2) {
                        numOfTwo++;
                    }
                    totalTwo += numOfTwo;
                }
            }
        }

 


        int points = totalTwo * 2;
   
        System.out.print("Points Gained: " + points + ". ");
        if (points>=10) {
            System.out.println("Your team has survived the game.");
        } else {
            System.out.println("Your team is out.");
        }
    }
