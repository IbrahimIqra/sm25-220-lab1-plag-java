int row = arena.length;
      int col = arena[0].length;
      int totalPoint = 0;

    for (int r = 0; r < row; r++){
        for (int c = 0; c < col; c++){
            if (arena[r][c] != 0){
                if (arena[r][c] % 50 == 0){ 


                    int point = 0;
                   if (r - 1 >= 0){
                       if (arena[r - 1][c] == 2){
                            point++;
                        }
                    }

                    if (r + 1 < row){
                        if (arena[r + 1][c] == 2){
                            point++;
                        }
                    }
                    if (c - 1 >= 0){
                        if (arena[r][c - 1] == 2){
                            point++;
                        }
                    }

                    if (c + 1 < col){
                        if (arena[r][c + 1] == 2){
                            point++;
                        }
                    }

                    totalPoint += point * 2;
                }
            }
        }
    }

    System.out.print("Points Gained: " + totalPoint + ". ");
    if (totalPoint >= 10){
        System.out.print("Your team has survived the game.\n");
    }
