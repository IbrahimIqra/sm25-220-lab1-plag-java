public static void mostWater( Integer[] height ){
        int mA=0;
        int i=0;
        int j=height.length-1;


        while(i<j){
            int b;
            if(height[i]<height[j]){
                b=height[i];
            }else{


                b=height[j]; 
            } 
            
            int l=j-i;
            int a=l*b;


            if(a>mA){
                mA=a;
            }


            if(height[i]<height[j]){
                i++;
            } else{
                j--;
            }
            
        }


        System.out.println(mA);


    }

public static Integer[][] compressMatrix(Integer[][] matrix) {
        int row = matrix.length;
        int col = matrix[0].length;
        int rr = row / 2;
        int cc = col / 2;


        Integer[][] comp = new Integer[rr][cc]; 


        int rrr = 0;


        int ccc = 0;


        for (int i = 0; i < row; i += 2) {
            for (int j = 0; j < col; j += 2) {
                int sum = matrix[i][j] + matrix[i + 1][j] +
                          matrix[i][j + 1] + matrix[i + 1][j + 1];
                comp[rrr][ccc] = sum;
                ccc++;
            }
            rrr++;
            ccc = 0;
        }


        return comp; 
    }

public static void playGame( Integer[][] arena ){
        int row= arena.length;
        int col= arena[0].length;
        int sum =0;
        for(int i=0; i<row; i++){
            for(int j=0; j<col; j++){
                if(arena[i][j]!=0 && arena[i][j]%50==0){
                    if(i>0 && arena[i-1][j]==2){
                        sum+=2;
                    }
                    if(i<row-1 && arena[i+1][j]==2){
                        sum+=2;
                    }
                    if(j>0 && arena[i][j-1]==2){
                        sum+=2;
                    }
                    if(j<col-1 && arena[i][j+1]==2){
                        sum+=2;
                    }
                }
            }
        }
        if(sum>=10){
            System.out.println("Points Gained: "+sum+" .Your team has survived the game.");
        }
        else{
            System.out.println("Points gained: "+sum+". Your team is out.");
        }


    }
