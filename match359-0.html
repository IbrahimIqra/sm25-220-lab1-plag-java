    public static void mostWater( Integer[] height ){
        int max = 0;
        int c1 = 0;
        int c2 = height.length-1;

        while (c1<c2) {
            int width = c2 - c1;
            int min = 0;


            if (height[c2] < height[c1]) {
                min = height[c2];
            }
            else {
                min = height[c1];
            }
            int area = width * min;
            if (area > max) {


                max = area;
            }
            if (height[c1]<height[c2]) {
                c1++;
            }
            else {
                c2--;
            }
        }
        System.out.println(max);
    }

    public static Integer rowRotation( Integer examWeek, String[][] matrix ){
        int a = matrix.length;
        int i = 0;
        int j = matrix.length-1;
        int m = (examWeek % a)-1;

        while(i<j){
            String[] temp = matrix[i];
            matrix[i] = matrix[j];
            matrix[j] = temp;
            i++;
            j--;
        }

        i = 0;
        j = m-1;
        while(i<j){
            String[] temp = matrix[i];
            matrix[i] = matrix[j];
            matrix[j] = temp;
            i++;
            j--;
        }

        i = m+1;
        j = matrix.length-1;
        while(i<j){
            String[] temp = matrix[i];
            matrix[i] = matrix[j];
            matrix[j] = temp;
            i++;
            j--;
        }
        for(int r=0; r<matrix.length; r++){
            for(int c=0; c<matrix[r].length; c++){
            if(matrix[r][c]=="AA"){
                return r+1;
            }
            }
        }


        return null;
    }

    public static Integer[][] compressMatrix( Integer[][] matrix ){
        int rows = matrix.length;
        int columns = matrix[0].length;
        int row = rows/2;
        int col = columns/2;
        int[][] compress = new Integer[row][col];  

        int rw=0;


        int cl=0;
        for (int r=0; r<row; r+=2) {
            for (int c=0; c<col; c+=2) {
                int sum = matrix[r][c]+matrix[r][c+1]+matrix[r+1][c]+matrix[r+1][c+1];
                compress[rw][cl] = sum;
                cl++;
            }
            rw++;
            cl=0;
        }


        return compress;
    }

    public static void playGame( Integer[][] arena ){
        int row = arena.length;
        int col = arena[0].length;
        int total = 0;

        for (int r=0; r<row; r++) {
            for (int c=0; c<col; c++) {
                if (arena[r][c]%50==0 && arena[r][c]!=0) {
                    if (r>0 && arena[r-1][c]==2) {
                        total+=2;
                    }
                    if (r<row-1 && arena[r+1][c]==2) {
                        total+=2;
                    }
                    if (c>0 && arena[r][c-1]==2) {
                        total+=2;
                    }
                    if (c<col-1 && arena[r][c+1]==2) {
                        total+=2;
                    }
                }
            }
        }
        if (sum >= 10) {
            System.out.println("Points Gained: " + sum + ". Your team has survived the game.");
          }
           else {
            System.out.println("Points Gained: " + sum + ". Your team is out.");
          }
    }
