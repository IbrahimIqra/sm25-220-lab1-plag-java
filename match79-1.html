public class SeatRotation {
    public static Integer rowRotation(Integer examWeek, String[][] seatStatus) {
        int allRows = seatStatus.length;
        int rotation = (examWeek - 1) % allRows;
        for (int i= 0; i < rotation; i++) 
        {
            String[] lastrow = seatStatus[allRows - 1];
            for (int j = allRows - 1; j > 0; j--) 
            {
                seatStatus[j] = seatStatus[j- 1];
            }
            seatStatus[0] = lastrow;
        }
        for (int x = 0; x < allRows; x++)
        {
            System.out.print("| ");
            for (int y = 0; y < seatStatus[0].length; y++) 
            {
                System.out.print(seatStatus[x][y] + "  | ");
            }
            System.out.println();
        }
        int rowAA = -1;
        for (int i = 0; i < allRows; i++)
        {
            for (int j = 0; j < seatStatus[0].length; j++) 
            {
                if (seatStatus[i][j].equals("AA")) {
                    rowAA = i + 1;
                    break;
                }
            }
            if (rowAA != -1) {
                break;
            }
        }

        return rowAA;
    }

public class MatrixCompression {


    public static Integer[][] compressMatrix(Integer[][] mat) {
        int allRows = mat.length;
        int allCols = mat[0].length;
        int newRows = allRows / 2;
        int newCols = allCols / 2;
        Integer[][] result = new Integer[newRows][newCols];
        int row1 = 0;
        for (int i = 0; i < allRows; i = i + 2) 
        {
            int col1 = 0;


            for (int j = 0; j < allCols; j = j + 2) 
            {
                int first = mat[i][j];
                int second = mat[i][j + 1];
                int third = mat[i + 1][j];
                int fourth = mat[i + 1][j + 1];
                int totalSum = first + second + third + fourth;
                result[row1][col1] = totalSum;
                col1 = col1 + 1;
            }
            row1 =row1 + 1;
        }
        return result;
    }

public static void playGame(Integer[][] arena) {
        int points = 0;
        int row = arena.length;
        int col = arena[0].length;

        for (int i = 0; i < row; i++) 
        {
            for (int j = 0; j < col; j++)
            {
                int val = arena[i][j];
                if (val != 0 && val % 50 == 0) 
                {
                    int count = 0;
                    if (i > 0 && arena[i - 1][j] == 2) {
                        count++;
                    }
                    if (i < row - 1 && arena[i + 1][j] == 2) {
                        count++;
                    }
                    if (j > 0 && arena[i][j - 1] == 2) {
                        count++;
                    }
                    if (j < col - 1 && arena[i][j + 1] == 2) {
                        count++;
                    }

                    points += count * 2;
                }
            }
        }

        System.out.print("Points Gained: " + points + ". ");
        if (points >= 10) {
            System.out.println("Your team has survived the game.");
        }
        else {
            System.out.println("Your team is out.");
        }
    }
