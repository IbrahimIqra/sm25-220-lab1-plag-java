public static void mostWater( Integer[] height ){
      Integer maxArea = 0;
      for (Integer i=0; i<height.length; i++){
        for(Integer j=i+1; j<height.length; j++){
          Integer hei = Math.min(height[i], height[j]);
          Integer wid = j-i;
          Integer area = hei*wid;
          maxArea = Math.max(maxArea, area);
        }
      }
      System.out.println(maxArea);
    }

public static Integer rowRotation( Integer examWeek, String[][] matrix ){
      Integer n = matrix.length;
      Integer x = (examWeek%n)-1;    //in case weeks > rows
      Integer i = 0;


      Integer j = matrix.length-1;
      while(i<j){
        String[] temp = matrix[i];
        matrix[i] = matrix[j];
        matrix[j] = temp;
        i++;
        j--;
      }
      i=0;
      j=x-1;
      while(i<j){
        String[] temp = matrix[i];
        matrix[i] = matrix[j];
        matrix[j] = temp;
        i++;
        j--;
      }
      i=x+1;
      j=matrix.length-1;
      while(i<j){
        String[] temp = matrix[i];
        matrix[i] = matrix[j];
        matrix[j] = temp;
        i++;
        j--;
      }
      for(Integer r=0; r<matrix.length; r++){
        for(Integer c=0; c<matrix[0].length; c++){
          if("AA".equals(matrix[r][c])){
            return r+1;
          }
        }
      }
        return null;
    }

   public static Integer[][] compressMatrix( Integer[][] matrix ){
      Integer row = matrix.length;
      Integer column = matrix[0].length;
      Integer r = row/2;
      Integer c = column/2;
      Integer[][]comp = new Integer[r][c];
      Integer r1 = 0;
      Integer c1 = 0;
      for(Integer i=0; i<row; i+=2){
        for(Integer j=0; j<column; j+=2){


          Integer sum = matrix[i][j]+matrix[i+1][j]+matrix[i][j+1]+matrix[i+1][j+1];
          comp[r1][c1] = sum;
          c1++;
        }
        r1++;
        c1=0;
      }
       return null;
    }

 public static void playGame( Integer[][] arena ){
      Integer row = arena.length;
      Integer col = arena[0].length;
      Integer sum = 0;
      for(Integer i=0; i<row; i++){
        for(Integer j=0; j<col; j++){


          if(arena[i][j]%50==0 && arena[i][j]!=0){
            if(i>0 && arena[i-1][j]==2){
              sum+=2;
            }
            if(i<row-1 && arena[i+1][j]==2){
              sum+=2;
            }
            if(j>0 && arena[i][j-1]==2){
              sum+=2;
            }
            if(j<col-1 && arena[i][j+1]==2){
              sum+=2;
            }
          }
        }
      }
      if(sum>=10){
        System.out.println("Points Gained: " +sum+ ". Your team has survived the game.");
      }
      else{
        System.out.println("Points Gained: " +sum+ ". Your team is out.");
      }
    }
