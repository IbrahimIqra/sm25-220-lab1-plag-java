//Assignment Task 01:
class AssgnTask1{


    public static void mostWater( Integer[] height ){
        
        int left= 0;
        int right=height.length-1;
        int maxArea=0;
        int height1=0;
        
        while(left<right){
            if (height[right]>height[left]){
                height1=height[right];
                left++;
            }
            else{
                height1=height[left];
                right--;
            }
            int width=right-left;
            int area=height1*width;
            
            if(maxArea<area){
                maxArea=area;
            }
        }
        
        System.out.println(maxArea);
    }

                }
            }

class AssgnTask3{


    public static Integer[][] compressMatrix( Integer[][] matrix ){

        int rows= matrix.length;
        int cols= matrix[0].length;
        int newRows= rows/2;
        int newCols= cols/2;
        
        Integer[][] compressed= new Integer[newRows][newCols];
        
        for (int i= 0;i<rows;i+=2){
            for(int j= 0;j<cols;j+= 2){
                int sum= matrix[i][j]+ matrix[i][j+1]+ matrix[i+1][j]+ matrix[i+1][j+1];


                compressed[i/2][j/2]= sum;
            }
        }
        return compressed;
    }

class AssgnTask4{
  public static void playGame(Integer[][] arena){

    int r= arena.length;
    int c= arena[0].length;
    int total=0;
    


    for(int i=0; i<r; i++){
      for(int j=0; j<c; j++){
        if(arena[i][j]%50==0 && arena[i][j]!=0){
          int points=0;
          
          if(i-1>=0 && arena[i-1][j]== 2){
            points++;}
          if(i+1<r && arena[i + 1][j] == 2){
            points++;}
          if(j-1>=0 && arena[i][j-1]==2){
            points++;}
          if(j+1<c && arena[i][j+1]==2){
            points++;}
          
          total+= points*2;
        }
      }
    }
    if(total>= 10){
      System.out.println("Points Gained: " + total+ ". Your team has survived the game.");}
    else{
      System.out.println("Points Gained: " + total+ ". Your team is out.");
    }
  }
