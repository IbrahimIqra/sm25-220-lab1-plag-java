public static void mostWater(Integer[] height) {
        int finalarea = 0;
        for (int i = 0; i < height.length; i++) {
            for (int j = i + 1; j < height.length; j++) {
              int heightarea ;
                if (height[i] < height[j]) {
                    heightarea = height[i];
                } else {
                    heightarea = height[j];
                }
                int lengtharea = j - i;
                int area = heightarea * lengtharea;
                if (area > finalarea) {
                    finalarea = area;
                }
            }
        }
        System.out.print(finalarea );
    }

 public static Integer rowRotation(Integer examWeek, String[][] matrix) {
        int rows = matrix.length;
        int cols = matrix[0].length;
        for (int i = 0; i < examWeek; i++) {
            String[] lastRow = matrix[rows - 1];
            for (int j = rows - 1; j > 0; j--) {
                matrix[j] = matrix[j - 1];
            }
            matrix[0] = lastRow;
        }
         for (int i = 0; i < rows; i++) {
            System.out.print("| ");
            for (int j = 0; j < cols; j++) {
                System.out.print(matrix[i][j] + " | ");
            }
            System.out.println();
        }
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if (matrix[i][j] == "AA") {
                    return i + 1;
                }
            }
        }
}

public static Integer[][] compressMatrix(Integer[][] matrix) {
        int row = matrix.length;
        int col = matrix[0].length;
        Integer[][] newarray = new Integer[row / 2][col / 2];
        for (int i = 0; i < row; i += 2) {
            for (int j = 0; j < col; j += 2) {
                int sum = matrix[i][j] + matrix[i][j+1] + matrix[i+1][j] + matrix[i+1][j+1];
                newarray[i / 2][j / 2] = sum;
            }
        }
        return newarray;
    }

public static void playGame(Integer[][] arena) {
        int rows = arena.length;
        int cols = arena[0].length;
        int totalPoints = 0;   
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if (arena[i][j] != 0 && arena[i][j] % 50 == 0) {
                    if (i - 1 >= 0 && arena[i - 1][j] == 2) {
                        totalPoints = totalPoints + 2;
                    }
                    if (i + 1 < rows && arena[i + 1][j] == 2) {
                        totalPoints = totalPoints + 2;
                    }    
                    if (j - 1 >= 0 && arena[i][j - 1] == 2) {
                        totalPoints = totalPoints + 2;
                    }   
                    if (j + 1 < cols && arena[i][j + 1] == 2) {
                        totalPoints = totalPoints + 2;
                    }
                }
            }
        }
        if (totalPoints >= 10) {
            System.out.println("Your team has survived the game");
        } else {
            System.out.println("Your team is out");
        }
    }
