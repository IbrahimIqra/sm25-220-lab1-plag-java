public static void mostWater( Integer[] height ){

        int maxArea = 0;
        for(int i=0; i<height.length; i++){
            for(int j=i+1; j<height.length; j++){
                int minHeight = height[i];
                if(height[j]<minHeight)
                    minHeight = height[j];
                int area = minHeight * (j-i);
                if(area > maxArea)
                    maxArea = area;
            }
        }
        System.out.println(maxArea);

    }

public static Integer rowRotation( Integer examWeek, String[][] matrix ){
        int rows = matrix.length;


        int shift = (examWeek - 1) % rows;

        for (int i = 0; i < shift; i++) {
            String[] temp = matrix[rows - 1];
            for (int j = rows - 1; j > 0; j--) {
                matrix[j] = matrix[j - 1];
            }
            matrix[0] = temp;
        }

        for (int i = 0; i < rows; i++) {
            System.out.print("| ");
            for (int j = 0; j < matrix[i].length; j++) {


                System.out.print(matrix[i][j] + " | ");
            }
            System.out.println();
        }

        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < matrix[i].length; j++) {
                if (matrix[i][j].equals("AA")) {
                    return i + 1;
                }
            }
        }

        return -1;
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){

        int rows = matrix.length;
        int cols = matrix[0].length;

        Integer[][] compressed = new Integer[rows / 2][cols / 2];

        for (int i = 0; i < rows; i += 2) {
            for (int j = 0; j < cols; j += 2) {
                int sum = matrix[i][j] + matrix[i][j + 1] + matrix[i + 1][j] + matrix[i + 1][j + 1];
                compressed[i / 2][j / 2] = sum;
            }
        }

        return compressed;
    }

public static void playGame( Integer[][] arena ){

       int rows = arena.length;
        int cols = arena[0].length;
        int totalPoints = 0;

        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {



                if (arena[i][j] % 50 == 0 && arena[i][j] != 0) {
                    int count = 0;

                    if (i - 1 >= 0 && arena[i - 1][j] == 2) count++;
                    if (i + 1 < rows && arena[i + 1][j] == 2) count++;
                    if (j - 1 >= 0 && arena[i][j - 1] == 2) count++;
                    if (j + 1 < cols && arena[i][j + 1] == 2) count++;

                    totalPoints += count * 2;
                }
            }
        }

        System.out.print("Points Gained: " + totalPoints + ". ");
        if (totalPoints >= 10) {
            System.out.println("Your team has survived the game.");
        } else {
            System.out.println("Your team is out.");
        }
    }
