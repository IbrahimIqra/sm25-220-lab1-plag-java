//Assignment Task 01: Container with Most Water
class AssgnTask1{
    public static void mostWater( Integer[] height ){
          int mxArea=0;
          int leftbar=0, rightbar=height.length-1;
          
          while(leftbar< rightbar){
            int hii;
            if(height[leftbar] <height[rightbar]){
              hii=height[leftbar];
            }
            else{
              hii=height[rightbar];
            }
            int s= rightbar-leftbar;
            int area=hii*s;
            if(area>mxArea){
              mxArea=area;
            }       
            if(height[leftbar] <height[rightbar]){
              leftbar++;
            }
            else{
              rightbar--;
            }
          }
          System.out.println(mxArea);
         
    }

//Assignment Task 02: Row Rotation Policy
class AssgnTask2{

    public static Integer rowRotation( Integer examWeek, String[][] matrix ){
         int rows=matrix.length, cols=matrix[0].length; 
         int count=examWeek-1;
         


         for(int i=0; i<count; i++){
           String [] n=new String[cols];
           for(int j=0; j<cols; j++){
             n[j]=matrix[rows-1][j];
           }
           for( int k=rows-1; k>0; k--){//
             for(int j=0; j<cols; j++){
             matrix[k][j]=matrix[k-1][j];
             }
           }
           for(int j=0; j<cols; j++){
             matrix[0][j]=n[j];
           }
         }
         //print
         for(int i=0; i<rows; i++){
           System.out.print(" | ");
           for(int j=0; j<cols; j++){
             System.out.print(matrix[i][j]+" | ");
           }
           System.out.println();
         }
           for(int i=0; i<rows; i++){          
           for(int j=0; j<cols; j++){
             if(matrix[i][j].equals("AA")){
               return i+1;
             }
           }
           }
           
        return -1;
    }

//Assignment Task 03: Matrix Compression
class AssgnTask3{


  public static Integer[][] compressMatrix( Integer[][] matrix ){
         int row=matrix.length;
         int col=matrix[0].length;
        Integer[][]cmprs=new Integer[row/2][col/2];


        
        for(int i=0; i<row; i+=2){
            for(int j=0; j<row; j+=2){


              int summm=matrix[i][j]+ matrix[i][j+1]+ matrix[i+1][j]+matrix[i+1][j+1];
              cmprs[i/2][j/2]=summm;
            }
        }
        return cmprs;
    }

//Assignment Task 04: Game Arena
class AssgnTask4{
 public static void playGame( Integer[][] arena ){
   int tp=0;
   
   for(int i=0; i<arena.length; i++){


       for(int j=0; j<arena[0].length; j++){
      
    if(arena[i][j]!=null&& arena[i][j]%50==0 && arena[i][j]>0){
        int count=0;
       
        if(i+1<arena.length && arena[i+1][j]!=null && arena[i+1][j]==2){
          count++;
        }
         if(i-1>=0 && arena[i-1][j]==2 && arena[i-1][j]!=null){
          count++;
        }
        if(j-1>=0 && arena[i][j-1]==2 && arena[i][j-1]!=null){
          count++;
        }
        if(j+1<arena[0].length && arena[i][j+1]==2 && arena[i][j+1]!=null){
          count++;
        }


        tp+=count*2;
      }
    }
  }
   if(tp<10){
      
     System.out.println("Points Gained: " +tp+". Your team is out.");

   }
   else{
     System.out.println("Points Gained: "+tp+ ". Your team has survived the game.");
     
    
   }
 }
