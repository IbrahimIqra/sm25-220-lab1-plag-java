   public static void mostWater( Integer[] height )
   {
      int area = 0;
      int max_area1 = 0;
      int max_area2 = 0;
      for(int i = 0; i < height.length; i++)
      {


          area = 0;
          for(int j = i+1; j < height.length; j++)
          {
              if(height[i] <= height[j])
              {
                  area = height[i] * (j-i);
              }
          }
          if(max_area1 == 0)
          {
              max_area1 = area;
          }
          else
          {
              if(area > max_area1)
              {
                  max_area1 = area;
              }
          }
      }
      for(int i = height.length - 1; i >= 0; i--)
      {


          area = 0;
          for(int j = i-1; j >= 0; j--)
          {
              if(height[i] <= height[j])
              {
                  area = height[i] * (i-j);
              }
          }
          if(max_area2 == 0)
          {
              max_area2 = area;
          }
          else
          {
              if(area > max_area2)
              {
                  max_area2 = area;
              }
          }
      }
      if(max_area1 > max_area2)
      {
          System.out.println(max_area1);
      }
      else
      {
          System.out.println(max_area2);
      }
   }

    public static void playGame( Integer[][] arena )
    {
        int countTotal = 0;
        int count = 0;


        int totalPoint = 0;
        for(int i = 0; i < arena.length; i++)
        {
            for(int j = 0; j < arena[i].length; j++)
            {
                if((arena[i][j] != 0) && (arena[i][j] % 50 == 0))
                {
                    count = 0;
                    //check top
                    if((i-1 >= 0) && (arena[i-1][j] == 2))
                    { count++; }
                    //check left
                    if((j-1 >= 0) && (arena[i][j-1] == 2))
                    { count++; }
                    //check right
                    if((j+1 < arena[i].length) && (arena[i][j+1] == 2))
                    { count++; }
                    //check bottom
                    if((i+1 < arena.length) && (arena[i+1][j] == 2))
                    { count++; }
                    countTotal += count;
                }
            }
        }
        totalPoint = countTotal*2;
        if(totalPoint < 10)
        {
            System.out.println("Points Gained: " + totalPoint + ". Your team is out.");
        }
        else
        {
            System.out.println("Points Gained: " + totalPoint + ". Your team has survived the game.");
        }            
    }
