    public static void mostWater( Integer[] height ){
    int a=0;
    int b=height.length-1;
    int maxarea=0;
    while(a<=b){
      int ch= Math.min(height[a],height[b]);
      int cw=b-a;
      int ca=ch*cw;
      if(ca>maxarea){
        maxarea=ca;
      }
      
      if(height[a]<height[b]){
        a++;
      }
      else{


        b--;
      }
    }
    System.out.println(maxarea);

    }

    public static Integer rowRotation( Integer exam_week, String[][] seat_status ){
        int r=seat_status.length;
        int c=seat_status[0].length;
        int cr=-1;


        for(int i=0;i<r;i++){
            for(int j=0;j<c;j++){
                if(seat_status[i][j].equals("AA")){
                    cr=i;
                    break;
                }
            }
            if(cr!=-1){
                break;
            }
        }


        int newr=(cr+exam_week)%r;
        for(int i=1;i<exam_week;i++){
            String[] last=seat_status[r-1];
            for(int j=r-1;j>0;j--){
                seat_status[j]=seat_status[j-1];
            }
            seat_status[0]=last;
        }
        Arr.print2D(seat_status);
        return newr;

    }

    public static Integer[][] compressMatrix( Integer[][] matrix ){
        int r=matrix.length;
        int c=matrix[0].length;
        int comprow=r/2;
        int compcol=c/2;
        Integer[][] compressedMatrix = new Integer[comprow][compcol];
        for(int i=0;i<r;i+=2){
          for(int j=0;j<c;j+=2){        
            int sum=0;
            sum+=matrix[i][j];   
            sum+=matrix[i][j+1];  
            sum+=matrix[i+1][j];   


            sum+=matrix[i+1][j+1]; 
            compressedMatrix[i/2][j/2]=sum;
            }
        }
        return compressedMatrix;
    }

    public static void playGame( Integer[][] arena ){
        int points=0;
        int row=arena.length;
        int col=arena[0].length;


        for(int i=0;i<row;i++){
          for(int j=0;j<col;j++){
            if(arena[i][j]%50==0 && arena[i][j]!=0){
              int countplayer=0;
              if(i-1>=0 && arena[i-1][j]==2){
                countplayer++;
              }
              if(i+1<row && arena[i+1][j]==2){
                countplayer++;
              }
              if(j-1>=0 && arena[i][j-1]==2){
                countplayer++;
              }
              if(j+1<col && arena[i][j+1]==2){
                countplayer++;
              }
              points+=countplayer* 2;
            }
          }
        }
        if (points>= 10){
          System.out.printf("Points Gained: %d. Your team has survived the game.\n", points);
        }
        else{
          System.out.printf("Points Gained: %d. Your team is out.\n", points);
        }


    }
