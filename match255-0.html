public static void mostWater( Integer[] height ){

        int end = height.length -1 ; 
        int start = 0 ; 
        int max = 0 ; 
        
        while(start < end){
            int h ; 
            if(height[start] < height[end]){
                h = height[start]; 
            }else{
                h = height[end]; 
            }
            int w = end - start ; 
            int area = h*w ; 
            if(area > max){
                max = area; 
            }

            if(height[start]< height[end]){
                start++; 
            }else{
                end--; 
            }
        }
        
        System.out.print(max);
    }

public static Integer[][] compressMatrix( Integer[][] matrix ) {
		int r = matrix.length;
		int c = matrix[0].length ;
		
		int l = r/2 , q = c/2 ; 
		Integer[][]com = new Integer[l][q];
		int sum = 0 ;
		
		for(int i = 0 ; i < r ; i+=2) {
			for(int j = 0 ; j< c ; j+=2) {


				sum = matrix[i][j] + matrix[i][j+1]+ matrix[i+1][j] + matrix[i+1][j+1];
				com[i/2][j/2] = sum;
			}
		}

		return com;

	}
