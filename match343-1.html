 public static void mostWater( Integer[] height ){


        //TO DO


        // My Code starts here
        int totalArea = 0;
        int starting = 0;
        int ending = height.length - 1;
       
        while (starting < ending) {
            int vertical = Math.min(height[starting], height[ending]);
            int horizontal = ending - starting;
            int area = vertical * horizontal;


            totalArea = Math.max(totalArea, area);
           
            if (height[starting] < height[ending]) {
                starting++;
            } else {
                ending--;
            }
        }
        System.out.print(totalArea);
        // My Code ends here


    }

    public static Integer[][] compressMatrix( Integer[][] matrix ){


        //For this task you'll need to create new 2D array
       
        //TO DO


        //remove the line below and return the newly created Compressed 2D matrix
       
        // ======My Code starts from here======
        int row = matrix.length;
        int col = matrix[0].length;
        Integer[][] newarr = new Integer[row / 2][col / 2];
        for (int i = 0; i < row; i += 2) {
            for (int j = 0; j < col; j += 2) {
                int sum = matrix[i][j] + matrix[i][j + 1] + matrix[i + 1][j] + matrix[i + 1][j + 1];
                newarr[i / 2][j / 2] = sum;
            }
        }
        return newarr;


        // ======My Code ends here======
    }

    public static void playGame( Integer[][] arena ){


        //For this task you don't need to create any new 2D array
        //just print the result inside the function
       
        //TO DO
       
        // ======My Code starts here======
        int row = arena.length;


        int col = arena[0].length;
        int tp = 0;
        for (int i = 0; i < row; i++) {
            for (int j = 0; j < col; j++) {
                if (arena[i][j] % 50 == 0 && arena[i][j] != 0) {


                    int points = 0;
                    if (i > 0 && arena[i - 1][j] == 2) {
                        points++;
                    }
                    if (i < row - 1 && arena[i + 1][j] == 2) {
                        points++;
                    }
                    if (j > 0 && arena[i][j - 1] == 2) {
                        points++;
                    }
                    if (j < col - 1 && arena[i][j + 1] == 2) {
                        points++;
                    }
                    tp += points * 2;


                }
            }
        }
        if (tp >= 10) {
            System.out.println("Points Gained: " + tp + ". Your team has survived the game.");
        } else {
            System.out.println("Points Gained: " + tp + ". Your team is out.");
        }


        // ======My Code ends here======
    }
