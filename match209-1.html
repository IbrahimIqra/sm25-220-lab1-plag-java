    public static void mostWater(Integer[] height){
    int maxWater = 0;
  for (int i = 0; i < height.length; i++) {
        for (int j = i + 1; j < height.length; j++){
            int width = j - i;


            int minHeight;
            if (height[i] < height[j]){
                minHeight = height[i];
            } 
            else{
                minHeight = height[j];
            }
            int currentArea = width * minHeight;
         
            if (currentArea > maxWater){
                maxWater = currentArea;
            }
        }
    }
    System.out.println(maxWater); 
}

class AssgnTask4{
public static void playGame(Integer[][] arena){
        int totalPoint = 0;
          for (int i = 0; i < arena.length; i++){
              for (int j = 0; j < arena[i].length; j++) {


                if (arena[i][j] % 50 == 0 && arena[i][j]!= 0){
                    if (i > 0 && arena[i-1][j] == 2) totalPoint += 2;
 if (i < arena.length-1 && arena[i+1][j] == 2) totalPoint += 2;
            
                    if (j > 0 && arena[i][j-1] == 2) totalPoint += 2;
               
                    if (j < arena[i].length-1 && arena[i][j+1] == 2) totalPoint += 2;
                }
            }
        }
        
        System.out.print( "Points Gained " + totalPoint );


        if (totalPoint >= 10){
            System.out.println(" Your team has survived the game");
        } 
        else{
            System.out.println(" Your team is out");
        }
    }
