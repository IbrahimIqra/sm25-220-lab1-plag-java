class AssgnTask1{


    public static void mostWater( Integer[] height ){
        int ma=0;// Max Area
        for(int i=0; i<height.length; i++){
            for(int j=i; j<height.length; j++){
                int b=0;// Breadth of Rectangle
                int l=0;// Length of Rectangle
                int ma2=0;// Dummy Max Area For comparision
                if(height[i]<height[j]){
                    l=j-i;
                    b=height[i];
                    ma2= l*b;
                    if(ma2>ma){
                        ma=ma2;
                    }
                }
                if(height[i]==height[j]){
                    l=j-i;
                    b=height[i];
                    ma2= l*b;
                    if(ma2>ma){
                        ma=ma2;
                    }
                }
                if(height[i]>height[j]){
                    l=j-i;
                    b=height[j];
                    ma2= l*b;
                    if(ma2>ma){
                        ma=ma2;
                    }
                }
            }
        }
        System.out.print(ma);
    }

            String [] temp = new String[matrix[0].length];
            String [] next = new String[matrix[0].length];
            for(int j=0; j<row; j++){
                
                    temp = matrix[(j+1)%row];
                    if(j==0){ 
                     matrix[(j+1)%row]=matrix[j];
                    }
                    else{
                        matrix[(j+1)%row]=next;
                    }
                    next= temp;
            }
            if(i==(examWeek-1)){
            for(int k=0; k<row; k++){
                System.out.print("[");
                for(int j=0; j<col; j++){
                    System.out.print(" "+matrix[k][j]+" ");
                }
                System.out.print("]");
                System.out.println();
            }
        }
        }
        for(int i=0; i<row; i++){
            for(int j=0; j<col; j++){
                if(matrix[i][j].equals("AA")){
                    rownum=i+1;
                }
            }
        }
        return rownum;
