    public static Integer[][] compressMatrix( Integer[][] matrix ){


        int row = matrix.length;
        int col = matrix[0].length;
        int row1 = row/2;
        int col1 = col/2;


        Integer[][] array = new Integer[row1][col1];
        for(int i=0;i<row;i+=2){
            for(int j=0;j<col;j+=2){
                int sum = matrix[i][j] + matrix[i][j+1] + matrix[i+1][j] + matrix[i+1][j+1];


                array[i/2][j/2] = sum;
            }
        }
        return array;
    }

    public static void playGame( Integer[][] arena ){


        int row = arena.length;
        int col = arena[0].length;
        int total = 0;


        for(int i=0;i<row;i++){
            for(int j=0;j<col;j++){
                if(arena[i][j] % 50 == 0 && arena[i][j] != 0){
                    int point = 0;


                    if(i>0 && arena[i-1][j] == 2){
                        point += 1;
                    }
                    if(i<row - 1 && arena[i+1][j] == 2){
                        point += 1;
                    }
                    if(j>0 && arena[i][j-1] == 2){
                        point += 1;
                    }
                    if(j<col -1 && arena[i][j+1] == 2){
                        point += 1;
                    }
                    total += point * 2;
                }
            }
        }


        if(total >=10){
            System.out.println("Points Gained: "+ total +". Your team has survived the game.");
        }
        else{
            System.out.println("Points Gained: "+ total +". Your team is out.");
        }
    }
