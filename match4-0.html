                if(height[i]<=height[j]){
                    if(height[i]*(j-i)>maxArea){

                    }
                }
                else{
                    if(height[j]*(j-i)>maxArea){
                        maxArea=height[j]*(j-i);
                    }
                }
            }
        }
        System.err.printf("%d\n", maxArea);

    }

    public static void playGame( Integer[][] arena ){

        for(int i=0;i<arena.length;i++){

                    if(i!=0&&arena[i-1][j]==2){
                        sum+=arena[i-1][j];
                    }
                    if(j!=0&&arena[i][j-1]==2){
                        sum+=arena[i][j-1];
                    }
                    if(i<arena.length-1&&arena[i+1][j]==2){
                        sum+=arena[i+1][j];
                    }
                    if(j<arena[0].length-1&&arena[i][j+1]==2){
                        sum+=arena[i][j+1];
                    }
                }
            }
        }
        if(sum>=10){
            System.out.printf("Points Gained: %d. Your team has survived the game.\n",sum);
        }
        else{
            System.out.printf("Points Gained: %d. Your team is out.\n",sum);
