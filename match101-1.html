public static void mostWater(Integer[] height) {
    //TO DO
    int largestArea = 0;
    int start = 0;
    int end = height.length - 1;

    while (start < end) {
        int currentWidth = end - start;
        int currentHeight = Math.min(height[start], height[end]);
        int area = currentWidth * currentHeight;
        if (area > largestArea) {
            largestArea = area;
        }
        
        if (height[start] < height[end]) {
            start++;
        } else {
            end--;
        }
    }
    System.out.print(largestArea);
}

public static Integer[][] compressMatrix(Integer[][] matrix) {

    //For this task you'll need to create new 2D array
        
    //TO DO
    int rows = matrix.length;
    int cols = matrix[0].length;
    int compressedRows = rows / 2;
    int compressedCols = cols / 2;

    Integer[][] result = new Integer[compressedRows][compressedCols];

    for (int i = 0; i < rows; i += 2) {
        for (int j = 0; j < cols; j += 2) {
            int sumBlock = 0;
            sumBlock += matrix[i][j];
            sumBlock += matrix[i][j + 1];
            sumBlock += matrix[i + 1][j];


            sumBlock += matrix[i + 1][j + 1];
            result[i / 2][j / 2] = sumBlock;
        }
    }
    return result;
}

public static void playGame(Integer[][] arena) {
    
    //For this task you don't need to create any new 2D array
    //just print the result inside the function
        
    //TO DO

    int totalPoints = 0;
    int numRows = arena.length;
    int numCols = arena[0].length;

    
    int[] dRow = {-1, 1, 0, 0};
    int[] dCol = {0, 0, -1, 1};

    for (int r = 0; r < numRows; r++) {
        for (int c = 0; c < numCols; c++) {
            Integer cell = arena[r][c];
            
            if (cell != null && cell > 0 && cell % 50 == 0) {
                
                for (int dir = 0; dir < 4; dir++) {
                    int adjRow = r + dRow[dir];
                    int adjCol = c + dCol[dir];
                    // Boundary check
                    if (adjRow >= 0 && adjRow < numRows && adjCol >= 0 && adjCol < numCols) {
                        if (arena[adjRow][adjCol] != null && arena[adjRow][adjCol] == 2) {
                            totalPoints += 2;
                        }
                    }
                }
            }
        }
    }

    if (totalPoints >= 10) {
        System.out.printf("Points Gained: %d. Your team has survived the game.\n", totalPoints);
    } else {
        System.out.printf("Points Gained: %d. Your team is out.\n", totalPoints);
    }
}
