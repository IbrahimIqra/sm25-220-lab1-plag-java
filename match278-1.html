public static void mostWater( Integer[] height ){


    int start=0;
    int end= height.length-1;
    int maximumarea=0;


    while(start<end){
        
        maximumarea= Math.max(maximumarea, (end- start)* Math.min(height[start],height[end] ));
        


        if (height[start]< height[end]) {


        start++;
        }
        else
        {
        end--;
    }


    }
    System.out.println(maximumarea);
}

 public static Integer rowRotation( Integer examWeek, String[][] matrix ){
        int rope=0;
        int start=0;
        int end=matrix.length-1;


        while(start<end){
            String[] temp= matrix[start];
            matrix[start]=matrix[end];
            matrix[end]=temp;
            start++;
            end--;
        }
        start=0;


        int d=(examWeek % matrix.length);


        end=d-1;


        while(start<end) {
            String[] temp= matrix[start];
            matrix[start]=matrix[end];
            matrix[end]=temp;
            start++;
            end--;
        }
        start= d+1;
        end= matrix.length-1;




        while(start<end) {
            String[] temp= matrix[start];
            matrix[start]=matrix[end];
            matrix[end]=temp;
            start++;
            end--;
        }


        for (int i = 0; i < matrix.length; i++) {
            for (int j = 0; j < matrix[0].length; j++) {
                if( matrix[i][j]=="AA")
                {
                   rope=i;
                }
            }


        }
return rope;


    }

 public static Integer[][] compressMatrix( Integer[][] matrix ){


        Integer [] [] newarr= new Integer[2][2];
        int tempsum=0;
        int cc =0,rr =0;
        for (int i = 0; i < matrix.length-1; i+=2) {
            for (int j = 0; j < matrix[0].length-1; j+=2) {


                    tempsum=matrix[i][j]+ matrix[i+1][j]+matrix[i][j+1]+matrix[i+1][j+1];
                    newarr[rr][cc]= tempsum;
                    cc++;
            }
            rr++;
            cc=0;
        }
        return newarr;




    }

 public static void playGame( Integer[][] arena ){
        int points=0;


        int row= arena.length;
        int col=arena[0].length;
        int last=0;
        for (int i = 0; i < row; i++) {
            for (int j = 0; j < col; j++) {
                if(arena[i][j]%50==0 && arena[i][j]!=0){
                    if( i>last && arena[i-1][j]==2){
                        points+=2;
                    }
                    if (i<row-1 && arena[i+1][j]==2  ){
                        points+=2;
                    }
                    if(j>last && arena[i][j-1]==2 ){
                        points+=2;
                    }


                    if(j<col-1 && arena[i][j+1]==2  ){
                        points+=2;
                    }
                }
            }


        }
       if(points>=10){
           System.out.println("Points Gained: "+ points+ " Your team has survived the game.");
       }
       else {
           System.out.println("Points Gained: "+ points+ " Your team is out.");


       }
    }
