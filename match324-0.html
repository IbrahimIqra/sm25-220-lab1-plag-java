public static void mostWater( Integer[] height ){
      int start = 0;
      int end = height.length -1;
      int bilay = 0;
      
      while (start< end){
        int temp;
        if ( height[start]< height[end]){
          temp= height[start];
        }else {
          temp= height[end];
        }
        int width = end-start;
        int area = temp * width;
        if (area> bilay){
          bilay = area;
        }
        if ( height[start] < height[end]){
          start+=1;
        }else{
          end--;
        }
      }
      System.out.println(bilay);
    }

    public static Integer[][] compressMatrix( Integer[][] matrix ){
      int rows = matrix.length;
      int col = matrix[0].length;
      int nRows = rows/2;
      int nCol = col/2;
      
      Integer[][]bilay = new Integer[nRows][nCol];
      for(int i=0; i<rows; i+=2){
        for(int j=0; j<col; j+=2){
          int sum= matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+matrix[i+1][j+1];
          bilay[i/2][j/2]=sum;
        }
      }




        return bilay;
    }

    public static void playGame( Integer[][] arena ){
      int rows = arena.length;
      int col = arena[0].length;
      int bilay = 0;
      for(int i=0; i<rows; i++){
        for(int j = 0; j<col; j++){
          if(arena[i][j]%50==0 && arena[i][j] != 0){


            int point = 0;
            if ((i>0 && arena[i-1][j] == 2)){ //up
              point++;
            }
            if ((i<rows-1 && arena[i+1][j] == 2)){ //down
              point++;
            }
            if ((j>0 && arena[i][j-1] == 2)){ //left
              point++;
            }
            if ((j<col-1 && arena[i][j+1] == 2)){ 
              point++;
            }
            
            bilay += point * 2;
          }
        }
      }
      if(bilay<10){
        System.out.println("Points Gained: "+ bilay+". Your team is out.");
      }else{
        System.out.println("Points Gained: "+ bilay+". Your team has survived the game.");
      }
    }
