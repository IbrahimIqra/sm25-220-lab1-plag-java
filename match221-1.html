class AssgnTask1{
    public static void mostWater( int[] height ){
        int maxWater = 0;
        int left = 0;
        int right = height.length - 1;

        while (left < right) {
            int h = Math.min(height[left], height[right]);
            int w = right - left;

            maxWater = Math.max(maxWater, h * w);

            if (height[left] < height[right])
                left++;
            else
                right--;
        }

        System.out.println(maxWater);
    }

public static Integer[][] compressMatrix(Integer[][] matrix) {
        int rows = matrix.length;
        int cols = matrix[0].length;
        Integer[][] temp = new Integer[rows / 2][cols / 2];

        for (int i = 0; i < rows; i += 2) {
            for (int j = 0; j < cols; j += 2) {
                int sum = matrix[i][j] + matrix[i + 1][j] + matrix[i][j + 1] + matrix[i + 1][j + 1];


                temp[i / 2][j / 2] = sum;
            }
        }

        return temp;
    }

 public static void playGame(Integer[][] arena) {
        int rows = arena.length;
        int cols = arena[0].length;
        int EarnedPoints  = 0;

        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                int CurrPoss = arena[i][j];

                    // The multiples of 50
                if (CurrPoss != 0 && CurrPoss % 50 == 0) {
                    


                    if (i - 1 >= 0 && arena[i - 1][j] == 2) {
                        EarnedPoints += 2; // To check Up
                    }
                  
                    if (i + 1 < rows && arena[i + 1][j] == 2) {
                        EarnedPoints += 2;   // To check Down
                    }
                    
                    if (j - 1 >= 0 && arena[i][j - 1] == 2) {
                        EarnedPoints += 2; // To check LEFT
                    }
                    
                    if (j + 1 < cols && arena[i][j + 1] == 2) {
                        EarnedPoints += 2; // To check RIGHT
                    }
                }
            }
        }

        if (EarnedPoints >= 10) {
            System.out.println("Points Gained: " + EarnedPoints + ". Your team has survived the game.");
        } else {
            System.out.println("Points Gained: " + EarnedPoints + ". Your team is out.");
        }
    }
