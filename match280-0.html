public static void mostWater( Integer[] height ){

        //TO DO
        int leftPointer = 0;
        int rightPointer = height.length - 1;
        int maxArea = 0;
        int distance = 0;
        int minHeight = 0;
        int currentArea = 0;
        

        while (leftPointer <= rightPointer) {
            distance = rightPointer - leftPointer;
            minHeight = Math.min(height[leftPointer], height[rightPointer]);


            currentArea = minHeight * distance;
            maxArea = Math.max(maxArea, currentArea);

            if (height[leftPointer] < height[rightPointer]){
                leftPointer++;
            } else {
                rightPointer--;
            }

        }
        System.out.println(maxArea);

    }

public static Integer rowRotation( Integer examWeek, String[][] matrix ){

        //TO DO
        int row = matrix.length;
        int col = matrix[0].length;

        for (int i = 0; i < (examWeek - 1) % row; i++) {


            String[] temp = matrix[row - 1];
            for (int r = row - 1; r > 0; r--) {
                matrix[r] = matrix[r - 1];
            }
            matrix[0] = temp;
        }
        

        int initialRow = -1;


        for (int r = 0; r < row; r++) {
            for (int c = 0; c < col; c++) {
                if (matrix[r][c].equals("AA")) {
                    initialRow = r + 1; 
                    break;
                }
            }
        }

        return initialRow;
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){

        
        //TO DO
        int row = matrix.length;
        int col = matrix[0].length;

        Integer[][] newMatrix = new Integer[row/2][col/2];

        for (int i = 0; i < row; i+=2) {
            for (int j = 0; j < col; j+=2) {
                int sum = 0;
                sum += matrix[i][j];
                sum += matrix[i+1][j];
                sum += matrix[i][j+1];


                sum += matrix[i+1][j+1];
                newMatrix[i/2][j/2] = sum;
            }
            
        }

        return newMatrix;
    }

public static void playGame( Integer[][] arena ){
        
        //TO DO
        int row = arena.length;
        int col = arena[0].length;
        int totalPoints = 0;

        int[] x = {-1, 1, 0, 0};
        int[] y = {0, 0, -1, 1};



        for (int i = 0; i < row; i++) {
            for (int j = 0; j < col; j++) {
                if (arena[i][j] % 50 == 0 && arena[i][j] != 0) {
                    for (int k = 0; k < 4; k++) {
                        int left_right = i + x[k];
                        int up_down = j + y[k];

                        if (left_right >= 0 && left_right < row && up_down >= 0 && up_down < col) {
                            if (arena[left_right][up_down] != null && arena[left_right][up_down] == 2) {


                                totalPoints += 2;
                            }
                        }
                    }
                }
            }
        }

    System.out.print("Points Gained: " + totalPoints + ". ");
    if (totalPoints >= 10) {
        System.out.println("Your team has survived the game.");
    } else {
        System.out.println("Your team is out.");
