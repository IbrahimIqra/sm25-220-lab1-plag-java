class AssgnTask1{
  public static void mostWater( Integer[] height ){
    int left = 0, right = height.length - 1, maxarea = 0;
    while (left < right) {
      int width = right - left;
      int minheight = 0;


      if (height[left] < height[right]){
       minheight = height[left];
      } 
      else {
        minheight = height[right];
      }
      int area = width * minheight;
      if (area > maxarea) {
        maxarea = area;
      }
      if (height[left] < height[right]) {
        left += 1;
      }
      else{
        right -= 1;
      }
    }
    System.out.println(maxarea); 
  }

class AssgnTask2{


  public static Integer rowRotation( Integer examWeek, String[][] matrix ){
    int ro =matrix.length;
    int col =matrix[0].length;
    int shifs = (examWeek - 1)%ro;
    for (int k =0;k < shifs; k++) {
      String[] temp= matrix[ro- 1];
      for (int i = ro - 1; i>0; i--) {
        matrix[i]=matrix[i-1];
      }
      matrix[0]=temp;
    }
    for (int i=0; i<ro;i++) {
      System.out.print("| ");
      for (int j=0;j < col; j++) {
        System.out.print(matrix[i][j] + " | ");
      }
      System.out.println();
    }
    for (int i = 0; i < ro; i++) {
      for (int j = 0; j < col; j++) {
        if ("AA".equals(matrix[i][j])) {
          return i;
        }
      }
    }
    return -1; 
  }


class AssgnTask3{


    public static Integer[][] compressMatrix( Integer[][] matrix ){
    int ro = matrix.length;
    int col = matrix[0].length;
    Integer[][] result =new Integer[ro / 2][col /2];
    for (int m = 0; m <ro / 2; m++) 
    {
        for (int r =0; r<col / 2; r++) 
        {
            int sum= matrix[m * 2][r * 2] +
                      matrix[m* 2][r * 2+ 1] +
                      matrix[m * 2 + 1][r *2] +
                      matrix[m * 2+ 1][r *2 + 1];
            result[m][r] =sum;
        }
    }
    return result;
}


class AssgnTask4{
  public static void playGame( Integer[][] arena ){
    int total= 0;
    int ro = arena.length;
    int col = arena[0].length;
    for (int k =0; k <ro; k++) 
    {
      for (int j = 0; j< col; j++) 
      {
        int valu = arena[k][j];


        if (valu >=50 && valu% 50== 0)
        {
          int count = 0;
          if (k - 1 >= 0 && arena[k- 1][j]== 2)count++; //up
          if (k + 1<ro && arena[k+1][j] == 2) count++; //down
          if (j - 1>= 0 && arena[k][j-1]== 2) count++; //lft       
          if (j+ 1< col && arena[k][j +1] ==2) count++; //rght
          
          total+=count * 2;
        }
      }
    }


    if 
      (total >= 10) {
        System.out.println("Points Gained: " + total + ". Your team has survived the game.");
    } 
    else {
        System.out.println("Points Gained: " + total + ". Your team is out.");
    }
}
