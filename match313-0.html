public static void mostWater(Integer[] height) {
    int area=0;
    int left=0;
    int right=height.length-1;
   
    while (left<right){
        int h;
        if(height[left]<height[right]){
            h=height[left];
        }else{


            h=height[right];
        }


        int w=right-left;
        int a=h*w;


        if(a>area){
            area=a;
        }


        if(height[left]<height[right]){
            left++;
        }else{
            right--;
        }
    }


    System.out.println(area);
}

public static Integer[][] compressMatrix(Integer[][] matrix) {
    int prevR=matrix.length;
    int prevC=matrix[0].length;
    int newR=prevR/2;
    int newC=prevC/2;
    Integer[][] newMatrix=new Integer[newR][newC];


    for(int i=0; i<prevR; i+=2){
        for(int j=0; j<prevC; j+=2){
            int sum=matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+matrix[i+1][j+1];


            newMatrix[i/2][j/2]=sum;
        }
    }
    return newMatrix;
}




    //DO NOT CHANGE ANY DRIVER CODE BELOW THIS LINE
    public static void main(String[] args){
        Integer[][] matrix = {
            { 1 , 2 , 3 , 4 },
            { 5 , 6 , 7 , 8 },
            { 1 , 3 , 5 , 2 },
            {-2 , 0 , 6 ,-3 }
        };
        System.out.println("Given Matrix: ");
        Arr.print2D(matrix);

        System.out.print("| 14 | 22 |\n| 2  | 10 |\n");
       
        System.out.print("\nYour Output:\n");
        Integer[][] returnedArray = compressMatrix( matrix );
        Arr.print2D( returnedArray );
    }
}

class AssgnTask4{

  public static void playGame(Integer[][] arena) {
    int row=arena.length;
    int col=arena.length;
    int total=0;


    for(int i=0; i<row; i++){
        for(int j=0; j<col; j++){


            if(arena[i][j]%50==0 && arena[i][j]!=0){
                if(i>0 && arena[i-1][j]==2)
                total+=2;
                if(i<row-1 && arena[i+1][j]==2)


                total+=2;
                if(j>0 && arena[i][j-1]==2)
                total+=2;
                if(j<col-1 && arena[i][j+1]==2)
                total+=2;
            }
        }
    }
    System.out.print("Points Gained: "+total+". ");
    if(total>=10){
        System.out.println("Your team has survived the game.");
    }else{
        System.out.println("Your team is out.");
    }
}
