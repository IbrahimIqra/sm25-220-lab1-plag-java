public static void mostWater( Integer[] height ){

        int start = 0;
        int last = height.length - 1;
        int maxw = 0;

        
        while (start < last) {
            int width = last - start;
            int h = Math.min(height[start], height[last]);
            int area = width * h;

            
            maxw = Math.max(maxw, area);

            
            if (height[start] < height[last]) {
                start++;
            } else {
                last--;
            }
        }

        
        System.out.println("Maximum water that can be stored: " + maxw);
    }

 public static Integer rowRotation( Integer examWeek, String[][] matrix ){
    int rows = matrix.length;
    int cols = matrix[0].length;

    
    for (int week = 0; week < examWeek; week++) {


        
        String[] last = matrix[rows - 1];

        
        for (int r = rows - 1; r > 0; r--) {
            matrix[r] = matrix[r - 1];  
        }

        
        matrix[0] = last;
    }

    
    System.out.println("Matrix after " + examWeek + " week(s) rotation:");
    for (int r = 0; r < rows; r++) {


        for (int c = 0; c < cols; c++) {
            System.out.print(matrix[r][c] + " ");
        }
        System.out.println();
    }

    


    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            if (matrix[r][c].equals("AA")) {
                
                return r + 1;  
            }
        }
    }

    
    return null;
}

public static Integer[][] compressMatrix(Integer[][] matrix) {
    int rows = matrix.length;
    int cols = matrix[0].length;

    
    if (rows % 2 != 0 || cols % 2 != 0) {
        return null;
    }

    


    int crows = rows / 2;
    int ccols = cols / 2;
    Integer[][] cmatrix = new Integer[crows][ccols];

    
    for (int r = 0; r < rows; r += 2) {
        for (int c = 0; c < cols; c += 2) {
            


            cmatrix[r / 2][c / 2] = matrix[r][c] + matrix[r][c + 1] + matrix[r + 1][c] + matrix[r + 1][c + 1];
        }
    }

    
    return cmatrix;
}

  public static void playGame(Integer[][] arena) {
    int rows = arena.length;
    int cols = arena[0].length;
    int fpoints = 0;  

    
    for (int R = 0; R < rows; R++) {
        for (int C = 0; C < cols; C++) {
            
            if (arena[R][C] % 50 == 0 && arena[R][C] > 0) {
                int points = 0; 

                
                if (R > 0 && arena[R - 1][C] == 2) points++; 
                if (R < rows - 1 && arena[R + 1][C] == 2) points++; 
                if (C > 0 && arena[R][C - 1] == 2) points++; 
                if (C < cols - 1 && arena[R][C + 1] == 2) points++; 

                
                points *= 2;

                
                fpoints += points;
            }
        }
    }

    
    if (fpoints >= 10) {
        System.out.println("Your team has survived the game & points " + fpoints);
    } else {
        System.out.println("Your team is out & points "+ fpoints);
    }
}
