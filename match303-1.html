public static void mostWater( Integer[] height ){

        int maxA = 0;
        int leftp = 0;
        int rightp = height.length-1;
        while(leftp<rightp)
        {
            int width = rightp-leftp;
            int smolh;
            if(height[leftp]> height[rightp])
            {
                smolh = height[rightp];
            }
            else
            {
                smolh = height[leftp];
            }
            int Area = width * smolh;
            if(Area > maxA)
            {
                maxA = Area;
            }
            if(height[leftp]> height[rightp])
            {
                rightp--;
            }
            else
            {
                leftp++;
            }
        }
        System.out.println("Max water: "+ maxA);
    }

public static Integer rowRotation(Integer examWeek, String[][] matrix) {
        int rotNUM = examWeek - 1;


        int rows = matrix.length;
        int cols = matrix[0].length;

        for (int a = 0; a < rotNUM; a++)
        {
            String[] lastRow = matrix[rows - 1];
            for (int i = rows - 1; i > 0; i--)
            {
                matrix[i] = matrix[i - 1];
            }

            matrix[0] = lastRow;
        }
        Arr.print2D(matrix);
        for (int i = 0; i < rows; i++)
        {
            for (int j = 0; j < cols; j++)
            {
                if (matrix[i][j].equals("AA"))
                {
                    return i+1;
                }
            }
        }
        return -1;
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){
        int rows = matrix.length;
        int cols = matrix[0].length;
        Integer [][] ans = new Integer[rows/2][cols/2];
        int sum;
        for(int i = 0; i<rows; i = i+2)
        {
            for(int j =0 ; j<cols; j = j+2)
            {


                sum = matrix[i][j] + matrix[i+1][j] + matrix[i][j+1]
                        + matrix[i+1][j+1];
                ans[i/2][j/2] = sum;
            }
        }
        return ans;
    }

public static void playGame( Integer[][] arena ){
        int rows = arena.length;
        int cols = arena[0].length;
        int totalp = 0;
        for(int i = 0; i<rows; i++)
        {
            for(int j= 0; j<cols ; j++)
            {
                if(arena[i][j] % 50 == 0 && arena[i][j] != 0)
                {
                    int playerPoints = 0;


                    if(i > 0 && arena[i-1][j] == 2)
                    {
                        playerPoints += 2;
                    }
                    if(i < rows - 1 && arena[i+1][j] == 2)
                    {
                        playerPoints += 2;
                    }
                    if(j > 0 && arena[i][j-1] == 2)
                    {
                        playerPoints += 2;
                    }
                    if(j < cols - 1 && arena[i][j+1] == 2)
                    {
                        playerPoints += 2;
                    }
                    totalp += playerPoints;
                }
            }
        }


        if(totalp>=10)
        {
            System.out.println("Points Gained: "+totalp+". Your team has survived the game.");
        }
        else
        {
            System.out.println("Points Gained: "+totalp+". Your team is out.");
        }
    }
