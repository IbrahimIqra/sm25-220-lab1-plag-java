class AssgnTask1
{
	public static void mostWater( Integer[] height )
	{
	  int container_area_max =0;
	  int container_area =0;
	  int container_width =0;
	  int pointer_1=0;
	  int pointer_2=height.length -1;
	  
	  
	  while (pointer_1 < pointer_2)
	  {
	    int container_height =0;
	    
	    if(height[pointer_1] < height[pointer_2])
	    {
	      container_height = height[pointer_1];
	      container_width = pointer_2 - pointer_1;
	      pointer_1++;
	    }
	    else
	    {
	      container_height = height[pointer_2];
	      container_width = pointer_2 - pointer_1;
	      pointer_2--;
	    }
	    
	    container_area = container_height*container_width;
	    
	    if (container_area > container_area_max)
	    {
	      container_area_max = container_area;
	    }
	  }
	  
	  System.out.println(container_area_max);

	}

}

class AssgnTask3
{
	public static Integer[][] compressMatrix( Integer[][] matrix )
	{
		int sum;
		int row = matrix.length;
		int col = matrix[0].length;

		Integer n_arr [] [] = new Integer [row/2] [col/2];

		for(int r=0; r<row; r+=2)
		{
			for(int c=0; c< col; c+=2)
			{
				sum = matrix[r][c] + matrix[r][c + 1] + matrix[r + 1][c] + matrix[r + 1][c + 1];
				n_arr[r/2][c/2]=sum;
			}
		}

    return n_arr;
  }
}
