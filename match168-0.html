   public static void mostWater( Integer[] height ){
       int maxArea = 0;
       int currentArea = 0;
       int left = 0;
       int right = height.length - 1;
       int currentHeight = 0;
       int currentWidth = 0;


       while (left < right){
           if (height[left] < height[right]){
               currentHeight = height[left];
           }  
           else{
               currentHeight = height[right];
           }
           currentWidth = right - left;
           currentArea = currentHeight * currentWidth;
           if (currentArea > maxArea){


               maxArea = currentArea;
           }
           if (height[left] < height[right]){
               left++;
           }
           else{
               right--;
           }
       }
       System.out.println(maxArea);


   }

   public static Integer rowRotation( Integer examWeek, String[][] matrix ){
       int r = matrix.length;
       int c = matrix[0].length;


       for (int i = 0; i<examWeek-1; i++){
           String [] lst = matrix[r-1];
          
           for (int j = r-1; j>0; j--){
               matrix[j]=matrix[j-1];


           }
           matrix[0] = lst;


       }


       for (int i = 0; i<r; i++){
           for (int j =0;j<c;j++){
               System.out.print("| " + matrix[i][j] + " ");


           }
           System.out.println("|");
       }


      
       for (int i = 0; i<r; i++){
           for (int j =0;j<c;j++){
               if (matrix[i][j]=="AA"){
                   return i+1;
               }
           }
       }
       return -1;
   }

   public static Integer[][] compressMatrix( Integer[][] matrix ){
       int r = matrix.length;
       int c = matrix[0].length;


       Integer [][] res = new Integer[r/2][c/2];


       for (int i=0;i<r;i+=2) {
           for (int j=0;j<c;j+=2) {
               int sum =matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+ matrix[i+1][j+1];
               res [i/2][j/2]=sum;
           }
       }
       return res;
  
  
   }

   public static void playGame( Integer[][] arena ){
       int r = arena.length;
       int c =arena[0].length;


       int points = 0;


       for (int i =0; i<r; i++){
           for (int j=0;j<c;j++){
               if (arena[i][j]%50==0 && arena[i][j]!=0){
                   if (i>0 && arena[i-1][j]==2){
                       points+= 2;
                   }
                   if (i<r-1 && arena[i+1][j]==2){
                       points+=2;
                   }
                   if(j>0 && arena[i][j-1]==2){
                       points+=2;
                   }


                   if (j<c-1 && arena[i][j+1] ==2){
                       points +=2;
                   }
               }
           }
       }


       System.out.print("Points Gained: " + points + ". ");
       if(points>=10){
           System.out.print("Your team has survived the game.");
       }
       else{
           System.out.print("Your team is out.");
       }
   }
