class AssgnTask1{

    


    public static void mostWater( Integer[] height ){

        int first = 0;
        int last = height.length-1;
        int max = 0;
        
        while(first<last){
          int h = 0;
          int w = last - first;
          int area = 0;
          if(height[first]>height[last]){
            h = height[last];
          }
          else {
            h = height[first];
          }
          area = h*w;
          if(area>max){
            max=area;
          }
          if(height[first]>height[last]){
            last--;
          }
          else{
            first++;
          }
        }
        System.out.print(max);
        

    }
}

      Integer [][] newmatrix = new Integer[newrow][newrow];
      
      
      for(int i=0; i<row; i+=2){
        int sum=0;
        for(int j=0; j<row; j+=2){


          sum+=matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+matrix[i+1][j+1];
          newmatrix[i/2][j/2]=sum;
        }
        
      }
      

        //remove the line below and return the newly created Compressed 2D matrix
