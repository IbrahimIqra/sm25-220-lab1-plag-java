public static void mostWater( Integer[] height ){
        int begin = 0;
        int end = height.length-1;
        int area1 = 0;
        int h = 0;
        while(begin<end) {


            if(height[begin]<height[end]){
                 h = height[begin];
            }
            else{
                 h = height[end];
            } 
            int w = end-begin;
            int area2 = h*w;
            if(area1<area2){
                area1 = area2;
            }
            if(height[begin]<height[end]){
                begin++;
            }
            else{
                end--;
            }
        }
        System.out.println(area1);
    }

public static Integer rowRotation( Integer examWeek, String[][] matrix ){
     int AArow = 5;
     int Row = (AArow + examWeek ) % matrix.length;
     String[][] rotated = new String[matrix.length][];
     for(int i=0;i<matrix.length;i++){
            int newPos =(i+examWeek)%matrix.length;
            rotated[newPos]=matrix[i];
     }
     for(int row=0;row<rotated.length;row++){
            System.out.print("| ");
            for(int col=0;col<rotated[row].length;col++){
                System.out.print(rotated[row][col] + " | ");
            }
            System.out.println();
     }
        return Row;
    }

    public static Integer[][] compressMatrix( Integer[][] matrix ){
        int row = matrix.length;
        int col = matrix[0].length;


        Integer[][] compressed = new Integer[row/2][col/2];      
        for(int i=0;i<row;i+=2){
            for(int j=0;j<col;j+=2){
                int sum = matrix[i][j]+matrix[i][j+1]+matrix[i+1][j]+matrix[i+1][j+1];
                compressed[i/2][j/2] = sum;
            }
        }
        return compressed;
    }

public static void playGame( Integer[][] arena ){
       int rows = arena.length;


       int cols = arena[0].length;
       int points = 0;
       for(int i=0; i<rows; i++){
         for(int j=0; j<cols; j++){
            if(arena[i][j]%50==0 && arena[i][j]!=0){
                if(i>0 && arena[i-1][j]==2){
                    points+=2;
                }
                if(i<rows-1 && arena[i+1][j]==2) {
                    points+=2;
                }
                if(j>0 && arena[i][j-1]==2){
                    points+=2;
                }
                if(j<cols-1 && arena[i][j+1]==2){
                    points+=2;
                }
            }
         }
       }
    System.out.print("Points Gained: " + points + ". ");
    if(points>=10){
        System.out.println("Your team has survived the game.");
    }
    else{
        System.out.println("Your team is out.");
    }
}
