//Assignment Task 01
class AssgnTask1{



        public static void mostWater( Integer[] height ){
       int start = 0;
       int end = height.length-1;
       int finalArea = 0;
       while(start<end){
         int a;


         if(height[start]<height[end]){
          a=height[start];
         }
         else{
          a=height[end];
         }
         int b = end-start;
         int area = a*b;
         if(area>finalArea){
          finalArea=area;
         }
         if(height[start]<height[end]){
          start++;
         }
         else{
          end--;
         }
       }
       System.out.println(finalArea);
    }
}

class AssgnTask2{



    public static Integer rowRotation( Integer examWeek, String[][] matrix ){

      int r=matrix.length;
      int c=matrix[0].length;
      
      for(int i=0;i<examWeek-1;i++){


        String row[] = new String[c];
        for(int j=0;j<c;j++){
         row[j]=matrix[r-1][j];
        }
        
        for(int k=r-1;k>0;k--){
          for(int l=0;l<c;l++){
           matrix[k][l]=matrix[k-1][l];
          }
        }
        
        for(int m=0;m<c;m++){
         matrix[0][m]= row[m];
        }
      } 
      for(int n=0;n<r;n++){
       System.out.print("| ");
       for(int o=0;o<c;o++){
        System.out.print(matrix[n][o]+" | ");


       }
       System.out.println();
      }
      for(int p=0;p<r;p++){
        for(int q=0;q<c;q++){
          if(matrix[p][q]=="AA"){
           return p+1;
          }
        }
      }
        return null;
    }
}

class AssgnTask3{

      public static Integer[][] compressMatrix( Integer[][] matrix ){

        int r = matrix.length/2;
        int c = matrix[0].length/2;
        Integer new2Darr [][] = new Integer[r][c];
        for(int i=0;i<matrix.length;i+=2){
          for(int j=0;j<matrix[0].length;j+=2){
           int sum=0;
           sum+= matrix[i][j];
           sum+= matrix[i][j+1];
           sum+= matrix[i+1][j];
           sum+= matrix[i+1][j+1];
           new2Darr[i/2][j/2]=sum;
          }
        }
          return new2Darr;
    }

   }

class AssgnTask4{



    public static void playGame( Integer[][] arena ){

      int r=arena.length;
      int c=arena[0].length;
      int total=0;
      
      for(int i=0;i<r;i++){
        for(int j=0;j<c;j++){
          if(arena[i][j]%50==0 && arena[i][j]!=0){
            int count=0;
            if(i-1>=0 && arena[i-1][j]==2){
             count++;
            }
            if(i+1<r && arena[i+1][j]==2){
             count++;
            }
            if(j-1>=0 && arena[i][j-1]==2){
             count++;
            }
            if(j+1<c && arena[i][j+1]==2){
             count++;
            }
            total+=count*2;
          }
        }
      }


      if(total>=10){
       System.out.println("Points Gained: "+total+". Your team has survived the game.");
      }
      else{
       System.out.println("Points Gained: "+total+". Your team is out.");
      }
    }
}
