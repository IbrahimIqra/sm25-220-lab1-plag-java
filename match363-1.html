public static void mostWater( Integer[] height ){
        int vertical_lines = height.length;
        int bamer_pointer = 0;
        int daner_pointer = vertical_lines - 1;




        int shobcheye_beshi_area = 0;


        while(bamer_pointer < daner_pointer){


            int container_height;
            if (height[bamer_pointer] < height[daner_pointer]){
                container_height = height[bamer_pointer];
            }
            else{
                container_height = height[daner_pointer];
            }


            int container_width = daner_pointer - bamer_pointer;
            int container_area = container_height * container_width;


            if(container_area > shobcheye_beshi_area){
                shobcheye_beshi_area = container_area;
            }


            if (height[bamer_pointer] < height[daner_pointer]){
                bamer_pointer++;
            }
            else{
                daner_pointer--;
            }
        }
        
        System.out.println(shobcheye_beshi_area);
    }

public static Integer rowRotation( Integer examWeek, String[][] matrix ){
        int rows = matrix.length;
        int columns = matrix[0].length;
        int koibar_ghurbe = examWeek -1;
        
        int ghura = 0;
        while(ghura < koibar_ghurbe){
           int c = 0;
           while(c < columns){
              String current_last = matrix[rows-1][c];
              int r = rows - 1;
              while(r > 0){
                  matrix[r][c] = matrix[r - 1][c];
                  r--;
                }
                matrix[0][c] = current_last;
                c++;
            }
          ghura++;
        }




       Arr.print2D(matrix);


       int peyechi_r = 0;
       for(int khujo_r = 0; khujo_r < rows; khujo_r++){
           for(int khujo_c = 0; khujo_c < columns; khujo_c++){
               if(matrix[khujo_r][khujo_c].equals("AA")){
                   peyechi_r = khujo_r;
               }
           }
       }
       return peyechi_r + 1;
    }

public static Integer[][] compressMatrix( Integer[][] matrix ){
        int original_rows = matrix.length;
        int original_columns = matrix[0].length;


        int chotto_rows = original_rows / 2;
        int chotto_columns = original_columns / 2;
        Integer[][] chotto_array = new Integer[chotto_rows][chotto_columns];


        int temp_sum;
        int r = 0;
        while(r < original_rows){
            int c = 0;
            while(c < original_columns){


                temp_sum = matrix[r][c] + matrix[r][c+1] + matrix[r+1][c] + matrix[r+1][c+1];
                chotto_array [r / 2][c / 2] = temp_sum;
                c += 2;
            }
            r += 2;
        }


        return chotto_array;
        }

   public static void playGame( Integer[][] arena ){


        int rows = arena.length;
        int columns = arena[0].length;


        int points = 0;
        for(int r = 0; r < rows; r++){
            for(int c = 0; c < columns; c++){
                int check_position = arena[r][c];
                if(check_position != 0 && check_position % 50 == 0){


                    if(c-1 >= 0 && arena[r][c-1] == 2){
                        points += 2;
                    }
                    if(c+1 < columns && arena[r][c+1] == 2){
                        points += 2;
                    }
                    if(r+1 < rows && arena[r+1][c] == 2){
                        points += 2;
                    }
                    if(r-1 >= 0 && arena[r-1][c] == 2){
                        points += 2;
                    }
                }
            }
        }


         if(points >= 10){
                System.out.print("Points Gained: " + points + ". ");
                System.out.print("Your team has survived the game.");
            }
            else{
                System.out.print("Points Gained: " + points + ". ");
                System.out.print("Your team is out.");
            }
    }
