    public static void mostWater( Integer[] height ) {
    int left = 0;
    int right = height.length - 1;
    int maxArea = 0;

    while (left <right) {
        int width = right - left;
        int minHeight = Math.min(height[left], height[right]);
        int area = width * minHeight;


        maxArea = Math.max(maxArea, area);
        if (height[left] < height[right]) {
            left++;
        }
         else {
            right--;
        }
    }
    System.out.println(maxArea);
}

    public static Integer rowRotation(Integer examWeek, String[][] matrix) {
    int rows = matrix.length;
    int rotations = (examWeek - 1) % rows;

    for (int r = 0; r < rotations; r++) {
        String[] lastRow = matrix[rows - 1];
        for (int i = rows - 1; i > 0; i--) {
            matrix[i] = matrix[i - 1];
        }
        matrix[0] = lastRow;
    }
   Arr.print2D(matrix);

                break;

        }
    }

}

public static Integer[][] compressMatrix( Integer[][] matrix ) {
    Integer[][] compressed = new Integer[matrix.length/2][matrix[0].length/2];
    for (int i = 0; i < matrix.length; i += 2) {


        for (int j = 0; j <matrix[0].length; j += 2) {
            int sum = matrix[i][j] + matrix[i][j+1] + matrix[i+1][j] + matrix[i+1][j+1];
            compressed[i/2][j/2] = sum;
        }
    }
    return compressed;
}

    public static void playGame(Integer[][] arena) {
    int rows = arena.length;
    int cols = arena[0].length;
    int totalPoints = 0;
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            int value = arena[i][j];


            if (value > 0 && value % 50 == 0) {
                int playerPoints = 0;
                if (i - 1 >= 0 && arena[i - 1][j] == 2) {
                    playerPoints += 2;
                }
                if (i + 1 < rows && arena[i + 1][j] == 2) {
                    playerPoints += 2;
                }
                if (j - 1 >= 0 && arena[i][j - 1] == 2) {
                    playerPoints += 2;
                }
                if (j + 1 < cols && arena[i][j + 1] == 2) {
                    playerPoints += 2;
                }
                totalPoints += playerPoints;
            }
        }
    }

    System.out.print("Points Gained: " + totalPoints + ". ");
    if (totalPoints >= 10) {
        System.out.print("Your team has survived the game");
    } else {
        System.out.print("Your team is out");
    }
}
