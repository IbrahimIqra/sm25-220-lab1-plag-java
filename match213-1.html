public static void mostWater( Integer[] height ){
        int left=0;
        int right=height.length-1;
        int maxArea=0;
        int vheight=0;
        while(right>left)
        {int width = right- left;
            if (height[left]>height[right])
            {
                vheight=height[right];
            }
            else
            {
                 vheight=height[left];
            }


            int currArea= width* vheight;
            if(currArea>maxArea)
            {
                maxArea=currArea;
            }
            
            if(height[left]<height[right])
            {
                left++;
            }
            else
            {
                right--;
            }
        }
        System.out.println(maxArea);
    }

 public static Integer[][] compressMatrix( Integer[][] matrix ){
       Integer [][]arr= new Integer[2][2];
       int rowlen=matrix.length;
       int colLen=matrix[0].length;


       int boxes=(rowlen*colLen)/(arr.length*arr[0].length);//4
       for(int i=0;i<boxes;i+=2)
       {


        for(int j=0;j<boxes;j+=2)
        {

            int sum=matrix[i][j]+matrix[i+1][j]+matrix[i][j+1]+matrix[i+1][j+1];


            arr[i/2][j/2]=sum;}
       }


        return arr;
    }

 public static void playGame( Integer[][] arena ){
        int score=0;
        for (int row=0;row<arena.length;row++)
        {


            for (int col=0;col<arena[0].length;col++)
            {
                if (arena[row][col]%50==0 &&arena[row][col]!=0)
                {
                    if(col-1>=0 && arena[row][col-1]==2 )
                    {
                        score+=2;
                    }
                    
                    if(col+1<arena.length &&arena[row][col+1]==2)
                    {
                        score+=2;
                    }
                    if(row-1>=0 && arena[row-1][col]==2)
                    {
                        score+=2;
                    }


                    if(row+1<arena.length && arena[row+1][col]==2)
                    {
                        score+=2;
                    }
                }
            }
        }
        if(score>=10)
        {
            System.out.println("Points Gained: "+score+". Your team has survived the game.");
        }
        else
        {
            System.out.println("Points Gained: "+score+". Your team is out.");
        }
        
    }
