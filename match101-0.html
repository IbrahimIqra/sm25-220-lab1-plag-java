##Assignment Task 01: Container with Most Water

class AssgnTask1{

    public static void mostWater( Integer[] height ){

       int maxA = 0 , l = 0 , r = height.length - 1;
	    
         while (l<r) {
             maxA=Math.max(maxA, (r-l) * Math.min(height[l], height[r]));

            if (height[l] < height[r]) {
                l++;
            } else
            
            {
                
            r--;
            }
        }

      System.out.println(maxA);

    }

##Lab Task 03: Decryption Process

class LabTask3 {


    public static int[][] decryptMatrix(int[][] matrix) {
        int row = matrix.length;
        int col = matrix[0].length;
        int newr =row/2;
        int newc =col/2;
        int[][] compressed = new int[newr][newc];

         for (int i=0; i<row; i+=2) {
             for (int j=0; j<col; j +=2) {
                int sum=matrix[i][j] + matrix[i][j+1] + matrix[i+1][j] + matrix[i+1][j+1];


                compressed[i/2][j/2]=sum;
            }
        }
        
        return compressed;
    }
