    public static void mostWater( Integer[] height ){

        int left =0;
        int right = height.length-1;
        int max = 0;

        while (left<right){
            int small;
            if(height[left]<height[right]){
                small = height[left];
            }
            else{


                small= height[right];
            }

            int width = right - left;
            int water = small*width;

            if (water> max){
                max = water;
            }
            if (height[left]<height[right]){
                left=left+1;

            }
            else{
                right = right-1;
            }


        }
        System.err.println(max);
       

    }






//Assignment Task 02: Row Rotation Policy

    public static Integer rowRotation( Integer examWeek, String[][] matrix ){

        int Row = matrix.length;
        int Column = matrix[0].length;


        int rotate = examWeek -1;

        for(int r =0; r<rotate ; r++){
            String[] latestRow = matrix[Row-1];

            
            for(int i= Row-1; i>0; i--){
                matrix[i]=matrix[i-1];
            }
            matrix[0]=latestRow;
        }

        for (int r=0; r<Row ; r++){


            for (int c=0; c< Column ; c++){
                System.out.print(matrix[r][c] + " ");
            }
            System.out.println();
        }

        for (int r=0; r<Row ; r++){
            for (int c=0; c< Column ; c++){
            
                if (matrix[r][c].equals("AA")){
                    return r+1;
                }
            }
        }

       
        return -1;
    }







//Assignment Task 03: Matrix Compression
class AssgnTask3{

    //Complete this method so that it gives the Expected Output
    //YOU ONLY HAVE TO SUBMIT THIS METHOD, NO OTHER DRIVER CODE


    public static Integer[][] compressMatrix( Integer[][] matrix ){
        int row = matrix.length;
        int column = matrix[0].length;

        int R = row/2;
        int C = column/2;

        Integer [][] comp = new Integer[R][C];

        int newR = 0;
        for (int r=0; r< row ; r+=2){
            int newC =0;
            for (int c=0; c< column ; c+=2){
            int sum =0;

            sum += matrix[r][c];
            sum += matrix[r][c+1];
            sum += matrix[r+1][c];
            sum += matrix[r+1][c+1];
            comp[newR][newC] = sum;

            newC++;

            } 
            newR++;       


        }

        
        return comp;
    }
