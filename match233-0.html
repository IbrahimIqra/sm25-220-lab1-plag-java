public static Integer rowRotation( Integer examWeek, String[][] matrix ){
        Integer rowNum = 0;
        Integer count = 1;
        while(count < examWeek){
            for(Integer i = 0; i < matrix.length; i++){
                if(i+1 == matrix.length - 1){
                    matrix[0] = matrix[matrix.length-1];
                    count+=1;
                    rowNum+=1;
                    break;
                }
                matrix[i+1] = matrix[i];
                }
                
            }
            return rowNum;
        }

 public static Integer[][] compressMatrix( Integer[][] matrix ){
        Integer np[][] = new Integer[matrix.length/2][matrix[0].length/2];
        Integer k = 0;
        Integer l = 0;
    for(Integer i = 0; i<np.length; i++){
        l = 0;
        for(Integer j = 0; j<np[0].length; j++){
            if(k < matrix.length && l < matrix[0].length){


                np[i][j] = matrix[k][l]+matrix[k+1][l]+matrix[k][l+1]+matrix[k+1][l+1];
            }
            l += 2;
        }
        k += 2;
    }
        return np;
    }
