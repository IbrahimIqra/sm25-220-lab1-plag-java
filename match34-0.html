package Lab1.HomeTask1;

 public class HomeTask1 {
    public static void main(String[] args) {
        Integer[] height = {1, 8, 6, 2, 5, 4, 8, 3, 7};
        calculateMaxArea(height); }
 public static void calculateMaxArea( Integer[] height) {
      int maxFor = 0;
      int maxBack = 0;
      for(int i = 0; i < height.length; i++) {


          int area = 0;
          for(int j = i+1; j < height.length; j++) {
              if(height[i] <= height[j]) {
                  area = height[i] * (j-i); }
          }
              if(area > maxFor){
                  maxFor = area;}
          }

      for(int i = height.length - 1; i >= 0; i--){


          int area = 0;
          for(int j = i-1; j >= 0; j--) {
              if(height[i] <= height[j]) {
                  area = height[i] * (i-j); }
          }
      if(area > maxBack) {
       maxBack = area; }
    } 
    int finalval = Math.max(maxFor, maxBack);
     System.out.println("Maximum area: " + finalval);
 }
 }

package Lab1.HomeTask4;

public class HomeTask4 {
    public static void evaluateArena(Integer[][] grid) {


    int earnedPoints = 0;

    for (int x = 0; x < grid.length; x++) {
        for (int y = 0; y < grid[x].length; y++) {

            if (grid[x][y] != 0 && grid[x][y] % 50 == 0) {
                int nearbyTwos = 0;

                if (x > 0 && grid[x - 1][y] == 2) {
                    nearbyTwos++; }
                if (x < grid.length - 1 && grid[x + 1][y] == 2) {
                    nearbyTwos++; }
                if (y > 0 && grid[x][y - 1] == 2) {
                    nearbyTwos++;}
                if (y < grid[x].length - 1 && grid[x][y + 1] == 2) {
                    nearbyTwos++;}

                earnedPoints += nearbyTwos;
            }}}

    int total = earnedPoints * 2;
    if (total >= 10) {
        System.out.println("Points Gained: " + total + ". Your team has survived the game.");
    } else {
        System.out.println("Points Gained: " + total + ". Your team is out.");
} }}
