    public static Integer[][] compressMatrix( Integer[][] matrix ){
        int rows=matrix.length;
        int cols=matrix[0].length;
        Integer[][] arr = new Integer[rows/2][cols/2];
        //arr er counting dhorbe
        int ii=0;


        int jj=0;
        for (int r=0;r<rows;r+=2){  //r==i
          for(int c=0;c<cols;c+=2){
            arr[ii][jj] = 0;  //surute khali thakbe


            arr[ii][jj]+=matrix[r][c]+matrix[r][c+1]+matrix[r+1][c]+matrix[r+1][c+1];
            jj+=1;   
            // ii+=1;   hbe na eta
          }
          ii+=1;
          jj=0;
        }
        return arr;
    }

    public static void playGame( Integer[][] arena ){
        int point = 0;
        int rows=arena.length;
        //int cols=arena[0].length;
        for (int i=0; i<rows; i++) {


            for (int j=0; j<arena[i].length; j++) {
                //if (arena[i][j] % 200 == 0 && arena[i][j] != 0) {----200 diye hbe na cz 200 ta 50 er included hoye jbe
                if (arena[i][j] % 50 == 0 && arena[i][j] != 0) {
                    // Rough note from sir: up(i-1, j), down(i+1, j), left(i, j-1), right(i, j+1)
                  if (i>0 && arena[i-1][j] == 2){
                    point += 2;
                  }
                  if (i<rows-1 && arena[i+1][j] == 2) {
                    point += 2;
                  } 
                  if (j>0 && arena[i][j-1] == 2){
                    point += 2;
                  } 
                  if (j<arena[i].length - 1 && arena[i][j+1] == 2){
                    point += 2;
                  }
                }
            }
        }
        if (point < 10) {
            System.out.println("Points Gained: "+point+". Your team is out.");
        } 
        else{
            System.out.println("Points Gained: "+point+". Your team has survived the game.");
        }
    }
